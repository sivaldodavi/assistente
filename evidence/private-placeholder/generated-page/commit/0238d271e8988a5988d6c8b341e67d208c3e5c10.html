+# Roteiro
+
+Site estático em Hugo para orientar a elaboração de cases e papers e gerar
+documentos em Typst no navegador.
+
+## Executar
+
+```sh
+npm install
+npm run dev
+```
+
+Gerar a versão estática:
+
+```sh
+npm run build
+```
+
+O resultado fica em `public/` e pode ser servido por qualquer hospedagem estática.
+Hugo monta o site e agrupa o pequeno frontend JavaScript e as dependências
+WebAssembly do Typst. Não há backend: o conteúdo do trabalho fica no navegador,
+salvo localmente, e não é enviado a um servidor. O motor pode carregar recursos
+tipográficos pela rede.
+
+## Escopo atual
+
+- Fluxos separados para case e paper, com etapas e dicas contextuais.
+- Rascunhos salvos localmente no navegador.
+- Prévia SVG e exportação PDF executadas no navegador com Typst.
+- Sem conta, backend ou sincronização entre dispositivos.
+
+Este protótipo organiza a escrita, mas não valida teses jurídicas, fontes, citações
+ou conformidade institucional. Confira as orientações do professor e as normas
+aplicáveis antes de entregar o documento.

+function escapeHtml(value) {
+  return String(value).replace(/[&<>"']/g, (char) => ({
+    "&": "&amp;",
+    "<": "&lt;",
+    ">": "&gt;",
+    '"': "&quot;",
+    "'": "&#39;",
+  })[char]);

+function documentMode() {
+  if (currentType === "case") return "case";
+  return draft.paperKind === "project" ? "project" : "paper";

+function isRequired(field) {
+  if (activePreset?.omitRequired?.includes(field.id)) return false;
+  if (field.required) return true;
+  const mode = documentMode();
+  if (field.requiredFor?.includes(mode)) return true;
+  if (field.requiredFor?.includes("case-closed")) {
+    return mode === "case" && draft.caseType === "Fechado";
+  }
+  if (field.requiredFor?.includes("facsure")) {
+    return draft.profile === "facsure";
+  }
+  return false;

+function renderField(field) {
+  const value = draft[field.id] ?? "";
+  const id = `field-${field.id}`;
+  const omittedByPreset = activePreset?.omitRequired?.includes(field.id);
+  const helpText = [
+    field.hint,
+    omittedByPreset ? "Ausente no PDF de referência; neste preenchimento temporário você pode deixar em branco." : "",
+  ].filter(Boolean).join(" ");
+  const hint = helpText ? `<p class="field-hint" id="${id}-hint">${escapeHtml(helpText)}</p>` : "";
+  const required = isRequired(field);
+  const projectSupport = !required
+    && documentMode() === "paper"
+    && draft.paperKind !== "project"
+    && field.requiredFor?.includes("project");
+  const badgeText = required ? "Obrigatório para este PDF" : omittedByPreset ? "Ausente no original" : projectSupport ? "Apoio · não incluído" : "Opcional / apoio";
+  const badge = `<span class="field-state ${required ? "is-required" : "is-optional"}">${badgeText}</span>`;
+  let input;
+  if (field.type === "textarea") {
+    input = `<textarea id="${id}" name="${field.id}" data-size="${field.size ?? ""}" ${required ? "required" : ""} aria-describedby="${field.hint ? `${id}-hint` : ""}"></textarea>`;
+  } else if (field.type === "select") {
+    input = `<select id="${id}" name="${field.id}" aria-describedby="${field.hint ? `${id}-hint` : ""}">
+      ${field.options.map((item) => {
+        const option = Array.isArray(item) ? item[0] : item;
+        const label = Array.isArray(item) ? item[1] : item;
+        return `<option value="${escapeHtml(option)}" ${value === option ? "selected" : ""}>${escapeHtml(label)}</option>`;
+      }).join("")}
+    </select>`;
+  } else {
+    input = `<input id="${id}" name="${field.id}" type="text" value="${escapeHtml(value)}" placeholder="${escapeHtml(field.placeholder ?? "")}" ${required ? "required" : ""} aria-describedby="${field.hint ? `${id}-hint` : ""}" />`;
+  }
+  return `<div class="field">
+    <div class="field-label-row"><label for="${id}">${escapeHtml(field.label)}${required ? " *" : ""}</label>${badge}</div>
+    ${input}
+    ${hint}
+  </div>`;

+function escapeTypstString(value) {

+function typstParagraphs(value) {
+  return String(value ?? "")
+    .split(/\n\s*\n/)
+    .map((paragraph) => paragraph.trim())
+    .filter(Boolean)
+    .map((paragraph) => `#par(justify: true)[#text(${escapeTypstString(paragraph.replace(/\s*\n\s*/g, " "))})]`)
+    .join("\n\n");

+function typstSection(title, content) {
+  const body = typstParagraphs(content);
+  if (!body) return "";
+  return `= #text(${escapeTypstString(title)})\n\n${body}`;

+function typstSubsection(level, title, content) {
+  const body = typstParagraphs(content);
+  return `${"=".repeat(level)} #text(${escapeTypstString(title)})${body ? `\n\n${body}` : ""}`;

+async function getTypst() {
+  if (!typstModule) {
+    const [{ $typst }, compilerWasm, rendererWasm] = await Promise.all([
+      import("@myriaddreamin/typst.ts/contrib/all-in-one-lite"),
+      fetch(document.body.dataset.compilerWasm),
+      fetch(document.body.dataset.rendererWasm),
+    ]);
+    if (!compilerWasm.ok || !rendererWasm.ok) {
+      throw new Error("Não foi possível carregar os recursos WebAssembly do Typst.");
+    }
+    const compilerModule = await compilerWasm.arrayBuffer();
+    const rendererModule = await rendererWasm.arrayBuffer();
+    $typst.setCompilerInitOptions({
+      getWrapper: () => import("@myriaddreamin/typst-ts-web-compiler"),
+      getModule: () => compilerModule,
+    });
+    $typst.setRendererInitOptions({
+      getWrapper: () => import("@myriaddreamin/typst-ts-renderer"),
+      getModule: () => rendererModule,
+    });
+    typstModule = $typst;
+  }
+  return typstModule;
+}

+async function generatePreview() {
+  const previewButton = document.querySelector("#preview-button");
+  const pdfButton = document.querySelector("#pdf-button");
+  const previewBody = document.querySelector("#preview-body");
+  const version = ++compileVersion;
+  const source = makeDocumentSource();

+  previewButton.disabled = true;
+  previewButton.textContent = "Compilando…";

+    const typst = await getTypst();
+    const vectorData = await typst.vector({ mainContent: source });
+    if (!vectorData) throw new Error("O compilador não retornou uma prévia.");
+    const svg = await typst.svg({ vectorData });
+    if (!svg) throw new Error("O renderizador não retornou páginas.");
+    if (version !== compileVersion) return;
+    if (previewUrl) URL.revokeObjectURL(previewUrl);
+    previewUrl = URL.createObjectURL(new Blob([svg], { type: "image/svg+xml" }));
+    previewBody.innerHTML = "";
+    const pages = document.createElement("div");
+    pages.className = "preview-pages";
+    const image = document.createElement("img");
+    image.src = previewUrl;
+    image.alt = `Prévia do ${workflows[currentType].name}`;
+    pages.append(image);
+    previewBody.append(pages);
+    generatedSource = source;
+    const missing = missingRequired();
+    pdfButton.disabled = missing.length > 0;
+    setPreviewStatus(missing.length
+      ? `Prévia parcial pronta. Preencha os ${missing.length} campos obrigatórios para liberar o PDF.`
+      : "Prévia pronta. O PDF será compilado quando você solicitar o download.");
+  } catch (error) {
+    if (version !== compileVersion) return;
+    previewBody.innerHTML = '<p class="preview-empty">Não foi possível gerar a prévia. Revise os campos e tente novamente.</p>';
+    setPreviewStatus(`Falha na compilação: ${error.message}`, "error");
+  } finally {
+    if (version === compileVersion) {
+      previewButton.disabled = false;
+      previewButton.textContent = "Atualizar";
+    }
+  }

+async function downloadPdf() {
+  const button = document.querySelector("#pdf-button");
+  if (!generatedSource) return;
+  const missing = missingRequired();
+  if (missing.length) {
+    setPreviewStatus(`PDF bloqueado: preencha os ${missing.length} campos obrigatórios indicados no roteiro.`, "error");
+    updateCompleteness();
+    return;
+  }
+  button.disabled = true;
+  setPreviewStatus("Gerando PDF no navegador…");
+  try {
+    const typst = await getTypst();
+    generatedPdf = await typst.pdf({ mainContent: generatedSource });
+    if (!generatedPdf) throw new Error("O compilador não retornou um PDF.");
+    const url = URL.createObjectURL(new Blob([generatedPdf], { type: "application/pdf" }));
+    const anchor = document.createElement("a");
+    const filename = (draft.title || workflows[currentType].short)
+      .normalize("NFD")
