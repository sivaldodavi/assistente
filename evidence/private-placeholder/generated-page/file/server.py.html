    346 def validated_jev_summary(value):
    347     if not isinstance(value, dict):
    348         raise ApiError(400, "invalid_jev_result", "Resultado de análise inválido.")
    349     score = value.get("score")
    350     confidence = value.get("confidence")
    351     probabilities = value.get("probabilities")
    352     if (
    353         isinstance(score, bool)
    354         or not isinstance(score, (int, float))
    355         or not math.isfinite(score)
    356         or not 0 <= score <= 3
    357         or isinstance(confidence, bool)
    358         or not isinstance(confidence, (int, float))
    359         or not math.isfinite(confidence)
    360         or not 0 <= confidence <= 1
    361         or not isinstance(probabilities, list)
    362         or len(probabilities) != 4
    363     ):
    364         raise ApiError(400, "invalid_jev_result", "Resultado de análise inválido.")
    365     if any(
    366         isinstance(probability, bool)
    367         or not isinstance(probability, (int, float))
    368         or not math.isfinite(probability)
    369         or not 0 <= probability <= 1
    370         for probability in probabilities
    371     ) or not 0.97 <= sum(probabilities) <= 1.03:
    372         raise ApiError(400, "invalid_jev_result", "Resultado de análise inválido.")
    373     return {
    374         "score": score,
    375         "confidence": confidence,
    376         "probabilities": probabilities,
    377     }
