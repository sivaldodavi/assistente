    680 function normalizeProject(raw, legacy = false) {
    681   const project = emptyProject(raw?.projectType === "paper" || raw?.type === "paper" ? "paper" : "case");
    682   const oldFields = raw?.fields && typeof raw.fields === "object" ? raw.fields : raw;
    683   for (const key of Object.keys(project.fields)) {
    684     if (typeof oldFields?.[key] === "string") project.fields[key] = oldFields[key];
    685   }
    686   project.fields.developmentSections = Array.isArray(oldFields?.developmentSections)
    687     ? oldFields.developmentSections
    688       .filter((section) => section && typeof section === "object")
    689       .map((section) => ({
    690         id: typeof section.id === "string" ? section.id : newId(),
    691         title: typeof section.title === "string" ? section.title : "",
    692         content: typeof section.content === "string" ? section.content : "",
    693       }))
    694     : [1, 2, 3]
    695       .map((index) => ({
    696         id: newId(),
    697         title: typeof oldFields?.[`development${index}Title`] === "string"
    698           ? oldFields[`development${index}Title`]
    699           : "",
    700         content: typeof oldFields?.[`development${index}`] === "string"
    701           ? oldFields[`development${index}`]
    702           : "",
    703       }))
    704       .filter((section) => section.title || section.content);
    705   if (!project.fields.developmentSections.length) {
    706     project.fields.developmentSections = [{ id: newId(), title: "", content: "" }];
    707   }
    708   project.sources = Array.isArray(raw?.sources)
    709     ? raw.sources.map((source) => ({
    710       id: String(source.id || newId()),
    711       category: source.category || "",
    712       referenceType: referenceTypes.some(([type]) => type === source.referenceType) ? source.referenceType : "manual",
    713       details: source.details && typeof source.details === "object"
    714         ? Object.fromEntries(Object.entries(source.details).filter(([, value]) => typeof value === "string"))
    715         : {},
    716       reference: String(source.reference || ""),
    717       theses: Array.isArray(source.theses) ? source.theses.filter((thesis) => ["A", "B"].includes(thesis)) : [],
    718       cited: Boolean(source.cited),
    719       library: Boolean(source.library),
    720     }))
    721     : [];
    722   if (legacy && !project.sources.length) project.sources = normalizedReferences(oldFields?.references);
    723   project.criteria = Array.isArray(raw?.criteria)
    724     ? raw.criteria.map((item) => ({
    725       id: String(item.id || newId()),
    726       term: String(item.term || ""),
    727       definition: String(item.definition || ""),
    728       basis: String(item.basis || ""),
    729     }))
    730     : [];
    731   if (legacy && !project.criteria.length && typeof oldFields?.criteria === "string" && oldFields.criteria.trim()) {
    732     project.criteria = [{
    733       id: newId(),
    734       term: "Conteúdo do rascunho anterior",
    735       definition: oldFields.criteria.trim(),
    736       basis: "",
    737     }];
    738   }
    739   project.checks = raw?.checks && typeof raw.checks === "object" ? { ...raw.checks } : {};
    740   project.suggestedChecks = raw?.suggestedChecks && typeof raw.suggestedChecks === "object"
    741     ? Object.fromEntries(Object.entries(raw.suggestedChecks)
    742       .filter(([key, value]) => /^[a-z0-9-]{1,80}$/.test(key) && (value === true || value === "suggested" || value === "stale"))
    743       .map(([key, value]) => [key, value === true ? "suggested" : value]))
    744     : {};
    745   project.currentStep = Number.isInteger(raw?.currentStep)
    746     ? Math.max(0, Math.min(getStages(project).length - 1, raw.currentStep))
    747     : 0;
    748   project.finalized = Boolean(raw?.finalized);
    749   return project;
    750 }

    804 function parseRouteHash() {
    805   let segments;
    806   try {
    807     segments = window.location.hash.replace(/^#\/?/, "").split("/").map(decodeURIComponent);
    808   } catch {
    809     return null;
    810   }
    811   if (segments.length === 1 && ["home", "login"].includes(segments[0])) {
    812     return { view: segments[0] };
    813   }
    814   if (segments[0] === "project" && segments.length === 4 && segments[2] === "step") {
    815     const step = Number(segments[3]);
    816     if (segments[1] && Number.isInteger(step) && step >= 0) {
    817       return { view: "project", id: segments[1], step };
    818     }
    819   }
    820   if (segments[0] === "model" && segments.length === 4 && segments[2] === "step") {
    821     const step = Number(segments[3]);
    822     if (segments[1] && Number.isInteger(step) && step >= 0) {
    823       return { view: "model", id: segments[1], step };
    824     }
    825   }
    826   if (segments[0] === "document" && segments.length === 2 && segments[1]) {
    827     return { view: "document", id: segments[1] };
    828   }
    829   return null;
    830 }

   2052 function formatReference(source) {
   2053   const value = (key) => String(source.details[key] ?? "").trim();
   2054   const sequence = (...parts) => parts.map((part) => String(part ?? "").trim()).filter(Boolean).join(" ");
   2055   const punctuation = (text, mark = ".") => {
   2056     const clean = String(text ?? "").trim();
   2057     if (!clean) return "";
   2058     if (mark === "." && /[.!?]$/.test(clean)) return clean;
   2059     return `${clean.replace(/[.,;:]+$/, "")}${mark}`;
   2060   };
   2061   const online = () => sequence(value("url") && `Disponível em: ${value("url")}.`, value("accessDate") && `Acesso em: ${value("accessDate")}.`);
   2062   switch (source.referenceType) {
   2063     case "book":
   2064       return sequence(
   2065         punctuation(value("authors")),
   2066         punctuation(value("title")),
   2067         punctuation(value("edition")),
   2068         sequence(punctuation(value("city"), ":"), punctuation(value("publisher"), ","), punctuation(value("year"))),
   2069       );
   2070     case "article":
   2071       return sequence(
   2072         punctuation(value("authors")),
   2073         punctuation(value("title")),
   2074         sequence(
   2075           punctuation(value("journal"), ","),
   2076           punctuation(value("city"), ","),
   2077           value("volume") && `v. ${value("volume")},`,
   2078           value("issue") && `n. ${value("issue")},`,
   2079           value("pages") && `p. ${value("pages")},`,
   2080           punctuation(value("monthYear")),
   2081         ),
   2082       );
   2083     case "online":
   2084       return sequence(
   2085         punctuation(value("authors")),
   2086         punctuation(value("title")),
   2087         sequence(punctuation(value("site"), ","), punctuation(value("year"))),
   2088         online(),
   2089       );
   2090     case "legislation":
   2091       return sequence(
   2092         value("jurisdiction") && `${value("jurisdiction").toLocaleUpperCase("pt-BR")}.`,
   2093         punctuation(`${value("title")}${value("date") ? `, de ${value("date")}` : ""}`),
   2094         punctuation(value("ementa")),
   2095         punctuation(value("publication")),
   2096         online(),
   2097       );
   2098     case "case-law":
   2099       return sequence(
   2100         value("jurisdiction") && `${value("jurisdiction").toLocaleUpperCase("pt-BR")}.`,
   2101         punctuation(value("tribunal")),
   2102         punctuation(value("party")),
   2103         sequence(value("processType"), value("processNumber") && `nº ${value("processNumber")}.`),
   2104         value("rapporteur") && punctuation(`Relator: ${value("rapporteur")}`),
   2105         punctuation(value("courtBody")),
   2106         value("judgedAt") && punctuation(`Julgado em: ${value("judgedAt")}`),
   2107         punctuation(value("publication")),
   2108         online(),
   2109       );
   2110     default:
   2111       return source.reference;
   2112   }
   2113 }
