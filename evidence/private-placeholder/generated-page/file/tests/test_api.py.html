     68     def request(self, method, path, payload=None, cookie=None):
     69         connection = http.client.HTTPConnection(*self.address)
     70         body = json.dumps(payload).encode() if payload is not None else None
     71         headers = {}
     72         if body is not None:
     73             headers["Content-Type"] = "application/json"
     74         if cookie:
     75             headers["Cookie"] = cookie
     76         connection.request(method, path, body, headers)
     77         response = connection.getresponse()
     78         data = response.read()
     79         try:
     80             result = json.loads(data) if data else None
     81         except json.JSONDecodeError:
     82             result = data
     83         set_cookie = response.getheader("Set-Cookie")
     84         status = response.status
     85         connection.close()
     86         return status, result, set_cookie
