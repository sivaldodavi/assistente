      1 #!/bin/bash
      2 # Uso: deploy-site <website|direito|teologia|assistente-academico> [ref]
      3 # Instalar em /opt/servidor/deploy-site. Push Git não dispara deploy.
      4 set -euo pipefail
      5 cd /
      6 unset GIT_DIR GIT_WORK_TREE
      7 SERVIDOR_DIR="/opt/servidor"
      8 REPOS_DIR="/home/git/repos"
      9 REPO="${1:-}"; REF="${2:-main}"
     10 case "$REPO" in
     11   website) SITE_DIR="pessoal"; SERVICE="pessoal" ;;
     12   direito) SITE_DIR="direito"; SERVICE="direito" ;;
     13   teologia) SITE_DIR="teologia"; SERVICE="teologia" ;;
     14   assistente-academico) SITE_DIR="assistente-academico"; SERVICE="assistente-academico" ;;
     15   *) echo "Uso: deploy-site <website|direito|teologia|assistente-academico> [ref]"; exit 1 ;;
     16 esac
     17 GIT_REPO="$REPOS_DIR/$REPO.git"
     18 DEST="$SERVIDOR_DIR/sites/$SITE_DIR"
     19 [ -d "$GIT_REPO" ] || { echo "ERRO: repositório não existe"; exit 1; }
     20 [ -d "$DEST" ] || { echo "ERRO: diretório de publicação não existe"; exit 1; }
     21 exec 9>"/tmp/deploy-site-$SITE_DIR.lock"
     22 flock 9
     23 STAGE="$(mktemp -d "$SERVIDOR_DIR/sites/.deploy-$SITE_DIR.XXXXXX")"
     24 trap 'rm -rf "$STAGE"' EXIT
     25 git --git-dir="$GIT_REPO" archive "$REF" | tar -x -C "$STAGE"
     26 if [ "$REPO" = "assistente-academico" ]; then
     27   for file in Dockerfile requirements.txt wsgi.py app/__init__.py app/store/schema.sql; do
     28     [ -s "$STAGE/$file" ] || { echo "ERRO: falta $file na aplicação Python; deploy cancelado"; exit 1; }
     29   done
     30   [ -d "$STAGE/templates" ] && [ -d "$STAGE/static" ] || exit 1
     31 elif [ ! -d "$STAGE/content" ] || [ -z "$(ls -A "$STAGE/content")" ]; then
     32   echo "ERRO: o commit não tem content/; deploy cancelado"
     33   exit 1
     34 fi
     35 if docker compose version >/dev/null 2>&1; then DC=(docker compose); else DC=(docker-compose); fi
     36 COMPOSE_ARGS=(--project-directory "$SERVIDOR_DIR" -f "$SERVIDOR_DIR/docker-compose.yml")
     37 if [ "$REPO" = "assistente-academico" ]; then
     38   COMPOSE_ARGS+=(-f "$SERVIDOR_DIR/docker-compose.production.yml")
     39 fi
     40 "${DC[@]}" "${COMPOSE_ARGS[@]}" config --quiet
     41 # Só código: banco, secrets e catálogo ficam fora de sites/.
     42 find "$DEST" -mindepth 1 -maxdepth 1 -exec rm -rf {} +
     43 cp -a "$STAGE"/. "$DEST"/
     44 echo "==> [$REPO] construindo e publicando $REF"
     45 "${DC[@]}" "${COMPOSE_ARGS[@]}" up -d --build --no-deps "$SERVICE"
     46 echo "==> [$REPO] container atualizado; verificar healthcheck e domínio."
