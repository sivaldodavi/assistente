     25 def checks() -> list[str]:
     26     issues = []
     27     try:
     28         request = Request(SITE_URL + "/healthz", method="GET")
     29         with urlopen(request, timeout=10) as response:
     30             if response.status != 200:

     33         issues.append("site_http")

     35     health, code = command("docker", "inspect", "--format", "{{.State.Health.Status}}", CONTAINER)

     37         issues.append("container_health")

     39     try:
     40         connection = sqlite3.connect(f"file:{DATABASE}?mode=ro", uri=True, timeout=3)
     41         if connection.execute("PRAGMA integrity_check").fetchone()[0] != "ok":
     42             issues.append("database_integrity")
     43         connection.close()
     44     except (OSError, sqlite3.Error):
     45         issues.append("database")

     47     usage = shutil.disk_usage("/opt/servidor")
