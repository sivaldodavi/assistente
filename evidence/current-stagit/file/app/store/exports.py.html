     16 def export_account_data(account_id: str) -> dict[str, Any]:
     17     """Exporta todos os dados da conta para LGPD/GDPR."""
     18     from .accounts import get_account
     19 
     20     account = get_account(account_id)
     21     if not account:
     22         raise ValueError("Conta não encontrada")
     23 
     24     # Projetos completos
     25     projects_data = []
     26     for project in list_projects(account_id):
     27         full_project = get_project(account_id, project["id"])
     28         if full_project:
     29             projects_data.append(full_project)
     30 
     31     # Revisões
     32     revisions = []
     33     for project in list_projects(account_id):
     34         rev_list = list_revisions(project["id"], account_id)
     35         for rev in rev_list:
     36             revisions.append({
     37                 "project_id": project["id"],
     38                 "snapshot_id": rev["snapshot_id"],
     39                 "revision": rev["revision"],
     40                 "title": rev["title"],
     41                 "project_type": rev["project_type"],
     42                 "payload": rev["payload"],
     43                 "created_at": rev["created_at"],
     44             })
     45 
     46     # Comentários recebidos (como proprietário dos projetos)
     47     comments = []
     48     for project in list_projects(account_id):
     49         proj_comments = list_comments(project["id"])
     50         for c in proj_comments:
     51             c = dict(c)
     52             c["project_id"] = project["id"]
     53             comments.append(c)
     54 
     55     # Fontes da biblioteca
     56     sources_data = list_sources(account_id)
     57 
     58     # Uso de IA - reservas
     59     ai_usage = get_db().execute(
     60         """SELECT request_id, action, units, status, created_at, updated_at
     61            FROM ai_usage_reservations WHERE account_id = ? ORDER BY created_at DESC""",
     62         (account_id,),
     63     ).fetchall()
     64 
     65     # Uso de IA - provedores
     66     provider_usage = get_db().execute(
     67         """SELECT request_id, provider, operation, model, input_tokens, output_tokens, status, created_at
     68            FROM provider_usage WHERE account_id = ? ORDER BY created_at DESC""",
     69         (account_id,),
     70     ).fetchall()
     71 
     72     # Grants
     73     grants = get_db().execute(
     74         """SELECT grant_id, plan_code, project_credits, project_consumed,
     75                   ai_analysis_credits, ai_analysis_consumed,
     76                   deepseek_explanation_credits, deepseek_explanation_consumed,
     77                   created_at, expires_at
     78            FROM account_grants WHERE account_id = ? ORDER BY created_at DESC""",
     79         (account_id,),
     80     ).fetchall()
     81 
     82     # Sessões
     83     sessions = get_db().execute(
     84         "SELECT id, created_at, expires_at FROM sessions WHERE account_id = ? ORDER BY created_at DESC",
     85         (account_id,),
     86     ).fetchall()
     87 
     88     # Pedido de exclusão
     89     deletion_request = get_db().execute(
     90         "SELECT status, created_at, resolved_at FROM deletion_requests WHERE account_id = ?",
     91         (account_id,),
     92     ).fetchone()
     93 
     94     # Preferências de compartilhamento
     95     sharing_prefs = get_db().execute(
     96         """SELECT sharing_preferences_version, sharing_allow_admin_view,
     97                   sharing_allow_admin_comments, sharing_inherit_default
     98            FROM accounts WHERE id = ?""",
     99         (account_id,),
    100     ).fetchone()
    101 
    102     return {
    103         "account": {
    104             **account,
    105             "sharing_preferences": dict(sharing_prefs) if sharing_prefs else {},
    106             "deletion_request": dict(deletion_request) if deletion_request else None,
    107         },
    108         "projects": projects_data,
    109         "revisions": revisions,
    110         "comments": comments,
    111         "sources": sources_data,
    112         "ai_usage_reservations": [dict(r) for r in ai_usage],
    113         "provider_usage": [dict(r) for r in provider_usage],
    114         "grants": [dict(r) for r in grants],
    115         "sessions": [dict(r) for r in sessions],
    116         "exported_at": int(time.time()),
    117     }
