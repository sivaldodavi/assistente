     40 def create_project(account_id, project_type):
     41     project_id = str(uuid4())
     42     now = int(time.time())
     43     from .accounts import get_account
     44     database = get_db()
     45     database.execute("BEGIN IMMEDIATE")
     46     account = get_account(account_id)
     47     if account is None:
     48         database.rollback()
     49         raise ProjectQuotaError("Conta não encontrada.")
     50     if not account["unlimited"] and int(account["project_remaining"] or 0) <= 0:
     51         database.rollback()
     52         raise ProjectQuotaError("A conta não possui crédito de projeto disponível.")
     53     profile = account.get("profile", {})
     54     fields = {"title": "", "authors": account.get("name", ""), **{key: profile.get(key, "") for key in ("institution", "course", "city")}}
     55     payload = {"version": 1, "fields": fields, "source_ids": [], "criteria": [], "deliveries": {}, "schedule": {"report_date": "", "debate_date": "", "dissertation_date": "", "final_date": "", "tasks": []}}
     56     database.execute(
     57         """INSERT INTO projects
     58            (id, account_id, project_type, title, revision, payload, created_at, updated_at)
     59            VALUES (?, ?, ?, '', 1, ?, ?, ?)""",
     60         (project_id, account_id, project_type, json.dumps(payload, ensure_ascii=False), now, now),
     61     )
     62     database.execute("UPDATE accounts SET project_used = project_used + 1 WHERE id = ?", (account_id,))
     63     database.commit()
     64     return project_id

     68     """Exclui um projeto do proprietário e reduz o contador de projetos."""

    120 def save_project(account_id, project_id, expected_revision, payload):
    121     database = get_db()
    122     now = int(time.time())
    123     title = str(payload.get("fields", {}).get("title") or "")[:240]
    124     encoded = json.dumps(payload, ensure_ascii=False, separators=(",", ":"))
    125     database.execute("BEGIN IMMEDIATE")
    126     row = database.execute(
    127         "SELECT revision FROM projects WHERE id = ? AND account_id = ?",
    128         (project_id, account_id),
    129     ).fetchone()
    130     if row is None:
    131         database.rollback()
    132         return None
    133     if int(expected_revision) != int(row["revision"]):
    134         database.rollback()
    135         raise ProjectConflict(row["revision"])
    136     next_revision = int(row["revision"]) + 1
    137     database.execute(
    138         """UPDATE projects SET title = ?, revision = ?, payload = ?, updated_at = ?
    139            WHERE id = ? AND account_id = ?""",
    140         (title, next_revision, encoded, now, project_id, account_id),
    141     )
    142     edit_date = _edit_date(now)
    143     database.execute(
    144         """INSERT INTO project_edit_days (account_id, project_id, edit_date, edit_count, updated_at)
    145            VALUES (?, ?, ?, 1, ?)
    146            ON CONFLICT(project_id, edit_date) DO UPDATE SET
    147              edit_count = project_edit_days.edit_count + 1,
    148              updated_at = excluded.updated_at""",
    149         (account_id, project_id, edit_date, now),
    150     )
    151     cutoff = (datetime.fromtimestamp(now, ZoneInfo("America/Sao_Paulo")) - timedelta(days=365)).date().isoformat() if ZoneInfo else ""
    152     if cutoff:
    153         database.execute("DELETE FROM project_edit_days WHERE account_id = ? AND edit_date < ?", (account_id, cutoff))
    154     database.commit()
    155     return next_revision

    169 def switch_paper_mode(account_id, project_id, project_type, expected_revision):
    170     """Alterna Projeto/Paper Final mantendo identidade, fontes e planejamento."""
    171     if project_type not in {"paper_project", "paper_final"}:
    172         raise ValueError("Roteiro de Paper inválido.")
    173     db = get_db()
    174     db.execute("BEGIN IMMEDIATE")
    175     row = db.execute("SELECT project_type, revision FROM projects WHERE id = ? AND account_id = ?", (project_id, account_id)).fetchone()
    176     if row is None or row["project_type"] not in {"paper_project", "paper_final"}:
    177         db.rollback()
    178         return None
    179     if int(expected_revision) != row["revision"]:
    180         db.rollback()
    181         raise ProjectConflict(row["revision"])
    182     revision = row["revision"] + 1
    183     db.execute("UPDATE projects SET project_type = ?, revision = ?, updated_at = ? WHERE id = ?", (project_type, revision, int(time.time()), project_id))
    184     db.commit()
    185     return revision
