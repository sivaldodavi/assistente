      9 def _migration_lock(database_path: Path):
     10     """File-based lock to prevent concurrent migrations in multi-worker environments."""
     11     lock_file = database_path.with_suffix(".migration.lock")
     12     lock_fd = os.open(lock_file, os.O_CREAT | os.O_RDWR)
     13     try:
     14         fcntl.flock(lock_fd, fcntl.LOCK_EX | fcntl.LOCK_NB)
     15         return lock_fd
     16     except BlockingIOError:
     17         os.close(lock_fd)
     18         return None

     46 def _remove_automatic_messaging(database):
     47     columns = {row["name"] for row in database.execute("PRAGMA table_info(accounts)")}
     48     outbox = database.execute(
     49         "SELECT 1 FROM sqlite_master WHERE type = 'table' AND name = 'notification_outbox'"
     50     ).fetchone()
     51     old_columns = {"whatsapp_phone", "whatsapp_opt_in", "whatsapp_opt_in_at", "whatsapp_opt_out_at"} & columns
     52     if not old_columns and outbox is None:
     53         return
     54     database.execute("PRAGMA foreign_keys = OFF")
     55     try:
     56         database.execute("BEGIN")
     57         if outbox is not None:
     58             database.execute("DROP TABLE notification_outbox")
     59         for name in sorted(old_columns):
     60             database.execute(f"ALTER TABLE accounts DROP COLUMN {name}")
     61         database.execute("UPDATE app_meta SET value = '11' WHERE key = 'schema_version'")
     62         database.commit()
     63     except Exception:
     64         database.rollback()
     65         raise
     66     finally:
     67         database.execute("PRAGMA foreign_keys = ON")
