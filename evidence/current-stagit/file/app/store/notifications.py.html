     24 def enqueue_notification(
     25     kind: str,
     26     text: str,
     27     account_id: str | None = None,
     28     project_id: str | None = None,
     29     destination: str | None = None,
     30     idempotency_key: str | None = None,
     31     next_attempt_at: int | None = None,
     32 ) -> int | None:
     33     """Adiciona notificação à fila de saída."""
     34     if kind not in NOTIFICATION_KINDS:
     35         raise ValueError(f"Tipo de notificação inválido: {kind}")
     36     if not text or not text.strip():
     37         raise ValueError("Texto da notificação não pode ser vazio")
     38 
     39     if idempotency_key is None:
     40         idempotency_key = f"{kind}:{account_id or 'global'}:{project_id or 'none'}:{uuid4().hex[:16]}"
     41 
     42     now = int(time.time())
     43     if next_attempt_at is None:
     44         next_attempt_at = now
     45 
     46     db = get_db()
     47     db.execute("BEGIN IMMEDIATE")
     48     try:
     49         cursor = db.execute(
     50             """INSERT INTO notification_outbox
     51                (account_id, destination, kind, project_id, idempotency_key, text, status, attempts, next_attempt_at, created_at)
     52                VALUES (?, ?, ?, ?, ?, ?, 'pending', 0, ?, ?)""",
     53             (account_id, destination, kind, project_id, idempotency_key, text.strip()[:3000], next_attempt_at, now),
     54         )
     55         notification_id = cursor.lastrowid
     56         db.commit()
     57         return notification_id
     58     except sqlite3.IntegrityError:
     59         # Chave idempotente duplicada - ignora silenciosamente
     60         db.rollback()
     61         return None
     62     except Exception:
     63         db.rollback()
     64         raise

     67 def get_pending_notifications(limit: int = 100) -> list[dict]:
     68     """Obtém notificações pendentes para processamento."""
     69     rows = get_db().execute(
     70         """SELECT id, account_id, destination, kind, project_id, idempotency_key, text, attempts, next_attempt_at, created_at
     71            FROM notification_outbox
     72            WHERE status = 'pending' AND next_attempt_at <= ?
     73            ORDER BY created_at ASC
     74            LIMIT ?""",
     75         (int(time.time()), limit),
     76     ).fetchall()
     77     return [dict(row) for row in rows]

    114 def list_account_notifications(account_id: str, limit: int = 50, status: str | None = None) -> list[dict]:
    115     """Lista notificações de uma conta (para exibição no perfil/admin)."""
    116     query = """SELECT id, kind, project_id, text, status, attempts, created_at, next_attempt_at
    117                FROM notification_outbox WHERE account_id = ?"""
    118     params = [account_id]
    119     if status:
    120         query += " AND status = ?"
    121         params.append(status)
    122     query += " ORDER BY created_at DESC LIMIT ?"
    123     params.append(limit)
    124 
    125     rows = get_db().execute(query, tuple(params)).fetchall()
    126     return [dict(row) for row in rows]

    129 def get_notification_stats() -> dict:
    130     """Estatísticas da fila de notificações (para admin dashboard)."""
    131     db = get_db()
    132     rows = db.execute(
    133         """SELECT status, COUNT(*) as count
    134            FROM notification_outbox
    135            GROUP BY status"""
    136     ).fetchall()
    137     stats = {row["status"]: row["count"] for row in rows}
    138     # Total de hoje
    139     today_start = int(time.time()) - (int(time.time()) % 86400)
    140     today_count = db.execute(
    141         "SELECT COUNT(*) as count FROM notification_outbox WHERE created_at >= ?",
    142         (today_start,),
    143     ).fetchone()
    144     stats["today"] = today_count["count"] if today_count else 0
    145     return stats
