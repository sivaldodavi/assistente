     16 def create_revision(
     17     account_id: str,
     18     project_id: str,
     19     revision: int,
     20     title: str,
     21     project_type: str,
     22     payload: dict,
     23 ) -> str:
     24     """Cria um snapshot de revisão do projeto."""
     25     snapshot_id = str(uuid4())
     26     now = int(time.time())
     27     db = get_db()
     28     db.execute(
     29         """INSERT INTO project_revisions
     30            (snapshot_id, project_id, account_id, revision, title, project_type, payload, created_at)
     31            VALUES (?, ?, ?, ?, ?, ?, ?, ?)""",
     32         (
     33             snapshot_id,
     34             project_id,
     35             account_id,
     36             revision,
     37             title[:240],
     38             project_type,
     39             json.dumps(payload, ensure_ascii=False, separators=(",", ":")),
     40             now,
     41         ),
     42     )
     43     # Limpeza: manter apenas as últimas MAX_REVISIONS_PER_PROJECT revisões
     44     db.execute(
     45         """DELETE FROM project_revisions
     46            WHERE project_id = ? AND snapshot_id NOT IN (
     47                SELECT snapshot_id FROM project_revisions
     48                WHERE project_id = ?
     49                ORDER BY revision DESC LIMIT ?
     50            )""",
     51         (project_id, project_id, MAX_REVISIONS_PER_PROJECT),
     52     )
     53     # Limpeza por idade (opcional, rodar periodicamente)
     54     db.commit()
     55     return snapshot_id

    100 def diff_revisions(
    101     account_id: str,
    102     project_id: str,
    103     snapshot_id_a: str,
    104     snapshot_id_b: str,
    105 ) -> dict | None:
    106     """Compara duas revisões e retorna diff simplificado."""
    107     rev_a = get_revision(account_id, project_id, snapshot_id_a)
    108     rev_b = get_revision(account_id, project_id, snapshot_id_b)
    109     if not rev_a or not rev_b:
    110         return None
    111 
    112     payload_a = rev_a.get("payload", {})
    113     payload_b = rev_b.get("payload", {})
    114 
    115     fields_a = payload_a.get("fields", {}) if isinstance(payload_a, dict) else {}
    116     fields_b = payload_b.get("fields", {}) if isinstance(payload_b, dict) else {}
    117 
    118     all_keys = set(fields_a.keys()) | set(fields_b.keys())
    119     diff = {}
    120     for key in sorted(all_keys):
    121         val_a = fields_a.get(key, "")
    122         val_b = fields_b.get(key, "")
    123         if val_a != val_b:
    124             diff[key] = {"from": val_a, "to": val_b}
    125 
    126     return {
    127         "revision_a": {"snapshot_id": snapshot_id_a, "revision": rev_a["revision"], "title": rev_a["title"]},
    128         "revision_b": {"snapshot_id": snapshot_id_b, "revision": rev_b["revision"], "title": rev_b["title"]},
    129         "fields_diff": diff,
    130     }

    133 def restore_revision(
    134     account_id: str,
    135     project_id: str,
    136     snapshot_id: str,
    137     expected_revision: int,
    138 ) -> int | None:
    139     """Restaura uma revisão anterior como nova revisão do projeto."""
    140     from .projects import save_project, ProjectConflict, get_project
    141 
    142     revision_data = get_revision(account_id, project_id, snapshot_id)
    143     if not revision_data:
    144         return None
    145 
    146     current = get_project(account_id, project_id)
    147     if current is None:
    148         return None
    149 
    150     # Usa o payload da revisão como novo payload
    151     payload = revision_data["payload"]
    152     try:
    153         next_revision = save_project(account_id, project_id, expected_revision, payload)
    154         return next_revision
    155     except ProjectConflict:
    156         raise
    157     except Exception:
    158         return None

    161 def cleanup_old_revisions(account_id: str | None = None) -> int:
    162     """Remove revisões antigas (> MAX_REVISION_AGE_DAYS). Retorna quantidade removida."""
    163     cutoff = int(time.time()) - (MAX_REVISION_AGE_DAYS * 86400)
    164     db = get_db()
    165     if account_id:
    166         cursor = db.execute(
    167             "DELETE FROM project_revisions WHERE account_id = ? AND created_at < ?",
    168             (account_id, cutoff),
    169         )
    170     else:
    171         cursor = db.execute("DELETE FROM project_revisions WHERE created_at < ?", (cutoff,))
    172     db.commit()
    173     return cursor.rowcount
