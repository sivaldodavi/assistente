     18         return None

     20         """INSERT INTO project_sharing (project_id, account_id, allow_admin_view, allow_admin_comments, inherit_account_default, updated_at)
     21            VALUES (?, ?, ?, ?, ?, ?) ON CONFLICT(project_id) DO UPDATE SET allow_admin_view=excluded.allow_admin_view, allow_admin_comments=excluded.allow_admin_comments, inherit_account_default=excluded.inherit_account_default, updated_at=excluded.updated_at""",
     22         (project_id, account_id, int(allow_view), int(allow_comments and allow_view), int(inherit_account_default), int(time.time())),
