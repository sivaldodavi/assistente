     77 def _account_dict(row: Any) -> dict[str, Any]:
     78     project_used = max(0, int(row["project_used"] or 0))
     79     project_limit = max(0, int(row["project_limit"] or 0))
     80     ai_used = max(0, int(row["ai_used"] or 0))
     81     ai_limit = max(0, int(row["ai_limit"] or 0))
     82     plan_code = row["plan_code"] if row["plan_code"] in PLAN_CODES else "livre"
     83     is_admin = _is_admin_email(row["email"])
     84     return {
     85         "id": row["id"],
     86         "email": row["email"],
     87         "name": row["name"],
     88         "plan_code": plan_code,
     89         "plan_label": PLAN_LABELS[plan_code],
     90         "project_limit": project_limit,
     91         "project_used": project_used,
     92         "project_remaining": None if is_admin else max(0, project_limit - project_used),
     93         "ai_limit": ai_limit,
     94         "ai_used": ai_used,
     95         "ai_remaining": None if is_admin else max(0, ai_limit - ai_used),
     96         "is_blocked": bool(row["is_blocked"]),
     97         "is_admin": is_admin,
     98         "unlimited": is_admin,
     99         "project_count": int(row["project_count"] or 0),
    100         "created_at": row["created_at"],
    101         "last_login": row["last_login"],
    102         # Novos campos v12
    103         "ai_analysis_limit": int(row["ai_analysis_limit"] or 0),
    104         "deepseek_explanation_limit": int(row["deepseek_explanation_limit"] or 0),
    105     }

    108 def list_admin_accounts() -> list[dict[str, Any]]:
    109     """Return account balances and identity metadata, never project payloads."""
    110     _require_admin_schema()
    111     rows = get_db().execute(
    112         """SELECT a.id, a.email, a.name, a.plan_code,
    113                   a.project_limit, a.project_used, a.ai_limit, a.ai_used,
    114                   a.is_blocked, a.created_at, a.last_login,
    115                   a.ai_analysis_limit, a.deepseek_explanation_limit,
    116                   COUNT(p.id) AS project_count
    117            FROM accounts AS a
    118            LEFT JOIN projects AS p ON p.account_id = a.id
    119            GROUP BY a.id, a.email, a.name, a.plan_code,
    120                     a.project_limit, a.project_used, a.ai_limit, a.ai_used,
    121                     a.is_blocked, a.created_at, a.last_login,
    122                     a.ai_analysis_limit, a.deepseek_explanation_limit
    123            ORDER BY a.created_at ASC, a.email ASC"""
    124     ).fetchall()
    125     return [_account_dict(row) for row in rows]

    142 def set_account_plan(account_id: str, plan_code: str) -> dict[str, Any]:
    143     """Change only the manually displayed plan badge; do not change balances."""
    144     _require_admin_schema()
    145     if not isinstance(plan_code, str) or plan_code not in PLAN_CODES:
    146         raise AdminValidationError("Plano inválido.")
    147     database = get_db()
    148     database.execute("BEGIN IMMEDIATE")
    149     row = database.execute("SELECT id FROM accounts WHERE id = ?", (account_id,)).fetchone()
    150     if row is None:
    151         database.rollback()
    152         return None
    153     database.execute("UPDATE accounts SET plan_code = ? WHERE id = ?", (plan_code, account_id))
    154     database.commit()
    155     return get_admin_account(account_id)

    158 def set_project_credits(account_id: str, project_remaining: int) -> dict[str, Any]:
    159     """Set available project credits independently of the plan badge."""
    160     _require_admin_schema()
    161     if (
    162         isinstance(project_remaining, bool)
    163         or not isinstance(project_remaining, int)
    164         or not 0 <= project_remaining <= MAX_ACCOUNT_LIMIT
    165     ):
    166         raise AdminValidationError(
    167             f"Os créditos de projeto devem ser um inteiro entre 0 e {MAX_ACCOUNT_LIMIT}."
    168         )
    169     database = get_db()
    170     database.execute("BEGIN IMMEDIATE")
    171     row = database.execute(
    172         "SELECT project_used FROM accounts WHERE id = ?", (account_id,)
    173     ).fetchone()
    174     if row is None:
    175         database.rollback()
    176         return None
    177     project_used = max(0, int(row["project_used"] or 0))
    178     if project_used + project_remaining > MAX_ACCOUNT_LIMIT:
    179         database.rollback()
    180         raise AdminValidationError(
    181             f"O total concedido não pode ultrapassar {MAX_ACCOUNT_LIMIT} projetos."
    182         )
    183     database.execute(
    184         "UPDATE accounts SET project_limit = ? WHERE id = ?",
    185         (project_used + project_remaining, account_id),
    186     )
    187     database.commit()
    188     return get_admin_account(account_id)

    191 def set_account_blocked(account_id: str, blocked: bool, actor_id: str | None = None) -> dict[str, Any]:
    192     """Set a manual block flag, protecting the current administrator from lockout."""
    193     _require_admin_schema()
    194     if type(blocked) is not bool:
    195         raise AdminValidationError("O estado de bloqueio é inválido.")
    196     database = get_db()
    197     database.execute("BEGIN IMMEDIATE")
    198     row = database.execute(
    199         """SELECT id, email, is_blocked
    200            FROM accounts WHERE id = ?""",
    201         (account_id,),
    202     ).fetchone()
    203     if row is None:
    204         database.rollback()
    205         return None
    206     if actor_id and actor_id == account_id:
    207         database.rollback()
    208         raise AdminValidationError("A conta administrativa atual não pode bloquear a própria sessão.")
    209     if blocked and _is_admin_email(row["email"]):
    210         admin_count = database.execute(
    211             "SELECT COUNT(*) FROM accounts WHERE lower(email) IN ({})".format(
    212                 ",".join("?" for _ in current_app.config.get("ADMIN_EMAILS", set()))
    213             ),
    214             tuple(sorted(current_app.config.get("ADMIN_EMAILS", set()))),
    215         ).fetchone()[0] if current_app.config.get("ADMIN_EMAILS", set()) else 0
    216         if admin_count <= 1:
    217             database.rollback()
    218             raise AdminValidationError("Não bloqueie a única conta administradora configurada.")
    219     database.execute(
    220         "UPDATE accounts SET is_blocked = ? WHERE id = ?", (int(blocked), account_id)
    221     )
    222     database.commit()
    223     return get_admin_account(account_id)

    226 def list_account_projects(account_id: str) -> list[dict[str, Any]] | None:
    227     """List safe project metadata; private payloads never leave the database."""
    228     _require_admin_schema()
    229     if get_db().execute("SELECT 1 FROM accounts WHERE id = ?", (account_id,)).fetchone() is None:
    230         return None
    231     rows = get_db().execute(
    232         """SELECT p.id, p.title, p.project_type, p.created_at, p.updated_at,
    233                   COALESCE(s.allow_admin_view, 0) AS allow_admin_view,
    234                   COALESCE(s.allow_admin_comments, 0) AS allow_admin_comments
    235            FROM projects p
    236            LEFT JOIN project_sharing s ON s.project_id = p.id AND s.account_id = p.account_id
    237            WHERE p.account_id = ?
    238            ORDER BY p.updated_at DESC, p.created_at DESC""",
    239         (account_id,),
    240     ).fetchall()
    241     return [
    242         {
    243             "id": row["id"],
    244             "title": row["title"],
    245             "project_type": row["project_type"],
    246             "created_at": row["created_at"],
    247             "updated_at": row["updated_at"],
    248             "allow_admin_view": bool(row["allow_admin_view"]),
    249             "allow_admin_comments": bool(row["allow_admin_view"] and row["allow_admin_comments"]),
    250         }
    251         for row in rows
    252     ]

    259 def global_usage() -> dict[str, Any]:
    260     """Retorna estatísticas globais de uso de IA (DeepSeek + JEV/Typesafe)."""
    261     db = get_db()
    262     # Usage reservations
    263     usage_stats = db.execute(
    264         """SELECT
    265             COUNT(*) as total_reservations,
    266             SUM(CASE WHEN status = 'complete' THEN units ELSE 0 END) as completed,
    267             SUM(CASE WHEN status = 'refunded' THEN units ELSE 0 END) as refunded,
    268             SUM(CASE WHEN status = 'reserved' THEN units ELSE 0 END) as reserved
    269            FROM ai_usage_reservations"""
    270     ).fetchone()
    271 
    272     # Provider usage
    273     provider_stats = db.execute(
    274         """SELECT provider, operation, COUNT(*) as calls,
    275                   SUM(input_tokens) as total_input_tokens,
    276                   SUM(output_tokens) as total_output_tokens
    277            FROM provider_usage
    278            GROUP BY provider, operation"""
    279     ).fetchall()
    280 
    281     # DeepSeek usage
    282     deepseek_stats = db.execute(
    283         """SELECT COUNT(*) as calls,
    284                   SUM(input_tokens) as total_input_tokens,
    285                   SUM(output_tokens) as total_output_tokens,
    286                   SUM(estimated_cost_usd) as total_cost
    287            FROM deepseek_usage"""
    288     ).fetchone()
    289 
    290     return {
    291         "reservations": dict(usage_stats) if usage_stats else {},
    292         "by_provider": [dict(row) for row in provider_stats],
    293         "deepseek": dict(deepseek_stats) if deepseek_stats else {},
    294     }

    297 def operational_health() -> dict[str, Any]:
    298     """Retorna health check operacional do sistema."""
    299     db = get_db()
    300 
    301     # DB integrity
    302     integrity = db.execute("PRAGMA integrity_check").fetchone()[0]
    303 
    304     # Notification queue stats
    305     notif_stats = db.execute(
    306         """SELECT status, COUNT(*) as count
    307            FROM notification_outbox
    308            GROUP BY status"""
    309     ).fetchall()
    310 
    311     # Active sessions (last 24h)
    312     now = int(time.time())
    313     day_ago = now - 86400
    314     active_sessions = db.execute(
    315         "SELECT COUNT(*) as count FROM sessions WHERE expires_at > ?", (day_ago,)
    316     ).fetchone()
    317 
    318     # Disk usage
    319     usage = shutil.disk_usage("/")
    320     disk = {
    321         "total_bytes": usage.total,
    322         "free_bytes": usage.free,
    323         "used_percent": round((usage.total - usage.free) / usage.total * 100, 1) if usage.total else 0,
    324     }
    325 
    326     # Database size
    327     db_path = current_app.config.get("DATABASE", "")
    328     db_size = 0
    329     if db_path:
    330         from pathlib import Path
    331         for path in (Path(db_path), Path(str(db_path) + "-wal"), Path(str(db_path) + "-shm")):
    332             if path.exists():
    333                 db_size += path.stat().st_size
    334 
    335     return {
    336         "database": {
    337             "integrity": integrity,
    338             "notifications": {row["status"]: row["count"] for row in notif_stats},
    339             "size_bytes": db_size,
    340         },
    341         "active_sessions_24h": active_sessions["count"] if active_sessions else 0,
    342         "disk": disk,
    343     }

    346 def set_account_limits(account_id: str, payload: dict) -> dict[str, Any] | None:
    347     """Admin define limites customizados da conta."""
    348     _require_admin_schema()
    349     db = get_db()
    350     db.execute("BEGIN IMMEDIATE")
    351 
    352     row = db.execute("SELECT id FROM accounts WHERE id = ?", (account_id,)).fetchone()
    353     if row is None:
    354         db.rollback()
    355         return None
    356 
    357     # Validação dos limites
    358     ai_analysis_limit = payload.get("ai_analysis_limit")
    359     deepseek_explanation_limit = payload.get("deepseek_explanation_limit")
    360     is_admin = payload.get("is_admin")
    361 
    362     if ai_analysis_limit is not None:
    363         if not isinstance(ai_analysis_limit, int) or not (0 <= ai_analysis_limit <= MAX_ACCOUNT_LIMIT):
    364             db.rollback()
    365             raise AdminValidationError(f"Limite de análise IA deve ser inteiro entre 0 e {MAX_ACCOUNT_LIMIT}.")
    366         db.execute("UPDATE accounts SET ai_analysis_limit = ? WHERE id = ?", (ai_analysis_limit, account_id))
    367 
    368     if deepseek_explanation_limit is not None:
    369         if not isinstance(deepseek_explanation_limit, int) or not (0 <= deepseek_explanation_limit <= MAX_ACCOUNT_LIMIT):
    370             db.rollback()
    371             raise AdminValidationError(f"Limite de explicação DeepSeek deve ser inteiro entre 0 e {MAX_ACCOUNT_LIMIT}.")
    372         db.execute("UPDATE accounts SET deepseek_explanation_limit = ? WHERE id = ?", (deepseek_explanation_limit, account_id))
    373 
    374     if is_admin is not None:
    375         if not isinstance(is_admin, bool):
    376             db.rollback()
    377             raise AdminValidationError("O campo is_admin deve ser booleano.")
    378         # Atualiza ai_limit baseado nos novos limites
    379         new_ai_limit = int(ai_analysis_limit or 0) + int(deepseek_explanation_limit or 0)
    380         db.execute("UPDATE accounts SET is_admin = ?, ai_limit = ? WHERE id = ?",
    381                    (int(is_admin), new_ai_limit, account_id))
    382 
    383     db.commit()
    384     return get_admin_account(account_id)
