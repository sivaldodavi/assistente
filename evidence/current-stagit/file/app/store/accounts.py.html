     36 def get_account(account_id: str) -> dict | None:

     47         "email": row["email"],

     58         "ai_limit": int(row["ai_limit"] or 0),
     59         "ai_used": int(row["ai_used"] or 0),
     60         "project_remaining": None if is_admin else max(0, int(row["project_limit"] or 0) - int(row["project_used"] or 0)),
     61         "ai_remaining": None if is_admin else max(0, int(row["ai_limit"] or 0) - int(row["ai_used"] or 0)),
     62         "unlimited": is_admin,
     63         # Novos campos v12
     64         "sharing_preferences_version": int(row["sharing_preferences_version"] or 1),
     65         "sharing_allow_admin_view": bool(row["sharing_allow_admin_view"]),
     66         "sharing_allow_admin_comments": bool(row["sharing_allow_admin_comments"]),

    130 def set_account_sharing_preferences(account_id: str, payload: dict) -> dict:
    131     """Atualiza preferências de compartilhamento no nível da conta.
    132 
    133     Regras de validação:
    134     - allow_admin_comments requer allow_admin_view
    135     - version é incrementado a cada mudança
    136     """
    137     allow_view = bool(payload.get("allow_admin_view", False))
    138     allow_comments = bool(payload.get("allow_admin_comments", False))
    139     inherit_default = bool(payload.get("inherit_account_default", True))
    140 
    141     if allow_comments and not allow_view:
    142         raise ValueError("Comentários administrativos requerem visualização administrativa.")
    143 
    144     now = int(time.time())
    145     get_db().execute(
    146         """UPDATE accounts
    147            SET sharing_allow_admin_view = ?, sharing_allow_admin_comments = ?,
    148                sharing_inherit_default = ?, sharing_preferences_version = sharing_preferences_version + 1
    149            WHERE id = ?""",
    150         (int(allow_view), int(allow_comments), int(inherit_default), account_id),
    151     )
    152     get_db().commit()
    153     return get_account_sharing_preferences(account_id)

    181 def request_self_deletion(account_id: str) -> dict:
    182     """Usuário solicita exclusão da própria conta."""
    183     now = int(time.time())
    184     db = get_db()
    185     db.execute("BEGIN IMMEDIATE")
    186     # Verifica se já existe pedido pendente
    187     existing = db.execute(
    188         "SELECT status FROM deletion_requests WHERE account_id = ?",
    189         (account_id,),
    190     ).fetchone()
    191     if existing and existing["status"] == "pending":
    192         db.rollback()
    193         raise ValueError("Já existe uma solicitação de exclusão pendente.")
    194 
    195     # Cria pedido de exclusão
    196     db.execute(
    197         """INSERT INTO deletion_requests (account_id, requested_by, status, created_at)
    198            VALUES (?, ?, 'pending', ?)""",
    199         (account_id, account_id, now),
    200     )
    201     # Marca conta como com pedido pendente
    202     db.execute(
    203         "UPDATE accounts SET deletion_request_pending = 1 WHERE id = ?",
    204         (account_id,),
    205     )
    206     db.commit()
    207     return {"status": "pending", "created_at": now}

    348 def export_account_data(account_id: str) -> dict:
    349     """Exporta todos os dados da conta para LGPD/GDPR."""
    350     from .projects import list_projects, get_project
    351     from .sharing import list_comments
    352     from .sources import list_sources
    353     from .ai_usage import get_db as get_ai_db
    354 
    355     account = get_account(account_id)
    356     if not account:
    357         raise ValueError("Conta não encontrada")
    358 
    359     projects_data = []
    360     for project in list_projects(account_id):
    361         full_project = get_project(account_id, project["id"])
    362         if full_project:
    363             projects_data.append(full_project)
    364 
    365     # Revisions
    366     revisions = []
    367     for project in list_projects(account_id):
    368         rev_rows = get_db().execute(
    369             """SELECT snapshot_id, revision, title, project_type, payload, created_at
    370                FROM project_revisions WHERE project_id = ? ORDER BY revision DESC""",
    371             (project["id"],),
    372         ).fetchall()
    373         for rev in rev_rows:
    374             try:
    375                 payload = json.loads(rev["payload"])
    376             except (TypeError, ValueError, json.JSONDecodeError):
    377                 payload = {}
    378             revisions.append({
    379                 "project_id": project["id"],
    380                 "snapshot_id": rev["snapshot_id"],
    381                 "revision": rev["revision"],
    382                 "title": rev["title"],
    383                 "project_type": rev["project_type"],
    384                 "payload": payload,
    385                 "created_at": rev["created_at"],
    386             })
    387 
    388     # Comments received (as owner)
    389     comments = []
    390     for project in list_projects(account_id):
    391         proj_comments = list_comments(project["id"])
    392         for c in proj_comments:
    393             c["project_id"] = project["id"]
    394             comments.append(c)
    395 
    396     # Sources
    397     sources_data = list_sources(account_id)
    398 
    399     # AI Usage
    400     ai_usage = get_ai_db().execute(
    401         """SELECT request_id, action, units, status, created_at, updated_at
    402            FROM ai_usage_reservations WHERE account_id = ? ORDER BY created_at DESC""",
    403         (account_id,),
    404     ).fetchall()
    405 
    406     provider_usage = get_ai_db().execute(
    407         """SELECT request_id, provider, operation, model, input_tokens, output_tokens, status, created_at
    408            FROM provider_usage WHERE account_id = ? ORDER BY created_at DESC""",
    409         (account_id,),
    410     ).fetchall()
    411 
    412     return {
    413         "account": account,
    414         "projects": projects_data,
    415         "revisions": revisions,
    416         "comments": comments,
    417         "sources": sources_data,
    418         "ai_usage_reservations": [dict(r) for r in ai_usage],
    419         "provider_usage": [dict(r) for r in provider_usage],
    420         "exported_at": int(time.time()),
    421     }
