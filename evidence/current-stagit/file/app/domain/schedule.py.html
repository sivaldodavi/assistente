     47 def _date(value: object) -> str:
     48     text = str(value or "").strip()
     49     if not text:
     50         return ""
     51     if len(text) == 10 and text[2] == "/" and text[5] == "/":
     52         text = f"{text[6:]}-{text[3:5]}-{text[:2]}"
     53     try:
     54         return date.fromisoformat(text).isoformat()
     55     except ValueError:
     56         return ""

     63 def _normalise_title(value: object) -> str:
     64     text = unicodedata.normalize("NFD", str(value or "").casefold())
     65     return "".join(character for character in text if unicodedata.category(character) != "Mn")

     92 def normalize_schedule(raw: object, fallback_date: object = "", project_type: str = "case") -> dict:
     93     source = raw if isinstance(raw, dict) else {}
     94     report_date = source.get("report_date") or source.get("deliveryDate") or source.get("paper_project_date")
     95     final_date = source.get("final_date") or source.get("finalDate") or source.get("paperFinalDate")
     96     tasks = []
     97     raw_tasks = source.get("tasks") if isinstance(source.get("tasks"), list) else []
     98     for item in raw_tasks[:100]:
     99         if not isinstance(item, dict):
    100             continue
    101         raw_date = item.get("date")
    102         due = _date(raw_date)
    103         if raw_date and not due:
    104             continue
    105         title = _text(item.get("title"), 180)
    106         identifier = _text(item.get("id"), 80) or str(uuid4())
    107         if not title and not due and not item.get("id"):
    108             continue
    109         raw_stage = item.get("stage")
    110         stage = raw_stage if isinstance(raw_stage, int) and raw_stage >= 0 else None
    111         template_key = _text(item.get("template_key") or item.get("templateKey"), 80)
    112         tasks.append({
    113             "id": identifier,
    114             "title": title,
    115             "date": due,
    116             "done": bool(item.get("done")),
    117             "stage": stage,
    118             "template_key": template_key or _template_key(project_type, title),
    119         })
    120     return {
    121         "report_date": _date(report_date) or _date(fallback_date),
    122         "debate_date": _date(source.get("debate_date") or source.get("debateDate")),
    123         "dissertation_date": _date(source.get("dissertation_date") or source.get("dissertationDate")),
    124         "final_date": _date(final_date),
    125         "tasks": tasks,
    126     }

    166 def _edit_level(count: int) -> int:

    178 def account_calendar(projects: list[dict], edit_activity: list[dict] | None = None, today: date | None = None) -> dict:
    179     if isinstance(edit_activity, date) and today is None:
    180         today, edit_activity = edit_activity, None
    181     today = today or date.today()
    182     entries = sorted(
    183         [entry for project in projects for entry in project.get("deadlines", [])],
    184         key=lambda item: (item["date"], item["title"], item["label"]),
    185     )
    186     activity = {
    187         str(item.get("date")): int(item.get("count") or 0)
    188         for item in (edit_activity or [])
    189         if isinstance(item, dict) and item.get("date")
    190     }
    191     first = date(today.year, today.month, 1)
    192     leading = [None] * ((first.weekday() + 1) % 7)
    193     cells = leading + [{"number": day} for day in range(1, calendar.monthrange(today.year, today.month)[1] + 1)]
    194     by_date = {}
    195     for entry in entries:
    196         by_date.setdefault(entry["date"], []).append(entry)
    197     for cell in cells:
    198         if cell is None:
    199             continue
    200         key = f"{today.year:04d}-{today.month:02d}-{cell['number']:02d}"
    201         cell["entries"] = by_date.get(key, [])[:2]
    202         cell["more"] = max(0, len(by_date.get(key, [])) - 2)
    203         cell["edit_count"] = activity.get(key, 0)
    204         cell["edit_level"] = _edit_level(cell["edit_count"])
    205         cell["edit_label"] = f"{cell['edit_count']} edição(ões) neste dia"
    206     upcoming = [entry for entry in entries if entry["date"] >= today.isoformat()]
    207     if not upcoming:
    208         upcoming = list(reversed(entries))
    209     return {
    210         "has_dates": bool(entries),
    211         "has_activity": bool(activity),
    212         "month_has_dates": any(cell.get("entries") for cell in cells if cell),
    213         "month_has_activity": any(cell.get("edit_count") for cell in cells if cell),
    214         "month_label": f"{_MONTHS[today.month - 1]} de {today.year}",
    215         "weekdays": ("Dom", "Seg", "Ter", "Qua", "Qui", "Sex", "Sáb"),
    216         "cells": cells,
    217         "upcoming": upcoming[:5],
    218         "upcoming_label": "Próximas datas" if any(entry["date"] >= today.isoformat() for entry in entries) else "Datas cadastradas",
    219     }
