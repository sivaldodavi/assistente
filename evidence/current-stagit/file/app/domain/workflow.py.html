     20 def review_groups(sections, checks, phase):
     21     """Agrupa as conferências da fase atual para a revisão final do roteiro."""
     22     source = checks if isinstance(checks, dict) else {}
     23     groups = []
     24     for section in sections:
     25         if section.get("phase") != phase or not section.get("checks"):
     26             continue
     27         items = [{"label": label, "done": bool(source.get(key))} for key, label in section["checks"]]
     28         groups.append({
     29             "title": section["title"],
     30             "items": items,
     31             "done": sum(item["done"] for item in items),
     32             "total": len(items),
     33         })
     34     done = sum(group["done"] for group in groups)
     35     total = sum(group["total"] for group in groups)
     36     return {"groups": groups, "done": done, "total": total, "percent": round(done / total * 100) if total else 100}
