     18 def _text(value: object, limit: int, name: str, required: bool = False) -> str:
     19     if not isinstance(value, str):
     20         if value is None and not required:
     21             return ""
     22         raise ValueError(f"{name} inválido")
     23     value = value.strip()
     24     if required and not value:
     25         raise ValueError(f"{name} obrigatório")
     26     if len(value) > limit:
     27         raise ValueError(f"{name} grande demais")
     28     return value

     31 def normalize_request(body: object) -> dict:
     32     if not isinstance(body, dict):
     33         raise ValueError("pedido inválido")
     34     request_id = body.get("request_id")
     35     if not isinstance(request_id, str) or not _UUID_RE.fullmatch(request_id):
     36         raise ValueError("request_id inválido")
     37     try:
     38         request_id = str(UUID(request_id))
     39     except ValueError as error:
     40         raise ValueError("request_id inválido") from error
     41     revision = body.get("project_revision")
     42     if isinstance(revision, bool) or not isinstance(revision, int) or revision < 1:
     43         raise ValueError("project_revision inválida")
     44     keywords = body.get("keywords", [])
     45     if not isinstance(keywords, list) or len(keywords) > MAX_KEYWORDS:
     46         raise ValueError("keywords inválidas")
     47     clean_keywords = []
     48     for keyword in keywords:
     49         value = _text(keyword, MAX_KEYWORD, "keyword", required=True)
     50         if value not in clean_keywords:
     51             clean_keywords.append(value)
     52     source_ids = body.get("catalog_source_ids", [])
     53     if not isinstance(source_ids, list) or len(source_ids) > MAX_SOURCE_IDS:
     54         raise ValueError("fontes de catálogo inválidas")
     55     clean_source_ids = []
     56     for source_id in source_ids:
     57         value = _text(source_id, 160, "source_id", required=True)
     58         if value not in clean_source_ids:
     59             clean_source_ids.append(value)
     60     return {
     61         "request_id": request_id,
     62         "project_revision": revision,
     63         "problem_question": _text(body.get("problem_question"), MAX_PROBLEM, "pergunta-problema", required=True),
     64         "professor_instructions": _text(body.get("professor_instructions"), MAX_INSTRUCTIONS, "instruções do professor"),
     65         "keywords": clean_keywords,
     66         "catalog_source_ids": clean_source_ids,
     67     }

     78 def validate_candidates(entries: list[dict]) -> list[dict]:
     79     clean = []
     80     seen = set()
     81     for entry in entries:
     82         if not isinstance(entry, dict):
     83             continue
     84         source_id = entry.get("id")
     85         kind = entry.get("kind")
     86         title = entry.get("title")
     87         if not isinstance(source_id, str) or not source_id or source_id in seen:
     88             continue
     89         if kind not in _KIND_TO_CATEGORY or not isinstance(title, str) or not title.strip():
     90             continue
     91         seen.add(source_id)
     92         clean.append({
     93             "id": source_id,
     94             "kind": kind,
     95             "title": title[:500],
     96             "reference": str(entry.get("reference") or "")[:1000],
     97             "author": str(entry.get("author") or "")[:500],
     98             "abstract": str(entry.get("abstract") or entry.get("ementa") or "")[:4000],
     99             "source_year": entry.get("source_year") if isinstance(entry.get("source_year"), int) else None,
    100             "source": str(entry.get("source") or "")[:500],
    101             "link": entry.get("link") if isinstance(entry.get("link"), str) and entry["link"].startswith("https://") else "",
    102         })
    103     return clean

    106 def validate_result(result: object, candidates: list[dict]) -> dict:
    107     if not isinstance(result, dict) or set(result) - {"law", "jurisprudence", "doctrine", "warnings"}:
    108         raise ValueError("resposta de recomendação inválida")
    109     by_id = {entry["id"]: entry for entry in candidates}
    110     output = {"law": [], "jurisprudence": [], "doctrine": []}
    111     used = set()
    112     for category in output:
    113         items = result.get(category, [])
    114         if not isinstance(items, list) or len(items) > MAX_RESULTS_PER_KIND:
    115             raise ValueError("lista de recomendações inválida")
    116         expected_kind = kind_for_category(category)
    117         for item in items:
    118             if not isinstance(item, dict) or set(item) != {"source_id", "why", "rank"}:
    119                 raise ValueError("item de recomendação inválido")
    120             source_id = item["source_id"]
    121             why = item["why"]
    122             rank = item["rank"]
    123             if source_id not in by_id or source_id in used or by_id[source_id]["kind"] != expected_kind:
    124                 raise ValueError("fonte recomendada não pertence ao catálogo permitido")
    125             if not isinstance(why, str) or not why.strip() or len(why) > 600:
    126                 raise ValueError("justificativa inválida")
    127             if isinstance(rank, bool) or not isinstance(rank, int) or not 1 <= rank <= MAX_RESULTS_PER_KIND:
    128                 raise ValueError("posição inválida")
    129             used.add(source_id)
    130             output[category].append({"source_id": source_id, **by_id[source_id], "why": why.strip(), "rank": rank})
    131     warnings = result.get("warnings", [])
    132     if not isinstance(warnings, list) or any(not isinstance(item, str) or len(item) > 300 for item in warnings):
    133         raise ValueError("avisos inválidos")
    134     return {"recommendations": output, "warnings": warnings[:5]}
