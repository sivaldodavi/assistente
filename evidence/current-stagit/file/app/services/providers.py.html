     25 def load_secret(env_name: str, file_env_name: str, default_file: str = "") -> str:
     26     value = os.environ.get(env_name, "").strip()
     27     path = os.environ.get(file_env_name, default_file)
     28     if not value and path:
     29         try:
     30             value = Path(path).read_text(encoding="utf-8").strip()
     31         except (OSError, UnicodeError):
     32             # Não vazar o caminho do arquivo na mensagem de erro
     33             raise ProviderError("secret_unavailable", "A configuração do provedor não pôde ser carregada.")
     34     if not value or len(value) > 4096 or "\n" in value or "\r" in value:
     35         raise ProviderError("secret_invalid", "A configuração do provedor é inválida.")
     36     return value

     39 class _JsonProvider:
     40     def __init__(self, api_key: str, api_url: str, model: str, opener=urlopen, timeout: int = DEFAULT_PROVIDER_TIMEOUT):
     41         self.api_key = api_key
     42         self.api_url = api_url
     43         self.model = model
     44         self.opener = opener
     45         self.timeout = timeout

     47     def post(self, payload: dict, timeout: int | None = None) -> dict:
     48         effective_timeout = timeout if timeout is not None else self.timeout
     49         body = json.dumps(payload, ensure_ascii=False, separators=(",", ":")).encode("utf-8")
     50         request = Request(
     51             self.api_url,
     52             data=body,
     53             headers={"Authorization": f"Bearer {self.api_key}", "Content-Type": "application/json"},
     54             method="POST",
     55         )
     56         try:
     57             with self.opener(request, timeout=effective_timeout) as response:
     58                 raw = response.read(4 * 1024 * 1024 + 1)
     59         except HTTPError as error:
     60             status = error.code
     61             error.close()
     62             if status in {408, 429, 529} or status >= 500:
     63                 raise ProviderError("provider_busy", "O provedor está temporariamente ocupado.", 503, retryable=True) from error
     64             if status == 401:
     65                 raise ProviderError("provider_auth", "A configuração do provedor foi recusada.", 502, retryable=True) from error
     66             if status == 402:
     67                 raise ProviderError("provider_quota", "O provedor não possui crédito disponível.", 503, retryable=True) from error
     68             raise ProviderError("provider_error", "O provedor não concluiu a solicitação.", 502) from error
     69         except (TimeoutError, URLError, OSError) as error:
     70             raise ProviderError("provider_unreachable", "Não foi possível conectar ao provedor.", 503) from error
     71         if len(raw) > 4 * 1024 * 1024:
     72             raise ProviderError("response_too_large", "A resposta do provedor é grande demais.", 502)
     73         try:
     74             data = json.loads(raw)
     75         except (TypeError, ValueError, json.JSONDecodeError) as error:
     76             raise ProviderError("invalid_response", "O provedor retornou uma resposta inválida.", 502, retryable=True) from error
     77         if not isinstance(data, dict):
     78             raise ProviderError("invalid_response", "O provedor retornou uma resposta inválida.", 502, retryable=True)
     79         return data
