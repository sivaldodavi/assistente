     22 def catalog_path(configured: str | os.PathLike[str] | None = None) -> Path:
     23     """Resolve o arquivo do catálogo sem criar diretórios ou tocar no banco de usuários.
     24 
     25     ``RESEARCH_CATALOG_PATH`` aponta para o arquivo montado em produção
     26     (normalmente ``/catalogs/catalog.sqlite3``). Os nomes antigos são somente
     27     fallbacks para desenvolvimento e testes de migração.
     28     """
     29     value = configured
     30     if value is None:
     31         try:
     32             from flask import current_app, has_app_context
     33 
     34             if has_app_context():
     35                 value = current_app.config.get("RESEARCH_CATALOG_PATH")
     36         except ImportError:
     37             pass
     38     if not value:
     39         value = os.environ.get("RESEARCH_CATALOG_PATH")
     40     if value:
     41         path = Path(value).expanduser()
     42         if path.exists() and path.is_dir():
     43             path = path / "catalog.sqlite3"
     44         return path
     45 
     46     catalog_dir = os.environ.get("ASSISTENTE_CATALOG_DIR")
     47     if catalog_dir:
     48         return Path(catalog_dir).expanduser() / "catalog.sqlite3"
     49     data_dir = os.environ.get("ASSISTENTE_DATA_DIR")
     50     return (Path(data_dir).expanduser() if data_dir else Path("data")) / "catalogs" / "catalog.sqlite3"

     53 def _fts_query(query: str) -> str:
     54     tokens = []
     55     for token in _WORD_RE.findall(query):
     56         clean = token.replace('"', "")
     57         if clean:
     58             tokens.append(f'"{clean}"*')
     59     return " AND ".join(tokens)

     62 def _metadata(path: Path, query: str, limit: int, returned: int) -> dict[str, Any]:
     63     result: dict[str, Any] = {}
     64     metadata_path = path.with_name("catalog_meta.json")
     65     try:
     66         loaded = json.loads(metadata_path.read_text(encoding="utf-8"))
     67         if isinstance(loaded, dict):
     68             result.update(loaded)
     69     except (OSError, json.JSONDecodeError):
     70         pass
     71     # Nunca retornamos o caminho físico do volume montado ao navegador.
     72     result.update(
     73         {
     74             "query": query,
     75             "limit": limit,
     76             "returned": returned,
     77             "scope": "Índice remissivo e metadados locais; pista inicial, não pesquisa concluída.",
     78         }
     79     )
     80     return result

     83 def _entry_from_row(row: sqlite3.Row) -> dict[str, Any]:
     84     payload = row[0]
     85     if payload:
     86         try:
     87             value = json.loads(payload)
     88         except (TypeError, json.JSONDecodeError):
     89             value = None
     90         if isinstance(value, dict):
     91             return value
     92     return {
     93         key: row[index]
     94         for index, key in enumerate(("id", "kind", "title", "reference", "author", "abstract", "source_year", "source"), 1)
     95         if len(row) > index and row[index] not in (None, "")
     96     }

     99 def search_catalog(
    100     query: str,
    101     kind: str | None = None,
    102     limit: int = 12,
    103     catalog_file: str | os.PathLike[str] | None = None,
    104     catalog_dir: str | os.PathLike[str] | None = None,
    105     letter: str | None = None,
    106 ) -> dict[str, Any]:
    107     """Busca no índice legado sem abrir o banco de contas e sem executar o legado."""
    108     if letter is not None and (len(letter) != 1 or letter.upper() not in "ABCDEFGHIJKLMNOPQRSTUVWXYZ"):
    109         raise ValueError("letra inicial inválida")
    110     if kind == "case_law":
    111         kind = "case-law"
    112     if kind is not None and kind not in RESEARCH_KINDS:
    113         raise ValueError("tipo de catálogo inválido")
    114     try:
    115         amount = int(limit)
    116     except (TypeError, ValueError) as exc:
    117         raise ValueError("limite inválido") from exc
    118     if amount <= 0:
    119         return {"entries": [], "metadata": {"query": query or "", "limit": amount, "returned": 0}}
    120     amount = min(amount, 100)
    121     text = str(query or "").strip()
    122     path = catalog_path(catalog_file if catalog_file is not None else catalog_dir)
    123     if not path.is_file():
    124         raise ResearchUnavailable("catálogo local não montado")
    125 
    126     uri = f"file:{quote(str(path.resolve()), safe='/')}?mode=ro"
    127     connection: sqlite3.Connection | None = None
    128     try:
    129         connection = sqlite3.connect(uri, uri=True)
    130         connection.row_factory = sqlite3.Row
    131         connection.create_function("initial", 1, lambda value: unicodedata.normalize("NFD", str(value or "").strip()).upper()[:1], deterministic=True)
    132         tables = {
    133             row[0]
    134             for row in connection.execute(
    135                 "SELECT name FROM sqlite_master WHERE type IN ('table', 'view')"
    136             ).fetchall()
    137         }
    138         if "entries" not in tables:
    139             raise ResearchUnavailable("catálogo sem tabela de entradas")
    140         terms = _fts_query(text)
    141         params: list[Any] = []
    142         where: list[str] = []
    143         if letter:
    144             where.append("initial(e.title) = ?")
    145             params.append(letter.upper())
    146         if kind:
    147             where.append("e.kind = ?")
    148             params.append(kind)
    149         if terms and "entries_fts" in tables:
    150             where.append("entries_fts MATCH ?")
    151             params.append(terms)
    152             sql = (
    153                 "SELECT e.payload, e.id, e.kind, e.title, e.reference, e.author, "
    154                 "e.abstract, e.source_year, e.source FROM entries_fts AS f "
    155                 "JOIN entries AS e ON e.id = f.id"
    156             )
    157         else:
    158             if terms:
    159                 where.append("e.search_text LIKE ?")
    160                 params.append("%" + "%".join(_WORD_RE.findall(text)) + "%")
    161             sql = (
    162                 "SELECT e.payload, e.id, e.kind, e.title, e.reference, e.author, "
    163                 "e.abstract, e.source_year, e.source FROM entries AS e"
    164             )
    165         if where:
    166             sql += " WHERE " + " AND ".join(where)
    167         sql += " ORDER BY e.title COLLATE NOCASE, e.id LIMIT ?"
    168         params.append(amount)
    169         try:
    170             rows = connection.execute(sql, params).fetchall()
    171         except sqlite3.OperationalError as exc:
    172             # Alguns índices antigos não têm FTS5; a coluna search_text é a
    173             # mesma usada pelo importador e permite fallback sem reindexar.
    174             if terms and "entries_fts" in tables and "no such table" not in str(exc).lower():
    175                 raise
    176             fallback_where: list[str] = []
    177             fallback_params: list[Any] = []
    178             if kind:
    179                 fallback_where.append("e.kind = ?")
    180                 fallback_params.append(kind)
    181             fallback_where.append("e.search_text LIKE ?")
    182             fallback_params.append("%" + "%".join(_WORD_RE.findall(text)) + "%")
    183             fallback_params.append(amount)
    184             rows = connection.execute(
    185                 "SELECT e.payload, e.id, e.kind, e.title, e.reference, e.author, "
    186                 "e.abstract, e.source_year, e.source FROM entries AS e WHERE "
    187                 + " AND ".join(fallback_where)
    188                 + " ORDER BY e.title COLLATE NOCASE, e.id LIMIT ?",
    189                 fallback_params,
    190             ).fetchall()
    191         entries = [_entry_from_row(row) for row in rows]
    192         return {
    193             "entries": entries,
    194             "metadata": _metadata(path, text, amount, len(entries)),
    195         }
    196     except ResearchUnavailable:
    197         raise
    198     except (OSError, sqlite3.Error) as exc:
    199         raise ResearchUnavailable("catálogo local indisponível") from exc
    200     finally:
    201         if connection is not None:
    202             connection.close()

    205 def entries_by_ids(source_ids: list[str], catalog_file: str | os.PathLike[str] | None = None) -> list[dict[str, Any]]:
    206     """Retorna somente registros canônicos do catálogo read-only."""
    207     ids = list(dict.fromkeys(value.strip() for value in source_ids if isinstance(value, str) and value.strip()))[:100]
    208     if not ids:
    209         return []
    210     path = catalog_path(catalog_file)
    211     if not path.is_file():
    212         raise ResearchUnavailable("catálogo local não montado")
    213     placeholders = ",".join("?" for _ in ids)
    214     connection = None
    215     try:
    216         uri = f"file:{quote(str(path.resolve()), safe='/')}?mode=ro"
    217         connection = sqlite3.connect(uri, uri=True)
    218         connection.row_factory = sqlite3.Row
    219         row = connection.execute("SELECT name FROM sqlite_master WHERE type = 'table' AND name = 'entries'").fetchone()
    220         if row is None:
    221             raise ResearchUnavailable("catálogo sem tabela de entradas")
    222         rows = connection.execute(
    223             "SELECT payload, id, kind, title, reference, author, abstract, source_year, source FROM entries WHERE id IN (" + placeholders + ")",
    224             ids,
    225         ).fetchall()
    226         by_id = {_entry_from_row(row)["id"]: _entry_from_row(row) for row in rows}
    227         return [by_id[source_id] for source_id in ids if source_id in by_id]
    228     except ResearchUnavailable:
    229         raise
    230     except (OSError, sqlite3.Error) as exc:
    231         raise ResearchUnavailable("catálogo local indisponível") from exc
    232     finally:
    233         if connection is not None:
    234             connection.close()
