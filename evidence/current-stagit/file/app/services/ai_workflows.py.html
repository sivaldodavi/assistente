     92 def action_context(project: dict, action: str, body: dict, *, sources: list[dict] | None = None) -> dict:
     93     state = legacy_state(project, sources=sources)
     94     fields = project.get("fields") if isinstance(project.get("fields"), dict) else {}
     95     project_type = "paper" if project.get("project_type") in {"paper_project", "paper_final"} else "case"
     96     context_fields = body.get("context_fields")
     97     if isinstance(context_fields, list):
     98         selected = [key for key in context_fields if isinstance(key, str) and key]
     99     else:
    100         selected = list(fields)
    101     project_context = {key: fields.get(key, "") for key in selected}
    102     context = {
    103         **state,
    104         "action": action,
    105         "message": str(body.get("message") or "")[:6000],
    106         "instruction": str(body.get("instruction") or "")[:3000],
    107         "messages": body.get("messages", [])[-12:] if isinstance(body.get("messages"), list) else [],
    108         "field_id": str(body.get("field_id") or "")[:120],
    109         "project_type": project_type,
    110         "project_context": project_context,
    111         "source_ids": project.get("source_ids", []) if isinstance(project.get("source_ids"), list) else [],
    112         "development_ids": [item.get("id") for item in project.get("development_sections", []) if isinstance(item, dict)],
    113         "criterion_ids": [item.get("id") for item in project.get("criteria", []) if isinstance(item, dict)],
    114         "checklist": body.get("checklist", []) if isinstance(body.get("checklist"), list) else [],
    115         "evidence_fields": [str(item) for item in selected[:20]],
    116     }
    117     if context.get("field_id") in fields:
    118         context["selected_text"] = _text(fields, context["field_id"])
    119     if body.get("allow_web") is True:
    120         context["search_web"] = False
    121     return context

    161 def validate_action_result(result: object, context: dict) -> dict:
    162     if not isinstance(result, dict):
    163         raise ValueError("resposta IA não é objeto")
    164     expected = {"answer", "suggestions", "rewrite", "checklist"}
    165     if set(result) != expected:
    166         raise ValueError("resposta IA fora do contrato")
    167     answer = result["answer"]
    168     rewrite = result["rewrite"]
    169     suggestions = result["suggestions"]
    170     checklist = result["checklist"]
    171     if not isinstance(answer, str) or not answer.strip() or len(answer) > 3000:
    172         raise ValueError("answer inválido")
    173     if not isinstance(rewrite, str) or len(rewrite) > 12000:
    174         raise ValueError("rewrite inválido")
    175     clean_suggestions = ai_support.validate_suggestion_items(
    176         context.get("action", ""), suggestions,
    177         project_type=context.get("project_type", "case"),
    178         project_context=context.get("project_context"),
    179         source_ids=context.get("source_ids"),
    180         development_ids=context.get("development_ids"),
    181         criterion_ids=context.get("criterion_ids"),
    182     )
    183     if len(clean_suggestions) > 4:
    184         raise ValueError("suggestions demais")
    185     expected_ids = {item.get("id") for item in context.get("checklist", []) if isinstance(item, dict)}
    186     clean_checklist = []
    187     if not isinstance(checklist, list) or len(checklist) > 30:
    188         raise ValueError("checklist inválida")
    189     for item in checklist:
    190         if not isinstance(item, dict) or set(item) != {"id", "confidence", "evidence"}:
    191             raise ValueError("item de checklist inválido")
    192         if item["id"] not in expected_ids or not _valid_number(item["confidence"], 0, 1):
    193             raise ValueError("item de checklist inválido")
    194         if not isinstance(item["evidence"], str) or len(item["evidence"]) > 500:
    195             raise ValueError("evidência de checklist inválida")
    196         clean_checklist.append({"id": item["id"], "confidence": item["confidence"], "evidence": item["evidence"].strip()})
    197     if context.get("action") != "checklist" and clean_checklist:
    198         raise ValueError("checklist inesperada")
    199     return {"answer": answer.strip(), "suggestions": clean_suggestions, "rewrite": rewrite.strip(), "checklist": clean_checklist}

    206 def jev_questions(section: str | None = None) -> dict:
    207     if section:
    208         definition = AI_CASE_CHECKLISTS.get(section)
    209         if definition is None:
    210             raise ValueError("etapa JEV inválida")
    211         return {
    212             check_id: {"type": "score", "instructions": item["instructions"], "criteria": AI_CASE_CHECKLIST_LEVELS}
    213             for check_id, item in definition["checks"].items()
    214         }
    215     questions = {
    216         question_id: {"type": "score", "instructions": definition["instructions"], "criteria": AI_CASE_SCORE_LEVELS}
    217         for question_id, definition in AI_CASE_RUBRIC.items()
    218     }
    219     questions["grade_range"] = {"type": "score", "instructions": AI_CASE_GRADE_QUESTION["instructions"], "criteria": AI_CASE_GRADE_RANGES}
    220     return questions

    223 def _read_score(answer: dict) -> dict:
    224     if not isinstance(answer, dict) or not _valid_number(answer.get("score"), 0, 3):
    225         raise ValueError("score JEV inválido")
    226     confidence = answer.get("confidence", 0.0)
    227     if not _valid_number(confidence, 0, 1):
    228         confidence = 0.0
    229     probabilities = answer.get("probabilities", [0.25, 0.25, 0.25, 0.25])
    230     if isinstance(probabilities, dict):
    231         probabilities = [probabilities.get(str(index), 0) for index in range(4)]
    232     if not isinstance(probabilities, list) or len(probabilities) != 4 or not all(_valid_number(item, 0, 1) for item in probabilities):
    233         probabilities = [0.25, 0.25, 0.25, 0.25]
    234     total = sum(probabilities)
    235     if total <= 0:
    236         probabilities = [0.25, 0.25, 0.25, 0.25]
    237     elif not 0.97 <= total <= 1.03:
    238         probabilities = [item / total for item in probabilities]
    239     observation = answer.get("observation", "")
    240     return {"score": answer["score"], "confidence": confidence, "probabilities": probabilities, "observation": str(observation or "")[:1000]}

    243 def normalize_jev_result(raw: dict, section: str | None = None) -> dict:
    244     answers = raw.get("answers") if isinstance(raw, dict) else None
    245     if not isinstance(answers, dict):
    246         raise ValueError("respostas JEV ausentes")
    247     if section:
    248         definition = AI_CASE_CHECKLISTS[section]
    249         criteria = []
    250         for check_id, item in definition["checks"].items():
    251             score = _read_score(answers.get(check_id, {}))
    252             criteria.append({"id": check_id, "label": item["label"], **score})
    253         return {
    254             "model": raw.get("model") or "jev-latest",
    255             "section": section,
    256             "criteria": criteria,
    257             "probability_threshold": AI_CASE_PROBABILITY_THRESHOLD,
    258             "unassessed_checks": list(AI_CASE_UNASSESSABLE_CHECKS.get(section, ())),
    259             "usage": raw.get("usage") or {},
    260         }
    261     criteria = []
    262     expected_points = 0.0
    263     for question_id, definition in AI_CASE_RUBRIC.items():
    264         score = _read_score(answers.get(question_id, {}))
    265         probabilities = score["probabilities"]
    266         expected_points += definition["max_points"] * ((probabilities[1] + probabilities[2]) / 2 + probabilities[3])
    267         criteria.append({"id": question_id, "title": definition["title"], "max_points": definition["max_points"], **score})
    268     grade = _read_score(answers.get("grade_range", {}))
    269     return {
    270         "model": raw.get("model") or "jev-latest",
    271         "criteria": criteria,
    272         "expected_points": round(expected_points, 2),
    273         "assessable_points": sum(item["max_points"] for item in AI_CASE_RUBRIC.values()),
    274         "grade_range": {"title": AI_CASE_GRADE_QUESTION["title"], **grade},
    275         "probability_threshold": AI_CASE_PROBABILITY_THRESHOLD,
    276         "unassessed_points": 0.5,
    277         "usage": raw.get("usage") or {},
    278     }

    281 def normalized_jev_suggestion(action: str, context: dict, raw: dict) -> dict:
    282     answers = raw.get("answers") if isinstance(raw, dict) else None
    283     if not isinstance(answers, dict):
    284         raise ValueError("respostas JEV ausentes")
    285     criteria = ai_support.JEV_SUGGESTION_CRITERIA.get(action)
    286     if not criteria:
    287         return {"status": "nao_avaliado", "observations": [], "proposal_present": bool(context.get("assignment_prompt"))}
    288     scores = []
    289     observations = []
    290     for criterion_id, _title in criteria:
    291         item = _read_score(answers.get(criterion_id, {}))
    292         scores.append(item["score"])
    293         observations.append(item["observation"] or f"Confira o critério {criterion_id} no texto antes de aplicar a sugestão.")
    294     proposal_present = bool(context.get("assignment_prompt"))
    295     return {"status": "nao_avaliado" if not proposal_present else ("adequado" if min(scores) >= 2 else "revisar"), "observations": observations, "proposal_present": proposal_present}
