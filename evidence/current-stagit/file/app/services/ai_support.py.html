    136 def suggestion_field_ids(
    137     action: str,
    138     project_type: str = "case",
    139     project_context: dict | None = None,
    140     *,
    141     source_ids: object = None,
    142     development_ids: object = None,
    143     criterion_ids: object = None,
    144 ) -> tuple[str, ...]:
    145     """Return only IDs that the actual project/frontend can address.
    146 
    147     ``project_context`` is the saved field map.  When present, aliases are
    148     resolved by retaining the ID that really exists (for example
    149     ``paperProblem`` instead of inventing ``problem``).  Dynamic forms such as
    150     sources/development/criteria are returned only with their saved IDs.
    151     """
    152     if action not in SUGGESTION_ACTIONS:
    153         return ()
    154     project_context = project_context if isinstance(project_context, dict) else None
    155     candidates = (_PAPER_FIELDS if project_type == "paper" else _CASE_FIELDS).get(action, ())
    156     if action == "review_references":
    157         values = source_ids if isinstance(source_ids, (list, tuple, set)) else ()
    158         result = tuple(
    159             f"source:{value}" for value in values
    160             if isinstance(value, str) and value and ":" not in value
    161         )
    162     elif action == "review_development":
    163         values = development_ids if isinstance(development_ids, (list, tuple, set)) else ()
    164         result = tuple(
    165             field_id
    166             for value in values
    167             if isinstance(value, str) and value and ":" not in value
    168             for field_id in (f"development:{value}", f"developmentText:{value}")
    169         )
    170     elif action == "review_criteria":
    171         values = criterion_ids if isinstance(criterion_ids, (list, tuple, set)) else ()
    172         result = tuple(
    173             f"criterion:{value}:{property_name}"
    174             for value in values
    175             if isinstance(value, str) and value and ":" not in value
    176             for property_name in ("term", "definition", "basis")
    177         )
    178     elif project_context is None:
    179         result = tuple(candidates)
    180     else:
    181         result = tuple(field_id for field_id in candidates if field_id in project_context)
    182     return tuple(field_id for field_id in result if field_id not in _UNSAFE_SUGGESTION_FIELD_IDS)

    185 def normalize_request_id(value: object) -> str | None:
    186     """Aceita somente UUID textual para idempotência de uma ação IA."""
    187     if not isinstance(value, str):
    188         return None
    189     import uuid
    190 
    191     try:
    192         parsed = uuid.UUID(value)
    193     except (ValueError, AttributeError, TypeError):
    194         return None
    195     return str(parsed)

    198 def validate_suggestion_items(
    199     action: str,
    200     suggestions: object,
    201     *,
    202     project_type: str = "case",
    203     project_context: dict | None = None,
    204     source_ids: object = None,
    205     development_ids: object = None,
    206     criterion_ids: object = None,
    207 ) -> list[dict[str, str]]:
    208     if not isinstance(suggestions, list) or len(suggestions) > 20:
    209         raise ValueError("suggestions inválidas")
    210     allowed = set(
    211         suggestion_field_ids(
    212             action,
    213             project_type,
    214             project_context,
    215             source_ids=source_ids,
    216             development_ids=development_ids,
    217             criterion_ids=criterion_ids,
    218         )
    219     )
    220     result = []
    221     seen = set()
    222     for item in suggestions:
    223         if not isinstance(item, dict) or set(item) != {"field_id", "text"}:
    224             raise ValueError("suggestion inválida")
    225         field_id = item["field_id"]
    226         text = item["text"]
    227         if (
    228             not isinstance(field_id, str)
    229             or field_id not in allowed
    230             or field_id in seen
    231             or field_id in _UNSAFE_SUGGESTION_FIELD_IDS
    232             or not isinstance(text, str)
    233             or not text.strip()
    234             or len(text) > 2000
    235         ):
    236             raise ValueError("suggestion inválida")
    237         if ":" not in field_id and not _SAFE_DIRECT_FIELD.fullmatch(field_id):
    238             raise ValueError("suggestion inválida")
    239         seen.add(field_id)
    240         result.append({"field_id": field_id, "text": text.strip()})
    241     return result

    249 def suggestion_validation_from_jev(
    250     action: str, context: dict, answers: dict
    251 ) -> ValidationResult:
    252     """Converte respostas score reais do JEV em status de UX.
    253 
    254     Não cria evidência local: somente observações textuais retornadas pelo JEV
    255     são expostas. A ausência de proposta impede o rótulo ``adequado`` mesmo
    256     quando clareza foi avaliada.
    257     """
    258     criteria = JEV_SUGGESTION_CRITERIA.get(action)
    259     if not criteria or not isinstance(answers, dict):
    260         raise ValueError("respostas JEV ausentes")
    261     observations = []
    262     scores = []
    263     for criterion_id, _title in criteria:
    264         answer = answers.get(criterion_id)
    265         if not isinstance(answer, dict):
    266             raise ValueError("critério JEV ausente")
    267         score = answer.get("score")
    268         confidence = answer.get("confidence")
    269         if (
    270             isinstance(score, bool)
    271             or not isinstance(score, (int, float))
    272             or not 0 <= score <= 3
    273             or isinstance(confidence, bool)
    274             or not isinstance(confidence, (int, float))
    275             or not 0 <= confidence <= 1
    276         ):
    277             raise ValueError("score JEV inválido")
    278         scores.append(score)
    279         observation = answer.get("observation")
    280         if isinstance(observation, str) and observation.strip():
    281             observations.append(observation.strip())
    282         else:
    283             observations.append(
    284                 f"O JEV não retornou observação textual para o critério {criterion_id}; "
    285                 "confira a evidência no texto antes de aplicar a sugestão."
    286             )
    287     proposal_present = bool(proposal_text(context))
    288     if not proposal_present:
    289         observations.append(
    290             "A proposta do professor não foi fornecida. Esclareça a atividade para avaliar adequação."
    291         )
    292         status = "nao_avaliado"
    293     else:
    294         status = "adequado" if min(scores) >= 2 else "revisar"
    295     return ValidationResult(status, observations, proposal_present)

    298 def validate_safe_web_request(value: object) -> dict | None:
    299     """Valida uma capacidade web opcional sem aceitar URL arbitrária."""
    300     if value in (None, False):
    301         return None
    302     if value is True:
    303         raise ValueError("a busca web exige backend permitido")
    304     if not isinstance(value, dict) or value.get("enabled") is not True:
    305         raise ValueError("configuração de busca web inválida")
    306     query = value.get("query")
    307     if not isinstance(query, str) or not query.strip():
    308         raise ValueError("consulta web inválida")
    309     domains = value.get("allowed_domains", [])
    310     if not isinstance(domains, list) or not domains or any(
    311         not isinstance(domain, str) or not domain.strip() for domain in domains
    312     ):
    313         raise ValueError("domínios web permitidos ausentes")
    314     normalized = []
    315     for domain in domains:
    316         hostname = domain.strip().casefold().rstrip(".")
    317         if "/" in hostname or ":" in hostname or "@" in hostname:
    318             raise ValueError("domínio web inválido")
    319         try:
    320             address = ipaddress.ip_address(hostname)
    321         except ValueError:
    322             address = None
    323         if address is not None and (
    324             address.is_private or address.is_loopback or address.is_link_local
    325         ):
    326             raise ValueError("domínio web privado")
    327         normalized.append(hostname)
    328     return {"query": query.strip(), "allowed_domains": normalized}

    331 def validate_backend_url(url: object, allowed_domains: list[str]) -> str:
    332     """Confere URL retornada por backend: HTTPS, host permitido e não privado."""
    333     if not isinstance(url, str):
    334         raise ValueError("URL web inválida")
    335     parsed = urlsplit(url)
    336     if parsed.scheme != "https" or not parsed.hostname or parsed.username or parsed.password:
    337         raise ValueError("URL web insegura")
    338     hostname = parsed.hostname.casefold().rstrip(".")
    339     if not any(hostname == domain or hostname.endswith("." + domain) for domain in allowed_domains):
    340         raise ValueError("URL fora da lista permitida")
    341     try:
    342         addresses = socket.getaddrinfo(hostname, 443, type=socket.SOCK_STREAM)
    343     except OSError as error:
    344         raise ValueError("host web indisponível") from error
    345     for address in addresses:
    346         ip = ipaddress.ip_address(address[4][0])
    347         if ip.is_private or ip.is_loopback or ip.is_link_local or ip.is_reserved:
    348             raise ValueError("URL aponta para rede privada")
    349     return url

    352 def research_entries_as_references(entries: object) -> list[dict]:
    353     """Conserva somente metadados e IDs retornados pelo catálogo local."""
    354     if not isinstance(entries, list):
    355         raise ValueError("catálogo inválido")
    356     references = []
    357     for entry in entries:
    358         if not isinstance(entry, dict) or not isinstance(entry.get("id"), str):
    359             raise ValueError("entrada de catálogo inválida")
    360         reference = {
    361             key: entry[key]
    362             for key in ("id", "kind", "title", "reference", "author", "abstract", "source_year", "source")
    363             if key in entry and entry[key] is not None
    364         }
    365         references.append(reference)
    366     return references
