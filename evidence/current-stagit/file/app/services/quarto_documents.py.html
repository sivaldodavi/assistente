     43 def _text(value: Any) -> str:
     44     if not isinstance(value, str):
     45         return ""
     46     # Remove control bytes that have no useful representation in a document.
     47     return re.sub(r"[\x00-\x08\x0b\x0c\x0e-\x1f\x7f]", "", value).replace("\r\n", "\n").replace("\r", "\n")

     56     """Encode a Typst-escaped string as a YAML/JSON-compatible quoted scalar."""

     61 def _markdown_text(value: Any) -> str:
     62     """Escape plain editor text so it cannot become Markdown, HTML, or a raw block."""
     63     value = _text(value)
     64     # Pandoc/Quarto Markdown treats backslash-escaped punctuation literally.
     65     return re.sub(r"([\\`*_{}\[\]<>()#+.!|~&+-])", r"\\\1", value)

     68 def _paragraphs(value: Any) -> str:
     69     paragraphs = re.split(r"\n[ \t]*\n+", _text(value).strip())
     70     rendered = []
     71     for paragraph in paragraphs:
     72         if not paragraph.strip():
     73             continue
     74         lines = paragraph.split("\n")
     75         rendered.append("  \n".join(_markdown_text(line) for line in lines))
     76     return "\n\n".join(rendered)

     94 def _brazilian_date(value: Any) -> str:
     95     value = _text(value).strip()
     96     match = re.fullmatch(r"(\d{4})-(\d{2})-(\d{2})", value)
     97     return f"{match.group(3)}/{match.group(2)}/{match.group(1)}" if match else value

    100 def _metadata(project: dict[str, Any]) -> str:
    101     fields = project["fields"]
    102     facsure = fields.get("profile") == "facsure"
    103     authors = _authors(project)
    104     institution = _text(fields.get("institution")) if facsure else ""
    105     course = _text(fields.get("course")) if facsure else ""
    106     schedule = project.get("schedule") if isinstance(project.get("schedule"), dict) else {}
    107     if project.get("projectType") == "paper" and fields.get("paperKind") == "final":
    108         delivery = schedule.get("finalDate") or fields.get("deliveryDate") or schedule.get("deliveryDate")
    109     else:
    110         delivery = fields.get("deliveryDate") or schedule.get("deliveryDate")
    111     details = [
    112         f"Disciplina: {_text(fields.get('discipline'))}" if _text(fields.get("discipline")).strip() else "",
    113         f"Professor(a): {_text(fields.get('teacher'))}" if _text(fields.get("teacher")).strip() else "",
    114         f"Turma: {_text(fields.get('classInfo'))}" if _text(fields.get("classInfo")).strip() else "",
    115         f"Data de entrega: {_brazilian_date(delivery)}" if _text(delivery).strip() else "",
    116     ]
    117     details = [item for item in details if item]
    118     title = _text(fields.get("title")).strip()
    119     year = _text(fields.get("year"))
    120     city = _text(fields.get("city"))
    121     subtitle = _text(fields.get("subtitle") or fields.get("subTitle"))
    122     metadata = [
    123         f"title: {_yaml_typst_scalar(title)}",
    124         "lang: pt-BR",
    125         "abnt-capa: true",
    126         "abnt-folha-rosto: false",
    127         f"abnt-instituicao: {_yaml_typst_scalar(institution)}",
    128         f"abnt-curso: {_yaml_typst_scalar(course)}",
    129         f"abnt-subtitulo: {_yaml_typst_scalar(subtitle)}",
    130         _yaml_list("abnt-autores", authors),
    131         _yaml_list("abnt-dados", details),
    132         f"abnt-local: {_yaml_typst_scalar(city)}",
    133         f"abnt-ano: {_yaml_typst_scalar(year)}",
    134         'abnt-numeracao: "1"',
    135         'abnt-numeracao-secoes: "1.1"',
    136         "abnt-sem-quebra-secoes: true",
    137         'abnt-fonte: "Liberation Sans"',
    138         "format: abnt-typst",
    139     ]
    140     return "---\n" + "\n".join(metadata) + "\n---\n"

    181 def _article_header(project: dict[str, Any]) -> str:
    182     fields = project["fields"]
    183     title = _text(fields.get("title")).strip().upper()
    184     subtitle = _text(fields.get("subtitle") or fields.get("subTitle")).strip()

    261 def _case_possible_solutions(fields: dict[str, Any]) -> str:
    262     blocks = [_paragraphs(fields.get("decisionBasis"))]
    263     for marker, key, ending in (("a", "decisionA", ";"), ("b", "decisionB", ".")):
    264         content = _text(fields.get(key)).strip()
    265         if not content:
    266             continue
    267         content = re.sub(r"[.;?!]\s*$", "", content).rstrip() + ending
    268         rendered = _paragraphs(content)
    269         if rendered:
    270             indented = rendered.replace("\n", "\n   ")
    271             blocks.append(f"{marker}) \u2003{indented}")
    272     return "\n\n".join(block for block in blocks if block)

    275 def _case_report(project: dict[str, Any]) -> str:
    276     fields = project["fields"]
    277     criteria = project.get("criteria") if isinstance(project.get("criteria"), list) else []
    278     criterion_text = []
    279     for item in criteria:
    280         if not isinstance(item, dict):
    281             continue
    282         term = _text(item.get("term")).strip()
    283         definition = _text(item.get("definition")).strip()
    284         basis = _text(item.get("basis")).strip()
    285         if not (term or definition or basis):
    286             continue
    287         sentence = f"{term}: {definition}" if term else definition
    288         if basis:
    289             sentence += f" Base normativa ou ética: {basis}"
    290         criterion_text.append(sentence.strip())
    291     sections = [
    292         _section("DESCRIÇÃO DO CASO", "\n\n".join(filter(None, [_text(fields.get("facts")), _text(fields.get("problem"))]))),
    293         "# IDENTIFICAÇÃO E ANÁLISE DO CASO",
    294         _case_possible_solutions(fields),
    295         "## Descrição dos argumentos",
    296         _section(_text(fields.get("thesisATitle")).strip() or "Argumento favorável à tese A", fields.get("thesisA"), 3),
    297         _section(_text(fields.get("thesisBTitle")).strip() or "Argumento favorável à tese B", fields.get("thesisB"), 3),
    298         _section("Descrição dos critérios e valores", "\n\n".join(criterion_text), 2),
    299     ]
    300     bibliography = _bibliography(project)
    301     return "\n\n".join(item for item in sections if item) + ("\n\n" + _raw_typst(bibliography) if bibliography else "")

    304 def _paper_project(project: dict[str, Any]) -> str:
    305     fields = project["fields"]
    306     objective_parts = [
    307         _section("Objetivo Geral", fields.get("objectiveGeneral"), 2),
    308     ]
    309     objectives = [
    310         _text(fields.get(f"objective{number}")).strip()
    311         for number in range(1, 4)
    312         if _text(fields.get(f"objective{number}")).strip()
    313     ]
    314     if objectives:
    315         objective_parts.append("## Objetivos específicos\n\n" + "\n\n".join(f"{number}. {_markdown_text(text)}" for number, text in enumerate(objectives, 1)))
    316     methodology = "\n\n".join(
    317         item for item in [
    318             f"Autor de metodologia adotado: {_text(fields.get('methodAuthor'))}." if _text(fields.get("methodAuthor")).strip() else "",
    319             f"Natureza: {_text(fields.get('nature'))}." if _text(fields.get("nature")).strip() else "",
    320             f"Abordagem: {_text(fields.get('approach'))}." if _text(fields.get("approach")).strip() else "",
    321             f"Fins: {_text(fields.get('ends'))}." if _text(fields.get("ends")).strip() else "",
    322             f"Procedimentos: {_text(fields.get('procedure'))}." if _text(fields.get("procedure")).strip() else "",
    323             _text(fields.get("instruments")),
    324         ] if item
    325     )
    326     sections = [
    327         _section("TEMA", fields.get("area")),
    328         _section("TÍTULO E DELIMITAÇÃO", "\n\n".join(filter(None, [_text(fields.get("delimitation")), _text(fields.get("researchContext")), _text(fields.get("researchObject"))]))),
    329         _section("PROBLEMA", "\n\n".join(filter(None, [_text(fields.get("problemContext")), _text(fields.get("problem"))]))),
    330         _section("HIPÓTESE", fields.get("hypothesis")),
    331         _section("JUSTIFICATIVA", fields.get("justification")),
    332         "# OBJETIVOS\n\n" + "\n\n".join(item for item in objective_parts if item),
    333         _section("REFERENCIAL TEÓRICO", fields.get("theory")),
    334         _section("METODOLOGIA", methodology),
    335     ]
    336     body = "\n\n".join(item for item in sections if item)
    337     bibliography = _bibliography(project)
    338     return body + ("\n\n" + _raw_typst(bibliography) if bibliography else "")

    341 def _paper_final(project: dict[str, Any]) -> str:
    342     fields = project["fields"]
    343     sections = [_section("INTRODUÇÃO", fields.get("introduction"))]
    344     development = fields.get("developmentSections") if isinstance(fields.get("developmentSections"), list) else []
    345     for index, item in enumerate(development):
    346         if not isinstance(item, dict):
    347             continue
    348         title = _text(item.get("title")).strip() or f"DESENVOLVIMENTO {index + 1}"
    349         sections.append(_section(title, item.get("content")))
    350     if _text(fields.get("aiNote")).strip():
    351         sections.append(_section("DECLARAÇÃO SOBRE USO DE IA", fields.get("aiNote")))
    352     sections.append(_section("CONCLUSÃO", fields.get("conclusion")))
    353     body = "\n\n".join(item for item in sections if item)
    354     bibliography = _bibliography(project)
    355     return body + ("\n\n" + _raw_typst(bibliography) if bibliography else "")

    358 def build_quarto_source(project: dict[str, Any], document_kind: str) -> str:
    359     """Return a .qmd document using the bundled ABNT Quarto format."""
    360     if not isinstance(project, dict) or not isinstance(project.get("fields"), dict):
    361         raise QuartoDocumentError("Projeto inválido para exportação.")
    362     if document_kind not in _DOCUMENT_KINDS:
    363         raise QuartoDocumentError("Tipo de documento inválido.")
    364     expected_type = _DOCUMENT_KINDS[document_kind]
    365     if project.get("projectType") != expected_type:
    366         raise QuartoDocumentError("O tipo de documento não corresponde ao projeto.")
    367 
    368     front_matter = _metadata(project)
    369     if document_kind == "case_report":
    370         body = _raw_typst(_case_header(project)) + "\n" + _case_report(project)
    371     elif document_kind == "paper_project":
    372         body = _raw_typst(_project_identification(project)) + "\n" + _paper_project(project)
    373     else:
    374         body = _raw_typst(_article_header(project)) + "\n" + _paper_final(project)
    375     return front_matter + "\n" + body.strip() + "\n"

    378 def render_document_pdf(project: dict[str, Any], document_kind: str) -> bytes:
    379     """Render an isolated QMD file with the local extension and return its PDF bytes."""
    380     qmd = build_quarto_source(project, document_kind)
    381     binary = os.environ.get("QUARTO_BINARY", "quarto")
    382     executable = shutil.which(binary)
    383     if not executable or not EXTENSION_DIR.is_dir():
    384         raise QuartoUnavailableError("O renderizador Quarto não está disponível.")
    385     if not _RENDER_SLOTS.acquire(blocking=False):
    386         raise QuartoBusyError("Há muitas renderizações em andamento.")
    387 
    388     try:
    389         with tempfile.TemporaryDirectory(prefix="assistente-quarto-") as temp_name:
    390             workspace = Path(temp_name)
    391             (workspace / "_extensions").mkdir()
    392             shutil.copytree(EXTENSION_DIR, workspace / "_extensions" / "abnt")
    393             source_path = workspace / "document.qmd"
    394             output_path = workspace / "document.pdf"
    395             source_path.write_text(qmd, encoding="utf-8")
    396             home_path = workspace / "home"
    397             home_path.mkdir(mode=0o700)
    398             env = {
    399                 "PATH": os.environ.get("PATH", "/usr/local/bin:/usr/bin:/bin"),
    400                 "HOME": str(home_path),
    401                 "TMPDIR": str(workspace),
    402                 "TMP": str(workspace),
    403                 "TEMP": str(workspace),
    404                 "LANG": os.environ.get("LANG", "C.UTF-8"),
    405                 "LC_ALL": os.environ.get("LC_ALL", "C.UTF-8"),
    406             }
    407             try:
    408                 completed = subprocess.run(
    409                     [executable, "render", source_path.name, "--output", output_path.name],
    410                     cwd=workspace,
    411                     env=env,
    412                     stdin=subprocess.DEVNULL,
    413                     stdout=subprocess.DEVNULL,
    414                     stderr=subprocess.DEVNULL,
    415                     timeout=RENDER_TIMEOUT_SECONDS,
    416                     check=False,
    417                 )
    418             except subprocess.TimeoutExpired as error:
    419                 raise QuartoRenderError("A renderização excedeu o tempo limite.") from error
    420             except OSError as error:
    421                 raise QuartoUnavailableError("Não foi possível iniciar o Quarto.") from error
    422             if completed.returncode != 0 or not output_path.is_file():
    423                 raise QuartoRenderError("O Quarto não conseguiu renderizar este documento.")
    424             if output_path.stat().st_size > MAX_PDF_BYTES:
    425                 raise QuartoRenderError("O PDF gerado excedeu o limite de tamanho.")
    426             pdf = output_path.read_bytes()
    427             if not pdf.startswith(b"%PDF-"):
    428                 raise QuartoRenderError("O renderizador não retornou um PDF válido.")
    429             return pdf
    430     finally:
    431         _RENDER_SLOTS.release()
