     31 def validate_result(result: object) -> dict:
     32     if not isinstance(result, dict):
     33         raise ValueError("resposta não é objeto")
     34     answer = result.get("answer")
     35     suggestions = result.get("suggestions")
     36     if not isinstance(answer, str) or not answer.strip() or len(answer) > 2000:
     37         raise ValueError("answer inválido")
     38     if not isinstance(suggestions, list) or len(suggestions) > 2:
     39         raise ValueError("suggestions inválidas")
     40     clean = []
     41     for item in suggestions:
     42         if not isinstance(item, dict) or set(item) != {"text"} or not isinstance(item["text"], str) or not item["text"].strip() or len(item["text"]) > 500:
     43             raise ValueError("suggestion inválida")
     44         clean.append({"text": item["text"].strip()})
     45     return {"answer": answer.strip(), "suggestions": clean}
