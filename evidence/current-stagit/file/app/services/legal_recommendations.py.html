     61 def recommend(request: dict, project: dict, candidates: list[dict], deepseek_factory, gemini_factory, record_attempt=None) -> dict:
     62     candidates = validate_candidates(candidates)
     63     if not candidates:
     64         return {
     65             "recommendations": {"law": [], "jurisprudence": [], "doctrine": []},
     66             "warnings": ["Nenhum candidato foi encontrado neste recorte do catálogo."],
     67             "provider": {"name": "none", "model": "", "fallback_used": False, "fallback_reason": None},
     68             "usage": {},
     69         }
     70     messages = build_messages(request, project, candidates)
     71     try:
     72         response = deepseek_factory().complete_json(messages, max_tokens=1600)
     73         result = _normalize_provider_result(response, candidates)
     74         if record_attempt:
     75             record_attempt("deepseek", result["model"], "complete", result["usage"])
     76         return _finish(result, "deepseek", False, None, candidates)
     77     except ProviderError as error:
     78         if record_attempt:
     79             record_attempt("deepseek", "", error.code, {})
     80         if not error.retryable and error.code not in _FALLBACK_ERROR_CODES:
     81             raise
     82         fallback_reason = error.code
     83     try:
     84         response = gemini_factory().complete_json(messages, max_tokens=1600)
     85         result = _normalize_provider_result(response, candidates)
     86         if record_attempt:
     87             record_attempt("gemini", result["model"], "complete", result["usage"])
     88         return _finish(result, "gemini", True, fallback_reason, candidates)
     89     except ProviderError as error:
     90         if record_attempt:
     91             record_attempt("gemini", "", error.code, {})
     92         raise

     95 def _finish(result: dict, provider: str, fallback_used: bool, fallback_reason: str | None, candidates: list[dict]) -> dict:
     96     counts = {"legislation": 0, "case-law": 0, "book": 0}
     97     for candidate in candidates:
     98         counts[candidate["kind"]] += 1
     99     if not result["warnings"]:
    100         result["warnings"] = []
    101     result["warnings"] = list(dict.fromkeys(result["warnings"] + [
    102         "Confira vigência, inteiro teor e pertinência antes de citar.",
    103     ]))[:5]
    104     result["provider"] = {
    105         "name": provider,
    106         "model": result.pop("model", ""),
    107         "fallback_used": fallback_used,
    108         "fallback_reason": fallback_reason,
    109     }
    110     result["coverage"] = {
    111         "candidate_count": len(candidates),
    112         "by_kind": counts,
    113         "scope": "catálogo jurídico local; pista inicial",
    114     }
    115     return result
