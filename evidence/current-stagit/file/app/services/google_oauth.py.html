     39 def valid_redirect_uri(value: str) -> bool:
     40     try:
     41         parsed = urlsplit(value)
     42         port = parsed.port
     43         return (
     44             parsed.scheme in {"https", "http"}
     45             and bool(parsed.hostname)
     46             and parsed.path == "/oauth/google/callback"
     47             and parsed.username is None
     48             and parsed.password is None
     49             and not parsed.query
     50             and not parsed.fragment
     51             and (port is None or 1 <= port <= 65535)
     52             and (parsed.scheme == "https" or parsed.hostname in {"localhost", "127.0.0.1", "::1"})
     53         )
     54     except ValueError:
     55         return False

     70 def pkce_challenge(verifier: str) -> str:
     71     digest = hashlib.sha256(verifier.encode("ascii")).digest()

     96     def exchange_code(self, code: str, code_verifier: str, expected_nonce: str) -> dict:
     97         if not self.config.configured:
     98             raise GoogleOAuthError("google_login_unconfigured")
     99         if not code or len(code) > 2048 or not code_verifier:
    100             raise GoogleOAuthError("google_state_failed")
    101         body = urlencode({
    102             "code": code,
    103             "client_id": self.config.client_id,
    104             "client_secret": self.config.client_secret,
    105             "redirect_uri": self.config.redirect_uri,
    106             "grant_type": "authorization_code",
    107             "code_verifier": code_verifier,
    108         }).encode("ascii")
    109         request = Request(
    110             self.config.token_endpoint,
    111             data=body,
    112             headers={"Accept": "application/json", "Content-Type": "application/x-www-form-urlencoded"},
    113             method="POST",
    114         )
    115         try:
    116             with urlopen(request, timeout=10) as response:
    117                 payload_bytes = response.read(64 * 1024 + 1)
    118             if len(payload_bytes) > 64 * 1024:
    119                 raise ValueError("OAuth response too large")
    120             payload = json.loads(payload_bytes)
    121         except (HTTPError, OSError, TimeoutError, ValueError, json.JSONDecodeError) as error:
    122             raise GoogleOAuthError("google_token_exchange_failed") from error
    123         token = payload.get("id_token") if isinstance(payload, dict) else None
    124         if not isinstance(token, str) or not token:
    125             raise GoogleOAuthError("google_token_exchange_failed")
    126         return self.verify_id_token(token, expected_nonce)

    128     def verify_id_token(self, token: str, expected_nonce: str) -> dict:
    129         try:
    130             from google.auth.transport import requests as google_requests
    131             from google.oauth2 import id_token
    132         except ImportError as error:
    133             raise GoogleOAuthError("google_verifier_unavailable") from error
    134         try:
    135             claims = id_token.verify_oauth2_token(
    136                 token,
    137                 google_requests.Request(),
    138                 self.config.client_id,
    139             )
    140         except Exception as error:
    141             raise GoogleOAuthError("invalid_google_credential") from error
    142         if (
    143             not isinstance(claims, dict)
    144             or claims.get("iss") not in {"accounts.google.com", "https://accounts.google.com"}
    145             or claims.get("email_verified") is not True
    146             or not isinstance(claims.get("sub"), str)
    147             or not claims["sub"]
    148             or len(claims["sub"]) > 255
    149             or not isinstance(claims.get("email"), str)
    150             or not claims["email"].strip()
    151             or not isinstance(claims.get("nonce"), str)
    152             or not secrets.compare_digest(claims["nonce"], expected_nonce)
    153         ):
    154             raise GoogleOAuthError("invalid_google_credential")
    155         return {
    156             "sub": claims["sub"],
    157             "email": claims["email"].strip().lower(),
    158             "name": claims.get("name", "") if isinstance(claims.get("name"), str) else "",
    159         }
