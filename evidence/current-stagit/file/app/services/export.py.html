     21 def _paragraphs(value):
     22     text = str(value or "").strip()
     23     if not text:
     24         return '<p class="empty">Não preenchido.</p>'
     25     return "".join(f"<p>{html.escape(part.strip())}</p>" for part in text.split("\n\n") if part.strip())

     35 def _document_sections(project, sources):
     36     """Produz as mesmas seções acadêmicas para HTML e PDF, na ordem do roteiro."""
     37     fields = project.get("fields", {})
     38     result = []
     39     for section in _sections(project):
     40         custom = section.get("custom")
     41         if custom == "review":
     42             continue
     43         content = [(field["label"], fields.get(field["id"], "")) for field in section["fields"] if field["id"] not in {"title", "authors"}]
     44         if custom == "criteria":
     45             content = [(item.get("title", "Critério"), (item.get("definition", "") + "\n\nBase: " + item.get("basis", ""))) for item in project.get("criteria", [])]
     46         elif custom == "dissertation":
     47             development = [(item.get("title", "Tópico"), item.get("body", "")) for item in project.get("development_sections", [])]
     48             if not development:
     49                 old = fields.get("dissertation_foundation") or fields.get("development")
     50                 if old:
     51                     development = [("Fundamentação", old)]
     52             content = [
     53                 ("Introdução", fields.get("dissertation_intro", "")),
     54                 *development,
     55                 ("Conclusão", fields.get("dissertation_conclusion", "")),
     56             ]
     57         elif custom == "development":
     58             content = [(item.get("title", "Tópico"), item.get("body", "")) for item in project.get("development_sections", [])]
     59             if not content:
     60                 old = fields.get("dissertation_foundation") or fields.get("development")
     61                 if old:
     62                     content = [("Fundamentação", old)]
     63         elif custom == "sources":
     64             content = [("", source.get("reference") or " — ".join(value for value in (source.get("author"), source.get("title")) if value)) for source in sorted(sources, key=lambda item: (item.get("author") or item.get("title") or "").casefold())]
     65         result.append((section["title"], content))
     66     return result

     69 def build_project_html(project, sources):
     70     fields = project.get("fields", {})
     71     title = fields.get("title") or project.get("title") or "Trabalho sem título"
     72     authors = fields.get("dissertation_author") if project.get("export_phase") == "dissertation" else fields.get("authors")
     73     authors = authors or "Autor não informado"
     74     parts = []
     75     for heading, content in _document_sections(project, sources):
     76         body = "".join((f"<h3>{html.escape(label)}</h3>" if label else "") + _paragraphs(value) for label, value in content)
     77         parts.append(f"<section><h2>{html.escape(heading)}</h2>{body or '<p class=empty>Não preenchido.</p>'}</section>")
     78     workflow = "".join(parts)
     79     return f'''<!doctype html>
     80 <html lang="pt-BR"><head><meta charset="utf-8"><title>{html.escape(title)}</title>
     81 <style>body{{font:16px/1.6 Georgia,serif;max-width:850px;margin:3rem auto;padding:0 1.5rem;color:#171717}}h1,h2,h3{{font-family:system-ui,sans-serif}}h1{{font-size:2.3rem}}h2{{margin-top:2.5rem;border-bottom:1px solid #bbb;padding-bottom:.35rem}}.meta{{color:#555}}.empty{{color:#666;font-style:italic}}</style></head>
     82 <body><header><h1>{html.escape(title)}</h1><p class="meta">{html.escape(authors).replace(chr(10), '<br>')}</p></header>{workflow}</body></html>'''
