     21 def _demo_workflow_key(model_id, workflow):
     22     if model_id == "civil-personality":
     23         return "case", "report"
     24     if model_id == "projeto-paper-modelo":
     25         return "paper_project", "paper_project"

     33 def _normalise_demo_title(value):
     34     value = unicodedata.normalize("NFD", str(value or "").casefold())
     35     value = "".join(character for character in value if unicodedata.category(character) != "Mn")
     36     return " ".join(value.replace("·", " ").replace(".", " ").split())

     39 def _demo_stage_sections(model_id, stage_title, sections):
     40     """Relaciona o conteúdo preenchido ao mesmo estágio usado no editor.
     41 
     42     Os modelos legados têm subtítulos vazios (por exemplo, cabeçalhos de
     43     agrupamento). Por isso, a associação não deve depender da posição bruta
     44     da lista de parágrafos: o título do estágio e o título textual da seção
     45     são as identidades estáveis.
     46     """
     47     maps = {
     48         "civil-personality": {
     49             "1 · Descrição do caso": ("1 descricao do caso",),
     50             "2.1 · Decisões possíveis": ("2 1 descricao das decisoes possiveis",),
     51             "2.2.1 · Argumentação da tese A": ("2 2 1",),
     52             "2.2.2 · Argumentação da tese B": ("2 2 2",),
     53             "2.3 · Critérios e valores": ("2 3 descricao dos criterios e valores",),
     54         },
     55         "projeto-paper-modelo": {
     56             "Identificação e delimitação": ("1 tema", "2 titulo e delimitacao do tema"),
     57             "Problema de pesquisa": ("3 problema",),
     58             "Hipótese": ("4 hipoteses",),
     59             "Justificativa": ("5 justificativa",),
     60             "Objetivos": ("6 objetivos gerais e especificos",),
     61             "Referencial teórico": ("7 referencial teorico",),
     62             "Metodologia": ("8 metodologia",),
     63             "Referências do projeto": ("9 referencias",),
     64         },
     65         "paper-final-modelo": {
     66             "Introdução": ("1 introducao",),
     67             "Desenvolvimento": ("2 o inquerito policial", "3 a lei", "4 valor probatorio"),
     68             "Conclusão": ("5 conclusao",),
     69             "Referências finais": ("referencias",),
     70         },
     71     }
     72     patterns = maps.get(model_id, {}).get(stage_title, ())
     73     if not patterns:
     74         return ()
     75     normalized_sections = [_normalise_demo_title(section.get("title")) for section in sections]
     76     return tuple(
     77         index
     78         for index, title in enumerate(normalized_sections)
     79         if any(title.startswith(pattern) for pattern in patterns)
     80     )

     83 def _build_demo_stages(model_id, workflow, sections):
     84     workflow_key, phase = _demo_workflow_key(model_id, workflow)
     85     values = workflow.get("fields", {}) if isinstance(workflow, dict) else {}
     86     stages = []
     87     for index, original in enumerate(workflow_sections(workflow_key)):
     88         if original.get("phase", "report") != phase:
     89             continue
     90         stage = deepcopy(original)
     91         fields = []
     92         for field in stage.get("fields", []):
     93             field = deepcopy(field)
     94             field_id = str(field.get("id") or "")
     95             candidates = [field_id, _camel_case(field_id)]
     96             if field_id == "authors":
     97                 candidates.insert(0, "author")
     98             field["value"] = next((values[key] for key in candidates if key in values), "")
     99             fields.append(field)
    100         stage["fields"] = fields
    101         if stage.get("custom") == "criteria":
    102             stage["criteria_text"] = values.get("criteriaText", "")
    103         if stage.get("custom") == "development":
    104             stage["development_sections"] = values.get("developmentSections", [])
    105         stage["demo_sections"] = [
    106             deepcopy(sections[item])
    107             for item in _demo_stage_sections(model_id, stage.get("title", ""), sections)
    108             if item < len(sections)
    109         ]
    110         stages.append(stage)
    111     return stages

    117 def _load_demo_models():
    118     source = Path(__file__).resolve().parents[1] / "domain" / "public-models.json"
    119     raw_models = json.loads(source.read_text(encoding="utf-8"))
    120     models = []
    121     for raw in raw_models:
    122         raw_sections = []
    123         for section in raw.get("sections", []):
    124             paragraphs = []
    125             for item in section.get("paragraphs", []):
    126                 if isinstance(item, dict) and item.get("quote"):
    127                     value = f'“{item["quote"]}”'
    128                 else:
    129                     value = str(item)
    130                 if value.strip():
    131                     paragraphs.append(value)
    132             paragraphs = tuple(paragraphs)
    133             raw_sections.append({"title": section.get("title", "Seção"), "paragraphs": paragraphs})
    134         sections = [section for section in raw_sections if section["paragraphs"]]
    135         model_id = raw.get("id", "")
    136         if model_id == "civil-personality":
    137             question_label = "Questão do caso"
    138             question = next((paragraph for section in sections for paragraph in section["paragraphs"] if "quais são os limites" in paragraph.lower()), "")
    139         elif model_id == "projeto-paper-modelo":
    140             question_label = "Problema de pesquisa"
    141             question = next((section["paragraphs"][-1] for section in sections if "problema" in section["title"].lower() and section["paragraphs"]), "")
    142         else:
    143             question_label = "Resumo"
    144             question = raw.get("abstract", "")
    145         workflow = deepcopy(raw.get("workflow", {}))
    146         field_values = dict(workflow.get("fields", {}))
    147         field_values.setdefault("title", raw.get("title", ""))
    148         field_values.setdefault("author", raw.get("authors", ""))
    149         field_values.setdefault("discipline", raw.get("discipline", ""))
    150         field_values.setdefault("teacher", raw.get("teacher", ""))
    151         field_values.setdefault("city", raw.get("city", ""))
    152         field_values.setdefault("year", raw.get("year", ""))
    153         class_info = raw.get("classInfo", "")
    154         if class_info:
    155             field_values.setdefault("classInfo", class_info)
    156             field_values.setdefault("shift", class_info.split(",", 1)[0].strip())
    157             if "," in class_info:
    158                 field_values.setdefault("period", class_info.split(",", 1)[1].strip())
    159         workflow["fields"] = field_values
    160         models.append({
    161             "id": model_id,
    162             "type_label": raw.get("typeLabel", "Modelo preenchido"),
    163             "title": raw.get("title", ""),
    164             "authors": raw.get("authors", ""),
    165             "discipline": raw.get("discipline", ""),
    166             "teacher": raw.get("teacher", ""),
    167             "class_info": raw.get("classInfo", ""),
    168             "delivery_date": raw.get("deliveryDate", ""),
    169             "city": raw.get("city", ""),
    170             "year": raw.get("year", ""),
    171             "question_label": question_label,
    172             "question": question,
    173             "summary": raw.get("summary", "Modelo preenchido em modo somente leitura."),
    174             "abstract": raw.get("abstract", ""),
    175             "keywords": raw.get("keywords", []),
    176             "sections": sections,
    177             "citations": raw.get("citations", []),
    178             "sources": raw.get("sources", []),
    179             "workflow": workflow,
    180             "stages": _build_demo_stages(model_id, workflow, raw_sections),
    181         })
    182     return tuple(models)

    250 def research_catalog():
    251     """Catálogo jurídico público com metadata para auxiliar IA/JEV."""
    252     q = request.args.get("q", "").strip()
    253     kind = request.args.get("kind")
    254     limit = request.args.get("limit", 20, type=int)
    255     limit = max(1, min(limit, 100))
    256 
    257     try:
    258         result = search_catalog(q, kind=kind, limit=limit)
    259     except ResearchUnavailable:
    260         return jsonify({"error": "catalog_unavailable", "message": "O catálogo jurídico não está disponível no momento."}), 503
    261 
    262     return jsonify({"ok": True, **result})

    267 def research_catalog_entries():
    268     """Obtém entradas completas do catálogo por IDs."""
    269     body = request.get_json(silent=True) or {}
    270     ids = body.get("ids")
    271     if not isinstance(ids, list) or not ids:
    272         return jsonify({"error": "invalid_ids", "message": "Informe uma lista de IDs."}), 400
    273     ids = [str(id_) for id_ in ids if isinstance(id_, str)][:50]
    274 
    275     try:
    276         entries = entries_by_ids(ids)
    277     except ResearchUnavailable:
    278         return jsonify({"error": "catalog_unavailable", "message": "O catálogo jurídico não está disponível no momento."}), 503
    279 
    280     return jsonify({"ok": True, "entries": entries})
