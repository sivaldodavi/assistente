     54 @account_bp.post("/account/settings/terms")
     55 @login_required
     56 @require_csrf
     57 def save_terms():
     58     if request.form.get("accept") != "on":
     59         abort(400, description="É necessário aceitar os termos para continuar.")
     60     try:
     61         accept_terms(
     62             current_account_id(),
     63             request.form.get("accept") == "on",
     64             request.form.get("ai_consent") == "on",
     65         )
     66     except ValueError as error:
     67         abort(400, description=str(error))
     68     return redirect(url_for("account.settings", saved="terms"))
