     24 @oauth_bp.get("/oauth/google/start")
     25 def google_start():
     26     client = _client()
     27     if not client.config.configured:
     28         return redirect(url_for("public.home", google_login="unconfigured"))
     29     state = new_state()
     30     nonce = new_nonce()

     33     try:
     34         location = client.authorization_url(state, nonce, code_verifier)
     35     except GoogleOAuthError as error:
     36         current_app.logger.warning("google_oauth_start_failed code=%s", str(error))
     37         session.pop("google_oauth", None)
     38         return redirect(url_for("public.home", google_login="failed"))

     43 def google_callback():

     45     state = request.args.get("state", "")
     46     code = request.args.get("code", "")
