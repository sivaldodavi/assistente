     36 @library_bp.post("/library/sources")
     37 @login_required
     38 @require_csrf
     39 def add_source():
     40     title = request.form.get("title", "").strip()
     41     if not title:
     42         abort(400, description="Informe o título da fonte.")
     43     create_source(
     44         current_account_id(),
     45         title,
     46         request.form.get("author", ""),
     47         request.form.get("kind", "other"),
     48         request.form.get("reference", ""),
     49         request.form.get("notes", ""),
     50     )
