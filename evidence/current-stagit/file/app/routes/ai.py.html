     86 def _complete_with_fallback(messages, validator, max_tokens):
     87     first_error = None
     88     try:
     89         client = _deepseek_for_action()
     90         response = client.complete_json(messages, max_tokens=max_tokens)
     91         validator(response.get("result"))
     92         response["provider"] = "deepseek"
     93         return response
     94     except (ProviderError, OSError) as deepseek_error:
     95         if isinstance(deepseek_error, ProviderError) and not _fallback_allowed(deepseek_error):
     96             raise
     97         first_error = deepseek_error if isinstance(deepseek_error, ProviderError) else ProviderError("provider_unreachable", "Não foi possível conectar ao DeepSeek.", 503, retryable=True)
     98     except (ValueError, TypeError) as deepseek_error:
     99         first_error = ProviderError("invalid_response", "A resposta DeepSeek não respeitou o contrato estruturado.", 502, retryable=True)
    100     try:
    101         client = _gemini_for_action()
    102         response = client.complete_json(messages, max_tokens=max_tokens)
    103         validator(response.get("result"))
    104         response["provider"] = "gemini"
    105         response["fallback_reason"] = first_error.code if first_error else "invalid_response"
    106         return response
    107     except (ValueError, TypeError) as error:
    108         raise ProviderError("invalid_response", "DeepSeek e Gemini não respeitaram o contrato estruturado.", 502, retryable=True) from error
    109     except ProviderError as error:
    110         raise error from first_error
