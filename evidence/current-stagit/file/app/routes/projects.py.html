    142     if not isinstance(payload, dict):
    143         abort(400, description="Projeto inválido.")

    145     fields = current_project.get("fields", {}) | fields

    149         normalized_fields = normalize_paper_fields(fields, current_project["project_type"])
    150     else:

    153     valid_project_source_ids = owned_source_ids(current_account_id(), requested_project_source_ids)
    154     project_source_ids = [source_id for source_id in requested_project_source_ids if source_id in valid_project_source_ids]
    155     criteria = normalize_criteria(payload.get("criteria")) if current_project["project_type"] == "case" else []
    156     schedule = normalize_schedule(
    157         payload.get("schedule"),
    158         normalized_fields.get("deliveryDate", ""),
    159         current_project["project_type"],
