    152 def _mask_email(email: str) -> str:
    153     """Mascarar email para LGPD - mostra apenas primeiro e último caractere do local-part."""
    154     if not email or "@" not in email:
    155         return email
    156     local, domain = email.split("@", 1)
    157     if len(local) <= 2:
    158         masked_local = local[0] + "*" * (len(local) - 1)
    159     else:
    160         masked_local = local[0] + "*" * (len(local) - 2) + local[-1]
    161     return f"{masked_local}@{domain}"
