     13 def _configured_secret(file_variable: str, value_variable: str, fallback: str = "", allow_dev_auth: bool = False) -> str:
     14     """Carrega secret de arquivo ou variável de ambiente. Falha se não configurado em produção."""
     15     path = os.environ.get(file_variable, "")
     16     if path:
     17         try:
     18             value = Path(path).read_text(encoding="utf-8").strip()
     19         except OSError:
     20             value = ""
     21         if value:
     22             return value
     23     value = os.environ.get(value_variable, fallback)
     24     if not value and not allow_dev_auth:
     25         raise RuntimeError(f"Secret obrigatório não configurado: {value_variable} ou {file_variable}")
     26     return value

     31 _rate_limit_windows = {}

     34 def _check_rate_limit(action: str, account_id: str) -> bool:
     35     """Verifica se a ação está dentro do limite de rate limiting."""
     36     limit = RATE_LIMITS.get(action)
     37     if limit is None:
     38         return True  # Sem limite definido = permitido
     39 
     40     now = time.monotonic()
     41     cutoff = now - RATE_LIMIT_WINDOW_SECONDS
     42     key = f"{account_id}:{action}"
     43 
     44     with _rate_limit_lock:
     45         timestamps = [ts for ts in _rate_limit_windows.get(key, []) if ts > cutoff]
     46         allowed = len(timestamps) < limit
     47         if allowed:
     48             timestamps.append(now)
     49         _rate_limit_windows[key] = timestamps
     50 
     51         # Limpeza periódica para evitar memory leak (lazy cleanup a cada 1000 requests)
     52         if len(_rate_limit_windows) > _RATE_LIMIT_MAX_KEYS:
     53             for k in list(_rate_limit_windows.keys()):
     54                 _rate_limit_windows[k] = [ts for ts in _rate_limit_windows[k] if ts > cutoff]
     55                 if not _rate_limit_windows[k]:
     56                     del _rate_limit_windows[k]
     57 
     58     return allowed
