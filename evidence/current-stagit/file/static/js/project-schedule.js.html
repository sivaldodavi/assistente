     99 export function collectSchedule(form) {
    100   const schedule = Object.fromEntries(
    101     [...form.querySelectorAll("[data-schedule-field]")].map((element) => [element.dataset.scheduleField, element.value]),
    102   );
    103   schedule.tasks = taskRows(form).map((task) => ({
    104     id: task.dataset.taskId || newTaskId(),
    105     title: task.querySelector('[data-task-field="title"]')?.value || "",
    106     date: task.querySelector('[data-task-field="date"]')?.value || "",
    107     done: Boolean(task.querySelector('[data-task-field="done"]')?.checked),
    108     stage: task.dataset.taskStage === "" ? null : Number(task.dataset.taskStage),
    109     template_key: task.dataset.taskTemplateKey || "",
    110   }));
    111   return schedule;
    112 }

    114 export function initScheduleEditor(form, onStructuralChange) {
    115   if (!form) return;
    116   updateMeta(form);
    117   form.addEventListener("input", (event) => {
    118     if (event.target.matches("[data-schedule-field], [data-task-field]")) updateMeta(form);
    119   });
    120   form.addEventListener("change", (event) => {
    121     if (event.target.matches("[data-schedule-field], [data-task-field]")) updateMeta(form);
    122   });
    123   form.addEventListener("click", (event) => {
    124     if (event.target.closest("[data-add-task]")) {
    125       createTaskRow(form, { id: newTaskId(), title: "", date: "", done: false }, taskRows(form).length);
    126       setStatus(form, "Tarefa adicionada.");
    127       updateMeta(form);
    128       onStructuralChange();
    129       form.querySelector("[data-task]:last-of-type [data-task-field=title]")?.focus();
    130       return;
    131     }
    132     if (event.target.closest("[data-add-suggested-tasks]")) {
    133       const existingKeys = new Set(taskRows(form).map((row) => row.dataset.taskTemplateKey).filter(Boolean));
    134       const existingTitles = new Set(taskRows(form).map((row) => row.querySelector('[data-task-field="title"]')?.value.trim()).filter(Boolean));
    135       const template = projectTemplate(form);
    136       let added = 0;
    137       template.forEach(([key, title], stage) => {
    138         if (existingKeys.has(key) || existingTitles.has(title)) return;
    139         createTaskRow(form, { id: newTaskId(), title, date: "", done: false, stage, template_key: key }, taskRows(form).length);
    140         added += 1;
    141       });
    142       setStatus(form, added ? `${added} tarefas do roteiro foram adicionadas.` : "O roteiro já está adicionado.");
    143       updateMeta(form);
    144       if (added) onStructuralChange();
    145       return;
    146     }
    147     if (event.target.closest("[data-remove-task]")) {
    148       event.target.closest("[data-task]")?.remove();
    149       showEmptyMessage(form);
    150       setStatus(form, "Tarefa removida.");
    151       updateMeta(form);
    152       onStructuralChange();
    153     }
    154   });
    155 }
