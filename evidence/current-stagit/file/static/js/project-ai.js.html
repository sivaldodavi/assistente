     37   function aiContextFields(stageId = "") {
     38     const stage = form?.querySelector(`[data-stage="${CSS.escape(stageId)}"]`);
     39     return [...(stage?.querySelectorAll("[data-case-field]") || [])].map((field) => field.dataset.caseField);
     40   }

     42   function applyRichField(fieldId, value) {
     43     if (!fieldId || !String(value).trim()) return false;
     44     const direct = form.querySelector(`[data-case-field="${CSS.escape(fieldId)}"]`);
     45     if (direct) {
     46       direct.value = value;
     47       direct.dispatchEvent(new Event("input", { bubbles: true }));
     48       return true;
     49     }
     50     const parts = String(fieldId).split(":");
     51     if (parts[0] === "criterion" && parts.length === 3) {
     52       const card = [...form.querySelectorAll("[data-criterion]")].find((item) => item.dataset.criterionId === parts[1]);
     53       const field = card?.querySelector(`[data-criterion-field="${CSS.escape(parts[2])}"]`);
     54       if (field) {
     55         field.value = value;
     56         field.dispatchEvent(new Event("input", { bubbles: true }));
     57         return true;
     58       }
     59     }
     60     return false;
     61   }

     63   function renderRichResponse(response, key) {
     64     const suggestions = (response?.suggestions || []).filter((item) => !item.declined);
     65     const suggestionHtml = suggestions.length ? `<div><strong>Possíveis ajustes</strong><ul class="ai-suggestion-list">${suggestions.map((item, index) => `<li class="ai-suggestion-item"><span>${escapeAi(item.text)}</span><small class="field-hint">Campo: ${escapeAi(item.field_id || "confira manualmente")}</small><div class="ai-suggestion-actions"><button type="button" class="text-button" data-rich-ai-apply="${escapeAi(key)}" data-rich-ai-index="${index}">Aplicar</button><button type="button" class="text-button" data-rich-ai-edit="${escapeAi(key)}" data-rich-ai-index="${index}">Editar e aplicar</button><button type="button" class="text-button" data-rich-ai-decline="${escapeAi(key)}" data-rich-ai-index="${index}">Ignorar</button></div></li>`).join("")}</ul></div>` : "";
     66     const rewriteHtml = response?.rewrite ? `<div class="ai-suggestion-item"><strong>Revisão opcional do campo</strong><p>${escapeAi(response.rewrite)}</p><button type="button" class="text-button" data-rich-ai-rewrite="${escapeAi(key)}">Substituir campo</button></div>` : "";
     67     const validation = response?.validation;
     68     const validationHtml = validation ? `<div class="ai-validation"><strong>Avaliação estruturada: ${escapeAi(validation.status || "não avaliado")}</strong>${(validation.observations || []).length ? `<ul>${validation.observations.map((item) => `<li>${escapeAi(item)}</li>`).join("")}</ul>` : ""}</div>` : "";
     69     const checklistHtml = (response?.checklist || []).length ? `<ul class="ai-suggestion-list">${response.checklist.map((item) => `<li class="ai-suggestion-item"><strong>${escapeAi(item.id)}</strong><span>Confiança: ${Math.round(Number(item.confidence || 0) * 100)}%</span><small>${escapeAi(item.evidence || "Sem evidência textual.")}</small></li>`).join("")}</ul>` : "";
     70     return `<article class="ai-response-card"><p>${escapeAi(response?.answer || "Não houve resposta textual.")}</p>${suggestionHtml}${rewriteHtml}${checklistHtml}${validationHtml}<small class="field-hint">Resposta gerada sob solicitação; confira tudo antes de usar.</small></article>`;
     71   }

     73   function renderJevResponse(analysis) {
     74     if (!analysis) return "";
     75     const items = (analysis.criteria || []).map((item) => {
     76       const probability = Array.isArray(item.probabilities) ? Math.max(...item.probabilities) : 0;
     77       return `<li class="ai-jev-item"><div><strong>${escapeAi(item.title || item.label || item.id)}</strong> <span class="ai-score">${escapeAi(item.score)} / 3</span></div><div class="ai-score-bar"><span style="width:${Math.round(probability * 100)}%"></span></div>${item.observation ? `<p>${escapeAi(item.observation)}</p>` : ""}<button type="button" class="text-button" data-jev-explain="${escapeAi(item.id)}">Entender esta observação</button></li>`;
     78     }).join("");
     79     const heading = analysis.expected_points !== undefined ? `<p><strong>${Number(analysis.expected_points).toFixed(2)} / ${Number(analysis.assessable_points || 3.5).toFixed(2)} pontos estimados.</strong></p>` : "<p>Leitura formativa da etapa; não é nota oficial.</p>";
     80     const grade = analysis.grade_range ? `<p><strong>Faixa provável:</strong> nível ${escapeAi(analysis.grade_range.score)} de 0 a 3.</p>` : "";
     81     return `<article class="ai-response-card"><h4>Resultado da avaliação</h4>${heading}${grade}<ul class="ai-jev-list">${items}</ul><small class="field-hint">A avaliação usa o texto enviado e deve ser conferida pelo estudante; não verifica automaticamente a correção jurídica das fontes.</small></article>`;
     82   }

     84   async function runRichAiAction(action, payload, target) {
     85     if (!target || !form?.dataset.aiActionUrl) return;
     86     abortCurrentRequest();
     87     currentAbortController = new AbortController();
     88     showLoading(target, "Salvando esta etapa e consultando a assistência…");
     89     try {
     90       if (!await flushSave()) throw new Error("Salve as alterações antes de pedir assistência.");
     91       const body = { action, project_revision: Number(form.dataset.revision), context_fields: aiContextFields(payload.stage_id), ...payload };
     92       const response = await fetch(form.dataset.aiActionUrl, { method: "POST", credentials: "same-origin", headers: { "Content-Type": "application/json", "X-CSRF-Token": form.querySelector("[name=csrf_token]").value }, body: JSON.stringify(body), signal: currentAbortController.signal });
     93       const result = await readJson(response);
     94       if (!response.ok) throw new Error(result.message || "Não foi possível consultar a assistência.");
     95       const key = `${action}:${payload.stage_id || payload.field_id || "project"}`;
     96       richAiResults.set(key, result.response || result);
     97       target.dataset.richAiKey = key;
     98       target.innerHTML = renderRichResponse(result.response || result, key);
     99     } catch (error) {
    100       if (error.name !== "AbortError") {
    101         target.textContent = error.message;
    102       }
    103     } finally {
    104       hideLoading(target);
    105       currentAbortController = null;
    106     }
    107   }

    223       const key = suggestionButton.dataset.richAiApply || suggestionButton.dataset.richAiEdit || suggestionButton.dataset.richAiDecline;
    224       const result = richAiResults.get(key);
    225       const index = Number(suggestionButton.dataset.richAiIndex);
    226       const suggestion = result?.suggestions?.[index];

    228       if (suggestionButton.hasAttribute("data-rich-ai-decline")) {
    229         suggestion.declined = true;

    232         if (suggestionButton.hasAttribute("data-rich-ai-edit")) value = window.prompt("Edite a sugestão antes de aplicar:", value);
    233         if (value === null || value === undefined) return;
    234         if (!applyRichField(suggestion.field_id, value)) suggestion.notice = "O campo não está visível nesta etapa; confira a sugestão manualmente.";
    235         else markDirty();
    236       }
    237       const owner = suggestionButton.closest("[data-ai-stage-result], [data-ai-field-result]");
    238       if (owner) owner.innerHTML = renderRichResponse(result, key);
    239       return;
    240     }
    241     const rewrite = event.target.closest("[data-rich-ai-rewrite]");
    242     if (rewrite) {
    243       const result = richAiResults.get(rewrite.dataset.richAiRewrite);
    244       const fieldId = rewrite.dataset.richAiRewrite.split(":").slice(1).join(":");
    245       if (result?.rewrite && window.confirm("Substituir o campo pela revisão sugerida?")) {
    246         if (applyRichField(fieldId, result.rewrite)) markDirty();
    247       }
    248     }
    249   });
