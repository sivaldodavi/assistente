     42   async function readJson(response) {
     43     const text = await response.text();
     44     if (!text) return {};
     45     try {
     46       return JSON.parse(text);
     47     } catch {
     48       return {};
     49     }

    144   const stages = [...(form?.querySelectorAll("[data-stage]") || [])];

    148   window.showStep = function showStep(index, focus = false) {
    149     if (!stages.length) return;
    150     activeStep = Math.max(0, Math.min(index, stages.length - 1));
    151     stages.forEach((stage, i) => { stage.hidden = i !== activeStep; });
    152     const phase = stages[activeStep].dataset.stagePhase;
    153     stepButtons.forEach((button, i) => {
    154       button.setAttribute("aria-current", i === activeStep ? "step" : "false");
    155       button.classList.toggle("is-other-phase", button.dataset.stepPhase !== phase);
    156     });
    157     form.querySelectorAll("[data-phase]").forEach((button) => button.setAttribute("aria-pressed", button.dataset.phase === phase ? "true" : "false"));
    158     form.querySelectorAll("[data-nav-phase]").forEach((node) => node.classList.toggle("is-other-phase", node.dataset.navPhase !== phase));
    159     form.querySelector("[data-previous-step]").disabled = activeStep === 0;
    160     form.querySelector("[data-next-step]").textContent = activeStep === stages.length - 1 ? "Voltar à identificação" : "Continuar";
    161     history.replaceState(null, "", `#stage-${stages[activeStep].dataset.stage}`);
    162     updateProgress(form, stages, activeStep, stepButtons, phase);
    163     if (focus) {
    164       stages[activeStep].querySelector("h2").focus({ preventScroll: true });
    165       form.querySelector(".workspace").scrollIntoView({ behavior: "auto", block: "start" });
    166     }
    167   };

    169   function updateProgress(form, stages, activeStep, stepButtons, phase) {
    170     if (!stages.length) return;
    171     const phaseStages = stages.filter((stage) => stage.dataset.stagePhase === phase);
    172     const required = phaseStages.flatMap((stage) => [...stage.querySelectorAll("[data-required-field]")]);
    173     const done = required.filter((input) => input.value.trim()).length;
    174     const percent = required.length ? Math.round(done / required.length * 100) : 0;
    175     form.querySelector("[data-progress-count]").textContent = `${done}/${required.length} campos · ${percent}%`;
    176     form.querySelector("[data-progress-bar]").style.width = `${percent}%`;
    177     const labels = { report: "Relatório", debate: "Debate", dissertation: "Dissertação", paper_project: "Projeto de Paper", paper_final: "Paper Final" };
    178     form.querySelector("[data-progress-label]").textContent = `Progresso · ${labels[phase]}`;
    179     stages.forEach((stage, i) => {
    180       const inputs = [...stage.querySelectorAll("[data-required-field], [data-workflow-check]")];
    181       const complete = inputs.length > 0 && inputs.every((input) => input.type === "checkbox" ? input.checked : input.value.trim());
    182       const status = stepButtons[i].querySelector(".step-status");
    183       status.textContent = complete ? "✓" : "·";
    184       status.setAttribute("aria-label", complete ? "Etapa completa" : "Etapa incompleta");
    185     });
    186     const title = form.querySelector('[data-case-field="title"]')?.value.trim();
    187     const heading = document.querySelector("[data-editor-title]");
    188     if (heading && title) heading.textContent = title;
    189   }
    190 
    191   stepButtons.forEach((button, i) => button.addEventListener("click", () => window.showStep(i, true)));
    192   form?.querySelectorAll("[data-phase]").forEach((button) => button.addEventListener("click", () => window.showStep(stages.findIndex((stage) => stage.dataset.stagePhase === button.dataset.phase), true)));
    193   form?.querySelector("[data-previous-step]")?.addEventListener("click", () => window.showStep(activeStep - 1, true));
    194   form?.querySelector("[data-next-step]")?.addEventListener("click", () => window.showStep(activeStep === stages.length - 1 ? 0 : activeStep + 1, true));
    195   const restoredStep = stages.findIndex((stage) => `#stage-${stage.dataset.stage}` === location.hash);
    196   window.showStep(restoredStep >= 0 ? restoredStep : 0);
    197   window.addEventListener("hashchange", () => { const index = stages.findIndex((stage) => `#stage-${stage.dataset.stage}` === location.hash); if (index >= 0) window.showStep(index); });
    198 }

    229   function appendProjectSource(source, checked = true) {
    230     const list = form.querySelector(".source-check-list");

    232     list.querySelector(".empty-state")?.remove();

    236     const input = row.querySelector("[data-project-source]");
    237     input.value = source.id;
    238     input.checked = checked;
    239     row.querySelector("strong").textContent = source.title || "Fonte sem título";
    240     row.querySelectorAll("small")[0].textContent = source.author || "";
    241     row.querySelectorAll("small")[1].textContent = source.reference || "";

    245   window.createInlineSource = async function createInlineSource(button) {
    246     const status = form.querySelector("[data-source-create-status]");
    247     const value = (name) => form.querySelector(`[data-source-create-field="${name}"]`)?.value?.trim() || "";
    248     const title = value("title");
    249     if (!title) {
    250       if (status) status.textContent = "Informe o título da fonte.";
    251       form.querySelector('[data-source-create-field="title"]')?.focus();
    252       return;
    253     }
    254     button.disabled = true;
    255     if (status) status.textContent = "Adicionando…";
    256     try {
    257       const response = await fetch(form.dataset.sourceCreateUrl, {
    258         method: "POST",
    259         credentials: "same-origin",
    260         headers: { "Content-Type": "application/json", "X-CSRF-Token": form.querySelector("[name=csrf_token]").value },
    261         body: JSON.stringify({ title, author: value("author"), kind: value("kind"), reference: value("reference"), notes: value("notes") }),
    262       });
    263       const result = await readJson(response);
    264       if (!response.ok) throw new Error(result.message || "Não foi possível adicionar a fonte.");
    265       appendProjectSource(result.source);
    266       ["title", "author", "reference", "notes"].forEach((name) => {
    267         const field = form.querySelector(`[data-source-create-field="${name}"]`);
    268         if (field) field.value = "";
    269       });
    270       if (status) status.textContent = "Fonte adicionada.";
    271       markDirty();
    272     } catch (error) {
    273       if (status) status.textContent = error.message;
    274     } finally {
    275       button.disabled = false;
    276     }
    277   };

    279   function renderLegalEntry(entry, attachUrl, csrf) {
    280     const card = document.createElement("article");
    281     card.className = "legal-result-card";
    282     const title = document.createElement("h4");
    283     title.textContent = entry.title || "Fonte sem título";
    284     card.append(title);
    285     const meta = document.createElement("p");
    286     meta.className = "legal-result-meta";
    287     meta.textContent = [entry.source, entry.source_year, entry.author, entry.reference].filter(Boolean).join(" · ");
    288     card.append(meta);
    289     const why = document.createElement("p");
    290     why.textContent = entry.why || "Pista contextual para conferência.";
    291     card.append(why);
    292     const actions = document.createElement("div");
    293     actions.className = "legal-result-actions";
    294     const attach = document.createElement("button");
    295     attach.type = "button";
    296     attach.className = "text-button";
    297     attach.textContent = "Adicionar à Minha Biblioteca";
    298     attach.addEventListener("click", async () => {
    299       attach.disabled = true;
    300       try {
    301         const response = await fetch(attachUrl, { method: "POST", credentials: "same-origin", headers: { "Content-Type": "application/json", "X-CSRF-Token": csrf }, body: JSON.stringify({ source_id: entry.id }) });
    302         const result = await readJson(response);
    303         if (!response.ok) throw new Error(result.message || "Não foi possível adicionar a fonte.");
    304         appendProjectSource({ id: result.source_id, title: result.title, author: result.author, reference: result.reference });
    305         markDirty();
    306         attach.textContent = "Adicionada à biblioteca";
    307       } catch (error) {
    308         attach.disabled = false;
    309         attach.textContent = error.message;
    310       }
    311     });
    312     const discard = document.createElement("button");
    313     discard.type = "button";
    314     discard.className = "text-button";
    315     discard.textContent = "Descartar";
    316     discard.addEventListener("click", () => card.remove());
    317     actions.append(attach, discard);
    318     card.append(actions);
    319     return card;
    320   }
