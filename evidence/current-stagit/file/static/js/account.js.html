     16 function initAccountTabs() {
     17   const tabButtons = document.querySelectorAll('.account-tabs [role="tab"]');
     18   const tabPanels = document.querySelectorAll('.tab-panel[role="tabpanel"]');
     19 
     20   tabButtons.forEach(btn => {
     21     btn.addEventListener('click', () => {
     22       const tabId = btn.dataset.tab;
     23       activateTab(tabId, tabButtons, tabPanels);
     24     });
     25   });
     26 
     27   // Activate first tab by default
     28   if (tabButtons[0]) {
     29     activateTab(tabButtons[0].dataset.tab, tabButtons, tabPanels);
     30   }
     31 }

     33 function activateTab(tabId, tabButtons, tabPanels) {
     34   tabButtons.forEach(btn => {
     35     const isActive = btn.dataset.tab === tabId;
     36     btn.setAttribute('aria-selected', isActive);
     37     btn.classList.toggle('is-active', isActive);
     38   });
     39 
     40   tabPanels.forEach(panel => {
     41     const isActive = panel.id === `tab-${tabId}`;
     42     panel.hidden = !isActive;
     43     panel.setAttribute('aria-hidden', !isActive);
     44   });
     45 }

     86 async function initSharingPreferences() {
     87   const form = document.getElementById('sharing-prefs-form');
     88   const statusEl = document.getElementById('sharing-prefs-status');
     89   const allowView = document.getElementById('allow_admin_view');
     90   const allowComments = document.getElementById('allow_admin_comments');
     91 
     92   if (!form) return;
     93 
     94   // Load current preferences
     95   await loadSharingPreferences();
     96 
     97   // Disable comments checkbox if view is not checked
     98   allowView?.addEventListener('change', () => {
     99     if (allowComments) {
    100       allowComments.disabled = !allowView.checked;
    101       if (!allowView.checked) {
    102         allowComments.checked = false;
    103       }
    104     }
    105   });
    106 
    107   form.addEventListener('submit', async (e) => {
    108     e.preventDefault();
    109     const formData = new FormData(form);
    110     const payload = {
    111       allow_admin_view: formData.get('allow_admin_view') === 'on',
    112       allow_admin_comments: formData.get('allow_admin_comments') === 'on',
    113       inherit_account_default: formData.get('inherit_account_default') === 'on'
    114     };
    115 
    116     try {
    117       const res = await fetch('/account/sharing', {
    118         method: 'POST',
    119         headers: { 'Content-Type': 'application/json', 'X-CSRF-Token': formData.get('csrf_token') },
    120         body: JSON.stringify(payload)
    121       });
    122       const data = await res.json();
    123       if (data.ok) {
    124         showStatus(statusEl, 'Preferências salvas com sucesso.', 'success');
    125         // Update UI to reflect saved values
    126         allowView.checked = data.preferences.allow_admin_view;
    127         allowComments.checked = data.preferences.allow_admin_comments;
    128         allowComments.disabled = !data.preferences.allow_admin_view;
    129         document.getElementById('inherit_account_default').checked = data.preferences.inherit_default;
    130       } else {
    131         showStatus(statusEl, data.message || 'Erro ao salvar preferências.', 'error');
    132       }
    133     } catch (err) {
    134       showStatus(statusEl, 'Erro de conexão.', 'error');
    135     }
    136   });
    137 }

    139 async function loadSharingPreferences() {
    140   try {
    141     const res = await fetch('/account/sharing');
    142     const data = await res.json();
    143     if (data.ok) {
    144       const prefs = data.preferences;
    145       document.getElementById('allow_admin_view').checked = prefs.allow_admin_view;
    146       document.getElementById('allow_admin_comments').checked = prefs.allow_admin_comments;
    147       document.getElementById('allow_admin_comments').disabled = !prefs.allow_admin_view;
    148       document.getElementById('inherit_account_default').checked = prefs.inherit_default;
    149     }
    150   } catch (err) {
    151     console.error('Failed to load sharing preferences:', err);
    152   }
    153 }

    159 async function initDeletion() {
    160   const statusEl = document.getElementById('deletion-status');
    161   const actionsEl = document.getElementById('deletion-actions');
    162   const btnRequest = document.getElementById('btn-request-deletion');
    163   const btnCancel = document.getElementById('btn-cancel-deletion');
    164   const btnConfirm = document.getElementById('btn-confirm-deletion');
    165 
    166   if (!statusEl) return;
    167 
    168   // Check for pending deletion request
    169   await checkDeletionStatus();
    170 
    171   // Forms use native submission, just handle UI state
    172   const formRequest = document.getElementById('form-request-deletion');
    173   const formCancel = document.getElementById('form-cancel-deletion');
    174   const formConfirm = document.getElementById('form-confirm-deletion');
    175 
    176   if (formRequest) {
    177     formRequest.addEventListener('submit', async (e) => {
    178       e.preventDefault();
    179       try {
    180         const res = await fetch('/account/deletion/request', {
    181           method: 'POST',
    182           headers: { 'Content-Type': 'application/json', 'X-CSRF-Token': new FormData(formRequest).get('csrf_token') }
    183         });
    184         const data = await res.json();
    185         if (data.ok) {
    186           updateDeletionUI('pending');
    187         } else {
    188           showStatus(statusEl, data.message || 'Erro ao solicitar exclusão.', 'error');
    189         }
    190       } catch (err) {
    191         showStatus(statusEl, 'Erro de conexão.', 'error');
    192       }
    193     });
    194   }
    195 
    196   if (formCancel) {
    197     formCancel.addEventListener('submit', async (e) => {
    198       e.preventDefault();
    199       try {
    200         const res = await fetch('/account/deletion/cancel', {
    201           method: 'POST',
    202           headers: { 'Content-Type': 'application/json', 'X-CSRF-Token': new FormData(formCancel).get('csrf_token') }
    203         });
    204         const data = await res.json();
    205         if (data.ok) {
    206           updateDeletionUI('none');
    207         } else {
    208           showStatus(statusEl, data.message || 'Erro ao cancelar.', 'error');
    209         }
    210       } catch (err) {
    211         showStatus(statusEl, 'Erro de conexão.', 'error');
    212       }
    213     });
    214   }
    215 
    216   if (formConfirm) {
    217     formConfirm.addEventListener('submit', async (e) => {
    218       // Let the native confirm dialog handle confirmation
    219       // If user confirms, form submits normally
    220     });
    221   }
    222 }

    224 async function checkDeletionStatus() {
    225   try {
    226     const res = await fetch('/account/sharing'); // Reusing endpoint that returns account info including deletion_request_pending
    227     const data = await res.json();
    228     if (data.ok && data.account) {
    229       const account = data.account;
    230       if (account.deletion_request_pending) {
    231         updateDeletionUI('pending');
    232       } else {
    233         updateDeletionUI('none');
    234       }
    235     }
    236   } catch (err) {
    237     console.error('Failed to check deletion status:', err);
    238   }
    239 }

    241 function updateDeletionUI(state) {
    242   const statusEl = document.getElementById('deletion-status');
    243   const btnRequest = document.getElementById('btn-request-deletion');
    244   const btnCancel = document.getElementById('btn-cancel-deletion');
    245   const btnConfirm = document.getElementById('btn-confirm-deletion');
    246   const formRequest = document.getElementById('form-request-deletion');
    247   const formCancel = document.getElementById('form-cancel-deletion');
    248   const formConfirm = document.getElementById('form-confirm-deletion');
    249 
    250   if (state === 'pending') {
    251     statusEl.textContent = '⚠️ Solicitação de exclusão pendente. Você pode cancelar ou confirmar a exclusão definitiva.';
    252     statusEl.className = 'deletion-status warning';
    253     if (formRequest) formRequest.style.display = 'none';
    254     if (formCancel) formCancel.style.display = 'inline';
    255     if (formConfirm) formConfirm.style.display = 'inline';
    256   } else {
    257     statusEl.textContent = 'Nenhuma solicitação de exclusão pendente.';
    258     statusEl.className = 'deletion-status';
    259     if (formRequest) formRequest.style.display = 'inline';
    260     if (formCancel) formCancel.style.display = 'none';
    261     if (formConfirm) formConfirm.style.display = 'none';
    262   }

    269 async function initGrants() {
    270   const listEl = document.getElementById('grants-list');
    271   if (!listEl) return;
    272 
    273   try {
    274     const res = await fetch('/account/grants');
    275     const data = await res.json();
    276     if (data.ok && data.grants) {
    277       renderGrants(data.grants, listEl);
    278     } else {
    279       listEl.innerHTML = '<p class="field-hint">Nenhum grant encontrado.</p>';
    280     }
    281   } catch (err) {
    282     listEl.innerHTML = '<p class="field-hint error">Erro ao carregar grants.</p>';
    283   }
    284 }

    286 function renderGrants(grants, container) {
    287   if (!grants.length) {
    288     container.innerHTML = '<p class="field-hint">Nenhum crédito concedido ainda.</p>';
    289     return;
    290   }
    291 
    292   container.innerHTML = grants.map(g => `
    293     <div class="grant-card">
    294       <div class="grant-header">
    295         <strong>${g.plan_code === 'livre' ? 'Sem plano' : g.plan_code}</strong>
    296         <span class="grant-date">Concedido em ${new Date(g.created_at * 1000).toLocaleDateString('pt-BR')}</span>
    297       </div>
    298       <div class="grant-credits">
    299         <span class="credit ${g.project_remaining > 0 ? 'has-credit' : ''}">Projetos: ${g.project_remaining}/${g.project_credits}</span>
    300         <span class="credit ${g.ai_analysis_remaining > 0 ? 'has-credit' : ''}">Análise IA: ${g.ai_analysis_remaining}/${g.ai_analysis_credits}</span>
    301         <span class="credit ${g.deepseek_explanation_remaining > 0 ? 'has-credit' : ''}">Explicação DeepSeek: ${g.deepseek_explanation_remaining}/${g.deepseek_explanation_credits}</span>
    302       </div>
    303       ${g.expires_at ? `<p class="grant-expiry">Expira em ${new Date(g.expires_at * 1000).toLocaleDateString('pt-BR')}</p>` : ''}
    304     </div>
    305   `).join('');
    306 }

    312 function showStatus(el, message, type) {
    313   if (!el) return;
    314   el.textContent = message;
    315   el.className = type === 'success' ? 'notice success' : 'notice error';
    316   el.hidden = false;
    317   setTimeout(() => { el.hidden = true; }, 5000);
