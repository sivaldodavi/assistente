      6 function escapeHtml(text) {
      7   return String(text ?? "")
      8     .replaceAll("&", "&")
      9     .replaceAll("<", "<")
     10     .replaceAll(">", ">")
     11     .replaceAll('"', """)
     12     .replaceAll("'", "&#039;");
     13 }

     15 async function loadRevisions(revisionsUrl) {
     16   try {
     17     const res = await fetch(revisionsUrl);
     18     const data = await res.json();
     19     if (!data.ok) throw new Error(data.error || "Erro ao carregar revisões.");
     20     return data.revisions || [];
     21   } catch (err) {
     22     console.error("Failed to load revisions:", err);
     23     return [];
     24   }
     25 }

     27 function renderRevisionsList(revisions, container) {
     28   if (!revisions.length) {
     29     container.innerHTML = '<p class="field-hint">Nenhuma revisão encontrada.</p>';
     30     return;
     31   }
     32   container.innerHTML = revisions.map(r => `
     33     <article class="revision-card" data-revision="${r.revision}">
     34       <header>
     35         <strong>Revisão ${r.revision}</strong>
     36         <time>${new Date(r.created_at * 1000).toLocaleString("pt-BR")}</time>
     37       </header>
     38       <div class="revision-actions">
     39         <button type="button" class="text-button" data-diff="${r.revision}">Ver diff</button>
     40         <button type="button" class="text-button" data-restore="${r.revision}">Restaurar</button>
     41       </div>
     42     </article>
     43   `).join("");
     44 }

     46 async function showDiff(revision, revisionsUrl, diffContainer, readJson) {
     47   if (!diffContainer) return;
     48   diffContainer.hidden = false;
     49   diffContainer.innerHTML = '<p class="field-hint">Carregando diff…</p>';
     50   try {
     51     const res = await fetch(`${revisionsUrl}/${revision}/diff`);
     52     const data = await res.json();
     53     if (!data.ok) throw new Error(data.error || "Erro ao carregar diff.");
     54     diffContainer.innerHTML = `<pre class="diff-output">${escapeHtml(data.diff)}</pre>`;
     55   } catch (err) {
     56     diffContainer.innerHTML = `<p class="field-hint error">${err.message}</p>`;
     57   }
     58 }

     60 async function restoreRevision(revision, revisionsUrl, currentRevision, form, readJson, setSaveState) {
     61   if (!confirm(`Tem certeza que deseja restaurar a revisão ${revision}? As alterações atuais não salvas serão perdidas.`)) return;
     62   try {
     63     const csrf = form.querySelector("[name=csrf_token]")?.value;
     64     if (!csrf) throw new Error("Sessão sem proteção CSRF.");
     65     const res = await fetch(`${revisionsUrl}/${revision}/restore`, {
     66       method: "POST",
     67       credentials: "same-origin",
     68       headers: { "Content-Type": "application/json", "X-CSRF-Token": csrf },
     69       body: JSON.stringify({ revision: Number(currentRevision) })
     70     });
     71     const data = await readJson(res);
     72     if (!res.ok) throw new Error(data.error || "Erro ao restaurar.");
     73     window.location.reload();
     74   } catch (err) {
     75     setSaveState(err.message);
     76   }
     77 }

     79 export function initRevisions({ form, readJson, setSaveState }) {
     80   if (!form || !form.dataset.revisionsUrl) return;
     81 
     82   const toggle = form.querySelector("[data-revisions-toggle]");
     83   const panel = form.querySelector("[data-revisions-panel]");
     84   const listEl = form.querySelector("[data-revisions-list]");
     85   const diffEl = form.querySelector("[data-revision-diff]");
     86 
     87   if (!toggle || !panel || !listEl) return;
     88 
     89   toggle.addEventListener("click", async () => {
     90     const isHidden = panel.hidden;
     91     if (isHidden) {
     92       panel.hidden = false;
     93       listEl.innerHTML = '<p class="field-hint">Carregando…</p>';
     94       const revisions = await loadRevisions(form.dataset.revisionsUrl);
     95       renderRevisionsList(revisions, listEl);
     96 
     97       listEl.querySelectorAll("[data-diff]").forEach(btn => {
     98         btn.addEventListener("click", () => showDiff(btn.dataset.diff, form.dataset.revisionsUrl, diffEl, readJson));
     99       });
    100       listEl.querySelectorAll("[data-restore]").forEach(btn => {
    101         btn.addEventListener("click", () => restoreRevision(btn.dataset.revision, form.dataset.revisionsUrl, form.dataset.revision, form, readJson, setSaveState));
    102       });
    103     } else {
    104       panel.hidden = true;
    105     }
    106   });
    107 }
