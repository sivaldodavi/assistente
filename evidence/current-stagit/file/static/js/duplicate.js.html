      3  * Handles project duplication functionality

      5 export function initDuplicate({ form, flushSave, readJson }) {
      6   if (!form || !form.dataset.duplicateUrl) return;
      7 
      8   form.querySelector("[data-duplicate-project]")?.addEventListener("click", async () => {
      9     const button = form.querySelector("[data-duplicate-project]");
     10     const saveState = form.querySelector("[data-save-state]");
     11     button.disabled = true;
     12     if (saveState) saveState.textContent = "Duplicando projeto…";
     13     try {
     14       if (!await flushSave()) throw new Error("Salve as alterações antes de duplicar.");
     15       const csrf = form.querySelector("[name=csrf_token]")?.value;
     16       if (!csrf) throw new Error("Sessão sem proteção CSRF.");
     17       const response = await fetch(form.dataset.duplicateUrl, {
     18         method: "POST",
     19         credentials: "same-origin",
     20         headers: { "Content-Type": "application/json", "X-CSRF-Token": csrf },
     21         body: JSON.stringify({ revision: Number(form.dataset.revision) })
     22       });
     23       const result = await readJson(response);
     24       if (!response.ok) throw new Error(result.error || "Erro ao duplicar.");
     25       if (result.project_url) {
     26         if (saveState) saveState.textContent = "Projeto duplicado. Redirecionando…";
     27         window.location.href = result.project_url;
     28       } else {
     29         throw new Error("URL do projeto não retornada.");
     30       }
     31     } catch (error) {
     32       if (saveState) saveState.textContent = error.message;
     33       button.disabled = false;
     34     }
     35   });
     36 }
