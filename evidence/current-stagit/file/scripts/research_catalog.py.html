     27     """Retorna o diretório de dados conforme o contrato, fora do repositório."""

     35 def _clean(value: Any) -> str:
     36     if value is None:
     37         return ""
     38     if isinstance(value, (list, tuple)):
     39         return "; ".join(_clean(item) for item in value if _clean(item))
     40     return " ".join(str(value).replace("\x00", " ").split())

     43 def _entry(entry: Mapping[str, Any]) -> dict[str, Any]:
     44     """Normaliza um registro sem eliminar campos oficiais específicos do STJ."""
     45     result: dict[str, Any] = {}
     46     for key, value in entry.items():
     47         if value is None or value == "" or value == []:
     48             continue
     49         if key in {"id", "kind", "title", "reference", "author", "abstract", "source", "link"}:
     50             if isinstance(value, (dict, list)):
     51                 result[key] = value
     52             else:
     53                 result[key] = _clean(value)
     54         elif key == "source_year":
     55             try:
     56                 result[key] = int(value)
     57             except (TypeError, ValueError):
     58                 continue
     59         else:
     60             result[key] = value
     61     if not result.get("id") or result.get("kind") not in KINDS or not result.get("title"):
     62         raise ValueError("cada entrada precisa de id, kind válido e title")
     63     return result

     66 def _record_files(catalog_dir: Path) -> list[Path]:
     67     return sorted(
     68         (path for path in catalog_dir.glob("*.jsonl") if path.is_file()),
     69         key=lambda path: path.name,
     70     )

     73 def _read_records(catalog_dir: Path) -> list[dict[str, Any]]:
     74     records: dict[str, dict[str, Any]] = {}
     75     for path in _record_files(catalog_dir):
     76         with path.open("r", encoding="utf-8") as handle:
     77             for line_number, line in enumerate(handle, 1):
     78                 if not line.strip():
     79                     continue
     80                 try:
     81                     value = _entry(json.loads(line))
     82                 except (json.JSONDecodeError, OSError, ValueError) as exc:
     83                     raise ValueError(f"registro inválido em {path}:{line_number}: {exc}") from exc
     84                 # A ordem dos arquivos é estável: o primeiro registro é a fonte
     85                 # vencedora em caso de ID repetido.
     86                 records.setdefault(str(value["id"]), value)
     87     return list(records.values())

     90 def _search_text(value: Any) -> str:
     91     if isinstance(value, Mapping):
     92         return " ".join(_search_text(item) for item in value.values())
     93     if isinstance(value, (list, tuple)):
     94         return " ".join(_search_text(item) for item in value)
     95     return _clean(value)

     98 def _signature(catalog_dir: Path) -> tuple[Any, ...]:
     99     paths = [catalog_dir / _INDEX_NAME, catalog_dir / _META_NAME, *_record_files(catalog_dir)]
    100     signature: list[Any] = []
    101     for path in paths:
    102         try:
    103             stat = path.stat()
    104         except FileNotFoundError:
    105             signature.append((str(path), None))
    106         else:
    107             signature.append((str(path), stat.st_mtime_ns, stat.st_size))
    108     return tuple(signature)

    111 def _create_schema(connection: sqlite3.Connection, use_fts: bool) -> None:
    112     connection.execute(
    113         """CREATE TABLE entries (
    114             id TEXT PRIMARY KEY,
    115             kind TEXT NOT NULL,
    116             title TEXT NOT NULL,
    117             reference TEXT,
    118             author TEXT,
    119             abstract TEXT,
    120             source_year INTEGER,
    121             source TEXT,
    122             search_text TEXT NOT NULL,
    123             payload TEXT NOT NULL
    124         )"""
    125     )
    126     if use_fts:
    127         connection.execute(
    128             """CREATE VIRTUAL TABLE entries_fts USING fts5(
    129                 id UNINDEXED, kind UNINDEXED, title, reference, author,
    130                 abstract, search_text, tokenize='unicode61 remove_diacritics 2'
    131             )"""
    132         )

    135 def _fill_schema(connection: sqlite3.Connection, records: Iterable[Mapping[str, Any]], use_fts: bool) -> int:
    136     count = 0
    137     for raw in records:
    138         value = _entry(raw)
    139         payload = json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":"))
    140         search_text = _search_text(value)
    141         connection.execute(
    142             """INSERT OR IGNORE INTO entries
    143                (id, kind, title, reference, author, abstract, source_year, source, search_text, payload)
    144                VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)""",
    145             (
    146                 str(value["id"]), value["kind"], value["title"], value.get("reference"),
    147                 value.get("author"), value.get("abstract"), value.get("source_year"),
    148                 value.get("source"), search_text, payload,
    149             ),
    150         )
    151         if use_fts:
    152             connection.execute(
    153                 """INSERT OR IGNORE INTO entries_fts
    154                    (id, kind, title, reference, author, abstract, search_text)
    155                    VALUES (?, ?, ?, ?, ?, ?, ?)""",
    156                 (
    157                     str(value["id"]), value["kind"], value.get("title", ""),
    158                     _search_text(value.get("reference")), _search_text(value.get("author")),
    159                     _search_text(value.get("abstract")), search_text,
    160                 ),
    161             )
    162         count += 1
    163     return count

    166 def _fts_available(connection: sqlite3.Connection) -> bool:
    167     try:
    168         return bool(connection.execute("SELECT sqlite_compileoption_used('ENABLE_FTS5')").fetchone()[0])
    169     except sqlite3.DatabaseError:
    170         return False

    173 def build_catalog(catalog_dir: str | os.PathLike[str], metadata: Mapping[str, Any] | None = None) -> dict[str, Any]:
    174     """Reconstrói atomicamente o índice a partir dos JSONL privados do diretório."""
    175     directory = Path(catalog_dir).expanduser()
    176     directory.mkdir(parents=True, exist_ok=True)
    177     records = _read_records(directory)
    178     fd, temporary = tempfile.mkstemp(prefix="catalog-", suffix=".sqlite3", dir=directory)
    179     os.close(fd)
    180     try:
    181         connection = sqlite3.connect(temporary)
    182         use_fts = _fts_available(connection)
    183         _create_schema(connection, use_fts)
    184         count = _fill_schema(connection, records, use_fts)
    185         connection.execute("CREATE INDEX entries_kind ON entries(kind)")
    186         connection.commit()
    187         connection.close()
    188         os.replace(temporary, directory / _INDEX_NAME)
    189     finally:
    190         try:
    191             os.unlink(temporary)
    192         except FileNotFoundError:
    193             pass
    194     sources = sorted({str(row.get("source")) for row in records if row.get("source")})
    195     years = sorted({int(row["source_year"]) for row in records if row.get("source_year") is not None})
    196     result: dict[str, Any] = {}
    197     meta_path = directory / _META_NAME
    198     if meta_path.exists():
    199         try:
    200             previous = json.loads(meta_path.read_text(encoding="utf-8"))
    201             if isinstance(previous, dict):
    202                 result.update(previous)
    203         except (OSError, json.JSONDecodeError):
    204             pass
    205     result.update({
    206         "count": count,
    207         "kinds": sorted({row["kind"] for row in records}),
    208         "source_years": years,
    209         "sources": sources,
    210         "fts5": use_fts,
    211     })
    212     if metadata:
    213         result.update(dict(metadata))
    214     (directory / _META_NAME).write_text(
    215         json.dumps(result, ensure_ascii=False, sort_keys=True, indent=2) + "\n", encoding="utf-8"
    216     )
    217     with _CACHE_LOCK:
    218         old = _CACHE.pop(str(directory.resolve()), None)
    219         if old:
    220             old[1].close()
    221     return result

    224 def write_catalog_file(
    225     catalog_dir: str | os.PathLike[str],
    226     filename: str,
    227     records: Iterable[Mapping[str, Any]],
    228     metadata: Mapping[str, Any] | None = None,
    229 ) -> dict[str, Any]:
    230     """Grava um JSONL de fonte e atualiza o índice combinado do diretório."""
    231     if not filename.endswith(".jsonl") or Path(filename).name != filename:
    232         raise ValueError("filename deve ser um nome JSONL simples")
    233     directory = Path(catalog_dir).expanduser()
    234     directory.mkdir(parents=True, exist_ok=True)
    235     normalized = [_entry(record) for record in records]
    236     temporary = directory / f".{filename}.tmp"
    237     with temporary.open("w", encoding="utf-8", newline="\n") as handle:
    238         for record in normalized:
    239             handle.write(json.dumps(record, ensure_ascii=False, sort_keys=True, separators=(",", ":")) + "\n")
    240     os.replace(temporary, directory / filename)
    241     return build_catalog(directory, metadata=metadata)

    244 def _open_index(directory: Path) -> sqlite3.Connection:
    245     """Abre índice existente; sem índice, constrói uma vez a partir de JSONL."""
    246     directory.mkdir(parents=True, exist_ok=True)
    247     index = directory / _INDEX_NAME
    248     index_mtime = index.stat().st_mtime_ns if index.exists() else -1
    249     source_changed = any(path.stat().st_mtime_ns > index_mtime for path in _record_files(directory))
    250     if not index.exists() or source_changed:
    251         build_catalog(directory)
    252     uri = f"file:{index.resolve()}?mode=ro"
    253     return sqlite3.connect(uri, uri=True, check_same_thread=False)

    256 def _connection(directory: Path) -> sqlite3.Connection:
    257     key = str(directory.resolve())
    258     signature = _signature(directory)
    259     with _CACHE_LOCK:
    260         cached = _CACHE.get(key)
    261         if cached and cached[0] == signature:
    262             return cached[1]
    263         if cached:
    264             cached[1].close()
    265         connection = _open_index(directory)
    266         _CACHE[key] = (_signature(directory), connection)
    267         return connection

    270 def _fts_query(query: str) -> str:
    271     # Tokens próprios, entre aspas, evitam que a consulta do usuário injete a
    272     # sintaxe MATCH. O prefixo permite buscar flexões sem gerar termos amplos.
    273     tokens = [_clean(token).replace('"', "") for token in _WORD_RE.findall(query)]
    274     return " AND ".join(f'"{token}"*' for token in tokens if token)

    277 def _public_payload(payload: str) -> dict[str, Any]:
    278     value = json.loads(payload)
    279     # Não vazamos colunas internas; os campos extras oficiais (p.ex. ementa e
    280     # numeroProcesso) continuam disponíveis para a etapa JEV selecionar IDs.
    281     return value

    284 def search_catalog(
    285     query: str,
    286     kind: str | None = None,
    287     limit: int = 12,
    288     catalog_dir: str | os.PathLike[str] | None = None,
    289 ) -> dict[str, Any]:
    290     """Busca metadados locais conforme o contrato Corpus API."""
    291     if kind is not None and kind not in KINDS:
    292         raise ValueError(f"kind inválido: {kind!r}")
    293     try:
    294         amount = int(limit)
    295     except (TypeError, ValueError) as exc:
    296         raise ValueError("limit deve ser inteiro") from exc
    297     if amount <= 0:
    298         return {"entries": [], "metadata": {"count": 0, "limit": amount}}
    299     amount = min(amount, 100)
    300     directory = Path(catalog_dir).expanduser() if catalog_dir is not None else default_catalog_dir()
    301     directory = directory.resolve()
    302     if not directory.exists():
    303         return {"entries": [], "metadata": {"count": 0, "catalog_dir": str(directory), "limit": amount}}
    304     connection = _connection(directory)
    305     terms = _fts_query(query or "")
    306     params: list[Any] = []
    307     where = []
    308     if kind:
    309         where.append("e.kind = ?")
    310         params.append(kind)
    311     if terms:
    312         where.append("entries_fts MATCH ?")
    313         params.append(terms)
    314     clause = " WHERE " + " AND ".join(where) if where else ""
    315     if terms:
    316         sql = (
    317             "SELECT e.payload FROM entries_fts f JOIN entries e ON e.id = f.id"
    318             f"{clause} ORDER BY bm25(entries_fts), e.title COLLATE NOCASE, e.id LIMIT ?"
    319         )
    320     else:
    321         sql = "SELECT e.payload FROM entries e" + clause + " ORDER BY e.title COLLATE NOCASE, e.id LIMIT ?"
    322     params.append(amount)
    323     try:
    324         rows = connection.execute(sql, params).fetchall()
    325     except sqlite3.OperationalError as exc:
    326         # Índices criados sem FTS5 continuam utilizáveis como busca leve, sem
    327         # reprocessar JSON por requisição.
    328         if terms and "no such table: entries_fts" in str(exc):
    329             like = "%" + "%".join(_clean(query).split()) + "%"
    330             fallback_where = []
    331             fallback_params: list[Any] = []
    332             if kind:
    333                 fallback_where.append("e.kind = ?")
    334                 fallback_params.append(kind)
    335             fallback_where.append("e.search_text LIKE ?")
    336             fallback_params.append(like)
    337             fallback_sql = "SELECT e.payload FROM entries e WHERE " + " AND ".join(fallback_where) + " ORDER BY e.title COLLATE NOCASE, e.id LIMIT ?"
    338             fallback_params.append(amount)
    339             rows = connection.execute(fallback_sql, fallback_params).fetchall()
    340         else:
    341             raise
    342     entries = [_public_payload(row[0]) for row in rows]
    343     meta_path = directory / _META_NAME
    344     metadata: dict[str, Any] = {}
    345     if meta_path.exists():
    346         try:
    347             metadata = json.loads(meta_path.read_text(encoding="utf-8"))
    348         except (OSError, json.JSONDecodeError):
    349             metadata = {}
    350     metadata.update({"catalog_dir": str(directory), "query": query or "", "limit": amount, "returned": len(entries)})
    351     return {"entries": entries, "metadata": metadata}
