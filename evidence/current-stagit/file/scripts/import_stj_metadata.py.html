     33 def _resource_date(resource: Mapping[str, Any]) -> dt.date | None:
     34     name = str(resource.get("name") or resource.get("url") or "")
     35     found = re.search(r"(?<!\d)(20\d{2})(\d{2})(\d{2})(?!\d)", name)
     36     if found:
     37         try:
     38             return dt.date(int(found.group(1)), int(found.group(2)), int(found.group(3)))
     39         except ValueError:
     40             pass
     41     for key in ("last_modified", "metadata_modified", "created"):
     42         value = str(resource.get(key) or "")[:10]
     43         try:
     44             return dt.date.fromisoformat(value)
     45         except ValueError:
     46             continue
     47     return None

     57 def select_recent_resource(resources: Iterable[Mapping[str, Any]], max_bytes: int = MAX_RESOURCE_BYTES) -> dict[str, Any]:
     58     """Escolhe um recurso JSON mensal, recente e pequeno; a seleção é única."""
     59     candidates: list[tuple[dt.date, int, Mapping[str, Any]]] = []
     60     for resource in resources:
     61         url = str(resource.get("url") or "")
     62         fmt = str(resource.get("format") or "").lower()
     63         name = str(resource.get("name") or url).lower()
     64         if not url.startswith(("http://", "https://")) or not (fmt == "json" or name.endswith(".json")):
     65             continue
     66         size = _resource_size(resource)
     67         if size is not None and size > max_bytes:
     68             continue
     69         candidates.append((_resource_date(resource) or dt.date.min, size if size is not None else max_bytes, resource))
     70     if not candidates:
     71         raise ValueError("pacote STJ não oferece recurso JSON mensal pequeno")
     72     # Data desc, depois tamanho asc e URL estável. Uma única data é baixada.
     73     candidates.sort(key=lambda item: (-item[0].toordinal(), item[1], str(item[2].get("url"))))
     74     return dict(candidates[0][2])

     83 def _year(value: str) -> int | None:
     84     found = re.match(r"(20\d{2})", value)

     88 def _record(entry: Mapping[str, Any], resource: Mapping[str, Any]) -> dict[str, Any] | None:
     89     # O ID numérico é a chave pública oficial do espelho; sem ele não há
     90     # candidato identificável e portanto não se cria um ID inventado.
     91     official_id = _text(entry.get("id"))
     92     ementa = _text(entry.get("ementa"))
     93     if not official_id or not ementa:
     94         return None
     95     processo = _text(entry.get("numeroProcesso"))
     96     registro = _text(entry.get("numeroRegistro"))
     97     orgao = _text(entry.get("nomeOrgaoJulgador")) or COLLECTION.upper()
     98     data_decisao = _text(entry.get("dataDecisao"))
     99     classe = _text(entry.get("siglaClasse"))
    100     title_prefix = classe or "Acórdão"
    101     title = f"{title_prefix} — Processo {processo}" if processo else f"{title_prefix} STJ — {official_id}"
    102     value: dict[str, Any] = {
    103         "id": official_id,
    104         "kind": "case-law",
    105         "title": title,
    106         "abstract": ementa,
    107         "ementa": ementa,
    108         "source": SOURCE,
    109         "license": "CC BY",
    110         "link": str(resource.get("url") or PACKAGE_URL),
    111         "numeroProcesso": processo,
    112         "numeroRegistro": registro,
    113         "nomeOrgaoJulgador": orgao,
    114         "dataDecisao": data_decisao,
    115         "termosAuxiliares": entry.get("termosAuxiliares"),
    116     }
    117     source_year = _year(data_decisao)
    118     if source_year is not None:
    119         value["source_year"] = source_year
    120     return value

    123 def records_from_json(data: Any, resource: Mapping[str, Any]) -> list[dict[str, Any]]:
    124     if isinstance(data, dict):
    125         for key in ("result", "records", "items", "data"):
    126             if isinstance(data.get(key), list):
    127                 data = data[key]
    128                 break
    129     if not isinstance(data, list):
    130         raise ValueError("recurso STJ não é uma lista JSON de acórdãos")
    131     dedup: dict[str, dict[str, Any]] = {}
    132     for item in data:
    133         if not isinstance(item, Mapping):
    134             continue
    135         value = _record(item, resource)
    136         if value:
    137             dedup.setdefault(value["id"], value)
    138     return [dedup[key] for key in sorted(dedup)]

    141 def import_stj(output_dir: str | os.PathLike[str], package_url: str = PACKAGE_URL) -> dict[str, Any]:
    142     package = fetch_json(package_url)
    143     if not isinstance(package, Mapping) or not package.get("success", True):
    144         raise ValueError("package_show do STJ não retornou sucesso")
    145     result = package.get("result")
    146     if not isinstance(result, Mapping):
    147         raise ValueError("package_show do STJ sem result")
    148     resource = select_recent_resource(result.get("resources", []))
    149     data = fetch_json(str(resource["url"]))
    150     entries = records_from_json(data, resource)
    151     return write_catalog_file(
    152         output_dir,
    153         "stj_quarta_turma.jsonl",
    154         entries,
    155         metadata={
    156             "stj_entry_count": len(entries),
    157             "stj_package": PACKAGE_NAME,
    158             "stj_package_url": package_url,
    159             "stj_resource_name": resource.get("name"),
    160             "stj_resource_url": resource.get("url"),
    161             "stj_resource_selection": "um JSON mensal recente, máximo 10 MB",
    162             "stj_license": "CC BY",
    163         },
    164     )
