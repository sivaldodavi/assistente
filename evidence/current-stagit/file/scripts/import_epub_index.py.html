     29 def _text(value: str) -> str:
     30     value = html.unescape(value).replace("\xa0", " ")
     31     return re.sub(r"\s+", " ", value).strip()

     34 class _IndexParser(HTMLParser):
     35     """Extrai relações h4_b -> h4_col -> p.index, sem navegar capítulos."""
     36 
     37     def __init__(self) -> None:
     38         super().__init__(convert_charrefs=True)
     39         self.term = ""
     40         self.norma = ""
     41         self.in_linebox = 0
     42         self.capture: tuple[str, str] | None = None
     43         self.buffer: list[str] = []
     44         self.rows: list[tuple[str, str, str]] = []
     45 
     46     @staticmethod
     47     def _classes(attrs: list[tuple[str, str | None]]) -> set[str]:
     48         raw = dict(attrs).get("class") or ""
     49         return set(raw.split())
     50 
     51     def handle_starttag(self, tag: str, attrs: list[tuple[str, str | None]]) -> None:
     52         classes = self._classes(attrs)
     53         if tag == "div" and "linebx" in classes:
     54             self.in_linebox += 1
     55             self.norma = ""
     56         if tag == "h4" and "h4_b" in classes:
     57             self.capture = ("h4", "term")
     58             self.buffer = []
     59         elif tag == "h4" and self.in_linebox and classes.intersection({"h4_col", "h4_col2", "h4_col3"}):
     60             self.capture = ("h4", "norma")
     61             self.buffer = []
     62         elif tag == "p" and self.in_linebox and "index" in classes:
     63             self.capture = ("p", "remissao")
     64             self.buffer = []
     65 
     66     def handle_data(self, data: str) -> None:
     67         if self.capture:
     68             self.buffer.append(data)
     69 
     70     def handle_endtag(self, tag: str) -> None:
     71         if self.capture and self.capture[0] == tag:
     72             field = self.capture[1]
     73             value = _text("".join(self.buffer))
     74             self.capture = None
     75             self.buffer = []
     76             if field == "term":
     77                 self.term = value
     78             elif field == "norma":
     79                 self.norma = value
     80             elif field == "remissao":
     81                 value = re.sub(r"^[-–—]\s*", "", value).strip()
     82                 if self.term and self.norma and value:
     83                     self.rows.append((self.term, self.norma, value))
     84         if tag == "div" and self.in_linebox:
     85             self.in_linebox -= 1
     86             if self.in_linebox == 0:
     87                 self.norma = ""

     90 def extract_index(epub_path: str | os.PathLike[str]) -> list[dict[str, Any]]:
     91     """Lê apenas bm01/bm01a/bm01b e retorna metadados de remissões."""
     92     rows: dict[str, dict[str, Any]] = {}
     93     with ZipFile(epub_path) as archive:
     94         names = set(archive.namelist())
     95         missing = [name for name in ALLOWED_PATHS if name not in names]
     96         if missing:
     97             raise ValueError("EPUB sem índice autorizado: " + ", ".join(missing))
     98         for name in ALLOWED_PATHS:
     99             parser = _IndexParser()
    100             with archive.open(name) as handle:
    101                 parser.feed(handle.read().decode("utf-8", errors="replace"))
    102             for term, norma, remissao in parser.rows:
    103                 key = "\x00".join((term, norma, remissao))
    104                 identifier = "vademecum-2023:" + hashlib.sha256(key.encode("utf-8")).hexdigest()[:20]
    105                 rows.setdefault(
    106                     identifier,
    107                     {
    108                         "id": identifier,
    109                         "kind": "legislation",
    110                         "title": term,
    111                         "reference": f"{norma} — {remissao}",
    112                         "source_year": 2023,
    113                         "source": SOURCE,
    114                         "term": term,
    115                         "norma": norma,
    116                         "remissao": remissao,
    117                     },
    118                 )
    119     return [rows[key] for key in sorted(rows)]

    122 def import_epub(epub_path: str | os.PathLike[str], output_dir: str | os.PathLike[str]) -> dict[str, Any]:
    123     entries = extract_index(epub_path)
    124     return write_catalog_file(
    125         output_dir,

    127         entries,
    128         metadata={
    129             "vademecum_paths": list(ALLOWED_PATHS),
    130             "vademecum_entry_count": len(entries),
    131             "reference_year": 2023,
    132             "source_book": "Vade Mecum Método",
    133             "source_book_edition": 16,
    134         },
    135     )
