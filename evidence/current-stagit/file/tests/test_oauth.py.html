     20 def test_google_start_uses_state_nonce_and_pkce(app):
     21     configure_google(app)
     22     with app.test_client() as client:
     23         response = client.get("/oauth/google/start")
     24         assert response.status_code == 302
     25         query = parse_qs(urlsplit(response.headers["Location"]).query)
     26         assert query["client_id"] == ["client-id"]
     27         assert query["code_challenge_method"] == ["S256"]
     28         assert len(query["state"][0]) >= 32
     29         assert len(query["nonce"][0]) >= 32
     30         assert len(query["code_challenge"][0]) == 43
     31         with client.session_transaction() as session:
