     87 def test_export_separates_report_and_dissertation(client):
     88     url, token, _ = start(client)
     89     client.post(url+'/save', headers={'X-CSRF-Token': token}, json={'revision': 1, 'payload': {'fields': {'facts': 'Narrativa do relatório', 'dissertation_intro': 'Introdução individual', 'dissertation_conclusion': 'Conclusão individual'}, 'development_sections': [{'id': 'topic-1', 'title': 'Fundamentação', 'body': 'Desenvolvimento individual'}]}})
     90     report = client.get(url+'/export?phase=report').get_data(as_text=True)
     91     dissertation = client.get(url+'/export?phase=dissertation').get_data(as_text=True)
     92     assert 'Narrativa do relatório' in report and 'Introdução individual' not in report
     93     assert 'Introdução individual' in dissertation and 'Narrativa do relatório' not in dissertation
     94     assert dissertation.index('Introdução individual') < dissertation.index('Desenvolvimento individual') < dissertation.index('Conclusão individual')
     95     assert client.get(url+'/export?phase=unrecognized').status_code == 400
