     14 // ---- Capa e folha de rosto ---------------------------------------
     15 // Layout medido no PDF do TGPC (Relatório de case):
     16 //   - instituição e curso no topo, centralizados
     17 //   - título em caixa alta e negrito no terço central
     18 //   - autores logo abaixo do título, alinhados à direita
     19 //   - bloco de dados (disciplina, professor, turma, data) à direita
     20 //   - local e ano centralizados na base
     21 // folha-rosto: true gera uma segunda página com a natureza do trabalho.
     22 #let abnt-capa(
     23   instituicao: "",
     24   curso: "",
     25   titulo: "",
     26   subtitulo: "",
     27   autores: (),
     28   dados: (),
     29   natureza: "",
     30   orientador: "",
     31   local: "",
     32   ano: "",
     33   folha-rosto: false,
     34 ) = {
     35   let auts = if autores.len() == 0 { "" } else { autores.join(", ", last: " e ") }
     36   let linha-entrelinha = abnt-entrelinha-1-5
     37 
     38   let topo = [
     39     #align(center)[
     40       #text(weight: "bold")[#upper(instituicao)] \
     41       #upper(curso)
     42     ]
     43   ]
     44 
     45   let meio = [
     46     #align(center)[#block(width: 84%)[
     47       #set align(center)
     48       #text(weight: "bold")[#upper(titulo)] \
     49       #if subtitulo != "" [#subtitulo \ ]
     50     ]]
     51   ]
     52 
     53   let direita = [
     54     #align(right)[
     55       #set par(first-line-indent: (amount: 0pt, all: true), leading: 1.04em, spacing: 0pt, justify: false)
     56       #auts #linebreak()
     57       #for d in dados [
     58         #d.replace("~", " ") #linebreak()
     59       ]
     60     ]
     61   ]
     62 
     63   let base = [
     64     #align(center)[#local \ #ano]
     65   ]
     66 
     67   page(numbering: none)[
     68     #set par(first-line-indent: (amount: 0pt, all: true), leading: linha-entrelinha, spacing: 0pt, justify: false)
     69     #v(3pt)
     70     #topo
     71     #v(198.7pt)
     72     #meio
     73     #v(12.5pt)
     74     #direita
     75     #place(bottom + center, dy: -6pt)[#base]
     76   ]
     77 
     78   if folha-rosto {
     79     page(numbering: none)[
     80       #set par(first-line-indent: (amount: 0pt, all: true), leading: linha-entrelinha, spacing: 0pt, justify: false)
     81       #topo
     82       #v(120pt)
     83       #meio
     84       #v(1fr)
     85       #block(width: 8cm)[
     86         #set par(first-line-indent: (amount: 0pt, all: true), leading: abnt-simples, spacing: 1em, justify: true)
     87         #set text(size: abnt-nota)
     88         #natureza
     89         #if orientador != "" [\ \ Orientador(a): #orientador]
     90       ]
     91       #v(1fr)
     92       #base
     93     ]
     94   }

     99   titulo: none,
    100   numeracao: "1",
    101   numeracao-secoes: "1.1",
    102   pagina-inicial: none,
    103   quebra-secoes: false,
    104   entrelinha: 1.04em,
    105   fonte: abnt-fonte-padrao,
    106   capa: none,
    107   sumario: false,
    108   corpo,
    109 ) = {
    110   let num-pagina = if numeracao == "none" { none } else { numeracao }
    111   let num-secao = if numeracao-secoes == "none" { none } else { numeracao-secoes }
    112   let familia = if type(fonte) == str { (fonte, "Liberation Sans") } else { fonte }
    113 
    114   set document(title: titulo) if titulo != none
    115 
    116   // A4; margens: sup. 3 cm, inf. 2 cm, esq. 3 cm, dir. 2 cm
    117   set page(
    118     paper: "a4",
    119     margin: (top: 3cm, left: 3cm, bottom: 2cm, right: 2cm),
    120     numbering: num-pagina,
    121     number-align: top + right,
    122   )
    123 
    124   // Fonte 12 pt, pt-BR, sem hifenização
    125   set text(font: familia, size: abnt-corpo, lang: "pt", region: "BR", hyphenate: false)
    126 
    127   // Parágrafo: entrelinha 1,5, recuo 1,25 cm na 1ª linha, justificado
    128   set par(
    129     leading: entrelinha,
    130     spacing: 1em,
    131     justify: true,
    132     first-line-indent: (amount: abnt-recuo, all: true),
    133   )
    134 
    135   if pagina-inicial != none {
    136     counter(page).update(pagina-inicial)

    141   // Títulos: nível 1 = caixa alta + negrito; nível 2 = negrito; nível 3 = itálico

    143     let numero = if it.numbering != none {
    144       counter(heading).display(it.numbering) + h(0.5em)
    145     } else { none }
    146 
    147     let conteudo = if it.level == 1 {
    148       text(weight: "bold", size: abnt-corpo)[#numero#upper(it.body)]
    149     } else if it.level == 2 {
    150       text(weight: "bold", size: abnt-corpo)[#numero#it.body]
    151     } else {
    152       text(weight: "regular", style: "italic", size: abnt-corpo)[#numero#it.body]
    153     }
    154 
    155     v(1em)
    156     block(width: 100%)[
    157       #set par(first-line-indent: (amount: 0pt, all: true))
    158       #conteudo
    159     ]
    160     v(1em)
    161   }
    162 
    163   // Notas de rodapé: 10 pt, simples, sem recuo
    164   show footnote.entry: set text(size: abnt-nota)
    165   show footnote.entry: set footnote.entry(indent: 0pt, gap: 0.5em)
    166   show footnote.entry: set par(
    167     leading: abnt-simples,
    168     spacing: 0pt,
    169     first-line-indent: (amount: 0pt, all: true),
    170     justify: true,
    171   )
    172 
    173   // Citação direta longa (> ...): recuo 4 cm, 10 pt, simples, sem aspas
    174   show quote.where(block: true): it => {
    175     v(1em)
    176     block(inset: (left: 4cm), width: 100%)[
    177       #set par(first-line-indent: (amount: 0pt, all: true), leading: abnt-simples, spacing: 0pt, justify: true)
    178       #set text(size: abnt-nota)
    179       #it.body
    180     ]
    181     v(1em)
    182   }
    183 
    184   if capa != none {
    185     abnt-capa(..capa)
    186   }
    187   if sumario {
    188     abnt-sumario()

    232 #let _ref(chave, texto) = (chave: chave, texto: texto)

    234 #let _chave(s) = {
    235   let t = lower(s)
    236   for p in ("o ", "a ", "os ", "as ", "um ", "uma ") {
    237     if t.starts-with(p) { t = t.slice(p.len()) }
    238   }
    239   t
    240 }

    242 // Autor: sobrenome em caixa alta. Suporta múltiplos autores separados por ";".
    243 #let _autor(a) = {
    244   let autores = a.split(";")
    245   autores.map(x => {
    246     let x = x.trim()
    247     if x == "et al." or x == "et al" { return x }
    248     let p = x.split(",")
    249     if p.len() > 1 {
    250       upper(p.at(0).trim()) + "," + p.slice(1).join(",")
    251     } else {
    252       upper(x)
    253     }
    254   }).join("; ")
    255 }

    257 #let _dispo(url, acesso) = {
    258   let d = if url != none { " Disponível em: " + url + "." } else { "" }
    259   let a = if acesso != none { " Acesso em: " + acesso + "." } else { "" }
    260   d + a
    261 }

    263 #let ref-livro(autor, titulo, local, editora, ano,
