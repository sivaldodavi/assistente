Fix: Add file-based migration lock to prevent SQLite locking issues in multi-worker gunicorn

The init_db() function was being called in each gunicorn worker process,
causing concurrent SQLite migrations that led to 'unable to open database file'
errors due to WAL mode locking conflicts. Added a file-based lock using fcntl
to ensure only one worker runs migrations at a time.

+def _migration_lock(database_path: Path):
+    """File-based lock to prevent concurrent migrations in multi-worker environments."""
+    lock_file = database_path.with_suffix(".migration.lock")
+    lock_fd = os.open(lock_file, os.O_CREAT | os.O_RDWR)
+    try:
+        fcntl.flock(lock_fd, fcntl.LOCK_EX | fcntl.LOCK_NB)
+        return lock_fd
+    except BlockingIOError:
+        os.close(lock_fd)
+        return None
+
+
+def _release_migration_lock(lock_fd):
+    if lock_fd is not None:
+        fcntl.flock(lock_fd, fcntl.LOCK_UN)
+        os.close(lock_fd)
