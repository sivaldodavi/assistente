+import time
+from uuid import uuid4
+
+from .db import get_db

+SOURCE_KINDS = {"book", "article", "legislation", "case_law", "website", "other"}
+
+
+def create_source(account_id: str, title: str, author: str, kind: str, reference: str, notes: str) -> str:
+    source_id = str(uuid4())
+    get_db().execute(
+        """INSERT INTO sources (id, account_id, title, author, kind, reference, notes, created_at, updated_at)
+           VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)""",
+        (
+            source_id,
+            account_id,
+            str(title or "").strip()[:240],
+            str(author or "").strip()[:240],
+            kind if kind in SOURCE_KINDS else "other",
+            str(reference or "").strip()[:500],
+            str(notes or "").strip()[:1000],
+            int(time.time()),
+            int(time.time()),
+        ),
+    )
+    get_db().commit()
+    return source_id
+
+
+def list_sources(account_id: str):
+    rows = get_db().execute(
+        """SELECT id, title, author, kind, reference, notes, created_at, updated_at
+           FROM sources WHERE account_id = ? ORDER BY updated_at DESC, title COLLATE NOCASE""",
+        (account_id,),
+    ).fetchall()
+    return [dict(row) for row in rows]
+
+
+def owned_source_ids(account_id: str, source_ids: list[str]) -> set[str]:
+    clean = {str(source_id) for source_id in source_ids if isinstance(source_id, str) and source_id}
+    if not clean:
+        return set()
+    placeholders = ",".join("?" for _ in clean)
+    rows = get_db().execute(
+        f"SELECT id FROM sources WHERE account_id = ? AND id IN ({placeholders})",
+        (account_id, *clean),
+    ).fetchall()
+    return {row["id"] for row in rows}

+# Biblioteca Pessoal
+
+## Objetivo
+
+A Biblioteca Pessoal guarda referências que pertencem à conta e podem ser reutilizadas em mais de um projeto. Ela não é uma lixeira de textos nem uma lista pública de obras.
+
+## Registro de fonte
+
+Uma fonte possui título obrigatório, autor/órgão opcional, tipo, referência bibliográfica ou link e nota pessoal curta. O tipo é controlado: livro, artigo, legislação, jurisprudência, site ou outro.
+
+A fonte não recebe uma descrição acadêmica longa. A análise, o fichamento e a relação com teses pertencem aos recursos que usam a fonte.
+
+## Vínculo com projeto
+
+A Linha do Tempo e as etapas acadêmicas guardam somente IDs de fontes. No servidor, cada ID é validado contra a conta proprietária antes de ser salvo. Um usuário não consegue vincular ou descobrir a fonte de outra conta enviando um ID arbitrário.
+
+O editor apresenta fontes disponíveis em seleção múltipla. Não é necessário digitar UUID ou texto solto para criar um vínculo.
+
+## Experiência visual
+
+A biblioteca usa uma coluna curta para cadastro e uma lista plana com divisórias para leitura. Título, autor/tipo e referência aparecem primeiro; a nota pessoal fica recolhida quando existe. O layout deve funcionar em uma coluna no celular e continuar usando os componentes escuros do site.
+
+## Próximas extensões
+
+Fichamento estruturado, classificação pessoal, busca e citações só entram depois que o cadastro e o vínculo básico estiverem estáveis. A tabela não será expandida com campos sem fluxo de usuário definido.
