+#!/bin/bash
+# Uso: deploy-site <website|direito|teologia|assistente-academico> [ref]

+  direito) SITE_DIR="direito"; SERVICE="direito" ;;
+  teologia) SITE_DIR="teologia"; SERVICE="teologia" ;;
+  assistente-academico) SITE_DIR="assistente-academico"; SERVICE="assistente-academico" ;;
+  *) echo "Uso: deploy-site <website|direito|teologia|assistente-academico> [ref]"; exit 1 ;;
+esac

+DEST="$SERVIDOR_DIR/sites/$SITE_DIR"

+[ -d "$DEST" ] || { echo "ERRO: diretório de publicação não existe"; exit 1; }

+  for file in Dockerfile requirements.txt wsgi.py app/__init__.py app/store/schema.sql; do
+    [ -s "$STAGE/$file" ] || { echo "ERRO: falta $file na aplicação Python; deploy cancelado"; exit 1; }

+  [ -d "$STAGE/templates" ] && [ -d "$STAGE/static" ] || exit 1

+COMPOSE_ARGS=(--project-directory "$SERVIDOR_DIR" -f "$SERVIDOR_DIR/docker-compose.yml")
