+from __future__ import annotations
+
+import json
+import os
+from pathlib import Path
+from urllib.error import HTTPError, URLError
+from urllib.request import Request, urlopen
+
+
+class ProviderError(RuntimeError):
+    def __init__(self, code: str, message: str, status: int = 503):
+        self.code = code
+        self.status = status
+        super().__init__(message)
+
+
+def load_secret(env_name: str, file_env_name: str, default_file: str = "") -> str:
+    value = os.environ.get(env_name, "").strip()
+    path = os.environ.get(file_env_name, default_file)
+    if not value and path:
+        try:
+            value = Path(path).read_text(encoding="utf-8").strip()
+        except (OSError, UnicodeError) as error:
+            raise ProviderError("secret_unavailable", "A configuração do provedor não pôde ser carregada.") from error
+    if not value or len(value) > 4096 or "\n" in value or "\r" in value:
+        raise ProviderError("secret_invalid", "A configuração do provedor é inválida.")
+    return value
+
+
+class _JsonProvider:
+    def __init__(self, api_key: str, api_url: str, model: str, opener=urlopen):
+        self.api_key = api_key
+        self.api_url = api_url
+        self.model = model
+        self.opener = opener
+
+    def post(self, payload: dict, timeout: int = 30) -> dict:
+        body = json.dumps(payload, ensure_ascii=False, separators=(",", ":")).encode("utf-8")
+        request = Request(
+            self.api_url,
+            data=body,
+            headers={"Authorization": f"Bearer {self.api_key}", "Content-Type": "application/json"},
+            method="POST",
+        )
+        try:
+            with self.opener(request, timeout=timeout) as response:
+                raw = response.read(4 * 1024 * 1024 + 1)
+        except HTTPError as error:
+            status = error.code
+            error.close()
+            if status in {429, 529}:
+                raise ProviderError("provider_busy", "O provedor está temporariamente ocupado.", 503) from error
+            if status == 401:
+                raise ProviderError("provider_auth", "A configuração do provedor foi recusada.", 502) from error
+            raise ProviderError("provider_error", "O provedor não concluiu a solicitação.", 502) from error
+        except (TimeoutError, URLError, OSError) as error:
+            raise ProviderError("provider_unreachable", "Não foi possível conectar ao provedor.", 503) from error
+        if len(raw) > 4 * 1024 * 1024:
+            raise ProviderError("response_too_large", "A resposta do provedor é grande demais.", 502)
+        try:
+            data = json.loads(raw)
+        except (TypeError, ValueError, json.JSONDecodeError) as error:
+            raise ProviderError("invalid_response", "O provedor retornou uma resposta inválida.", 502) from error
+        if not isinstance(data, dict):
+            raise ProviderError("invalid_response", "O provedor retornou uma resposta inválida.", 502)
+        return data
+
+
+class DeepSeekClient(_JsonProvider):
+    def __init__(self, api_key: str, api_url: str = "https://api.deepseek.com/chat/completions", model: str = "deepseek-chat", opener=urlopen):
+        super().__init__(api_key, api_url, model, opener)
+
+    def complete_json(self, messages: list[dict], max_tokens: int = 1400) -> dict:
+        response = self.post({"model": self.model, "messages": messages, "response_format": {"type": "json_object"}, "max_tokens": max_tokens, "stream": False})
+        choices = response.get("choices")
+        content = choices[0].get("message", {}).get("content") if isinstance(choices, list) and len(choices) == 1 else None
+        if not isinstance(content, str):
+            raise ProviderError("invalid_response", "A resposta DeepSeek não trouxe conteúdo JSON.", 502)
+        try:
+            result = json.loads(content)
+        except (TypeError, ValueError, json.JSONDecodeError) as error:
+            raise ProviderError("invalid_response", "A resposta DeepSeek não trouxe JSON válido.", 502) from error
+        if not isinstance(result, dict):
+            raise ProviderError("invalid_response", "A resposta DeepSeek não trouxe um objeto JSON.", 502)
+        return {"model": response.get("model") or self.model, "result": result, "usage": response.get("usage", {})}
+
+
+class JevClient(_JsonProvider):
+    def __init__(self, api_key: str, api_url: str = "https://api.typesafe.ai/v1/systemone", model: str = "jev-latest", opener=urlopen):
+        super().__init__(api_key, api_url, model, opener)
+
+    def score(self, state: dict, questions: dict) -> dict:
+        response = self.post({"state": state, "model": self.model, "questions": questions})
+        answers = response.get("answers")
+        if not isinstance(answers, dict):
+            raise ProviderError("invalid_response", "A resposta JEV não trouxe respostas estruturadas.", 502)
+        return {"model": response.get("model") or self.model, "answers": answers, "usage": response.get("usage", {})}

+# Provedores externos de IA
+
+## Fronteira
+
+DeepSeek e JEV/Typesafe são serviços externos atrás de adaptadores em `app/services/providers.py`. Rotas e domínio não montam `urllib.Request` diretamente.
+
+Os adaptadores recebem a chave, a URL e um transporte injetável. O transporte falso é usado na suíte; testes não fazem chamadas reais.
+
+## DeepSeek
+
+O contrato envia `model`, `messages`, `response_format` JSON, `max_tokens` e `stream: false` ao endpoint configurado. A resposta precisa ter uma única escolha com conteúdo JSON objeto. O adaptador devolve modelo, resultado estruturado e uso; não interpreta a regra acadêmica.
+
+## JEV/Typesafe
+
+O contrato envia `state`, `model` e `questions` ao endpoint configurado. A resposta precisa conter `answers` como objeto. A validação detalhada de rubric, itens permitidos, confiança e persistência de uso pertence ao serviço de aplicação que chamará o adaptador, não ao transporte.
+
+## Secrets
+
+A chave pode vir de variável de ambiente ou de arquivo indicado por `*_API_KEY_FILE`, compatível com secrets montados pelo Docker. O valor não é impresso, salvo no banco, enviado ao frontend ou incluído em exceções.
+
+## Falhas
+
+401 vira erro de configuração; 429/529 vira indisponibilidade temporária; timeout e erro de rede não são convertidos em resposta de sucesso. Corpo de erro do provedor não é registrado no log.
+
+## Limites do produto
+
+Nenhum adaptador libera sozinho uma chamada para um projeto. Consentimento de IA, reserva de uso, contexto mínimo, ação permitida, validação da resposta e registro de uso serão contratos separados. O provedor nunca decide o conteúdo do projeto sem confirmação explícita do usuário.

+import json
+
+import pytest
+
+from app.services.providers import DeepSeekClient, JevClient, ProviderError, load_secret
+
+
+class FakeResponse:
+    def __init__(self, payload):
+        self.payload = json.dumps(payload).encode()
+
+    def __enter__(self):
+        return self
+
+    def __exit__(self, *_):
+        return None
+
+    def read(self, _limit):
+        return self.payload
+
+
+def fake_opener(payload):
+    def opener(request, timeout):
+        assert request.get_method() == "POST"
+        assert request.headers["Authorization"] == "Bearer secret"
+        assert timeout == 30
+        return FakeResponse(payload)
+    return opener
+
+
+def test_deepseek_client_parses_structured_json_without_network():
+    client = DeepSeekClient("secret", opener=fake_opener({"model": "deepseek-chat", "choices": [{"message": {"content": '{"answer":"ok"}'}}], "usage": {"prompt_tokens": 1}}))
+    result = client.complete_json([{"role": "user", "content": "teste"}])
+    assert result["result"] == {"answer": "ok"}
+
+
+def test_jev_client_preserves_structured_answers():
+    client = JevClient("secret", opener=fake_opener({"model": "jev-latest", "answers": {"q1": {"score": 2}}, "usage": {"input_tokens": 1}}))
+    result = client.score({"text": "teste"}, {"q1": {"type": "score"}})
+    assert result["answers"]["q1"]["score"] == 2
+
+
+def test_provider_rejects_invalid_json_content():
+    client = DeepSeekClient("secret", opener=fake_opener({"choices": [{"message": {"content": "não-json"}}]}))
+    with pytest.raises(ProviderError) as error:
+        client.complete_json([])
+    assert error.value.code == "invalid_response"
+
