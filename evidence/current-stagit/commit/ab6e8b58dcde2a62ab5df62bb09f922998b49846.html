+@oauth_bp.get("/oauth/google/start")
+def google_start():
+    client = _client()
+    if not client.config.configured:
+        return redirect(url_for("public.home", google_login="unconfigured"))
+    state = new_state()
+    nonce = new_nonce()
+    code_verifier = new_code_verifier()
+    session["google_oauth"] = {"state": state, "nonce": nonce, "code_verifier": code_verifier}
+    try:
+        location = client.authorization_url(state, nonce, code_verifier)
+    except GoogleOAuthError:
+        session.pop("google_oauth", None)
+        return redirect(url_for("public.home", google_login="failed"))
+    return redirect(location)
+
+
+@oauth_bp.get("/oauth/google/callback")
+def google_callback():
+    flow = session.pop("google_oauth", None)
+    state = request.args.get("state", "")
+    code = request.args.get("code", "")
+    if (
+        not isinstance(flow, dict)
+        or not isinstance(state, str)
+        or not state
+        or not secrets.compare_digest(state, str(flow.get("state", "")))
+        or request.args.get("error")
+        or not isinstance(code, str)
+        or not code
+        or len(code) > 2048
+    ):
+        return redirect(url_for("public.home", google_login="failed"))
+    try:
+        claims = _client().exchange_code(code, str(flow["code_verifier"]), str(flow["nonce"]))
+        account_id = upsert_google_account(claims)
+    except (GoogleOAuthError, KeyError, TypeError, ValueError):
+        return redirect(url_for("public.home", google_login="failed"))
+    session.clear()
+    session["account_id"] = account_id
+    csrf_token()
+    return redirect(url_for("projects.account"))

+import base64
+import hashlib

+from urllib.parse import urlencode, urlsplit

+class GoogleOAuthError(Exception):
+    """Erro esperado ao iniciar ou validar um fluxo OAuth Google."""
+
+
+@dataclass(frozen=True)
+class GoogleOAuthConfig:
+    client_id: str
+    client_secret: str
+    redirect_uri: str
+    authorization_endpoint: str = "https://accounts.google.com/o/oauth2/v2/auth"
+    token_endpoint: str = "https://oauth2.googleapis.com/token"
+
+    @classmethod
+    def from_mapping(cls, mapping):
+        return cls(
+            client_id=str(mapping.get("GOOGLE_CLIENT_ID", "") or "").strip(),
+            client_secret=str(mapping.get("GOOGLE_CLIENT_SECRET", "") or "").strip(),
+            redirect_uri=str(mapping.get("GOOGLE_REDIRECT_URI", "") or "").strip(),
+        )
+
+    @property
+    def configured(self):
+        return bool(self.client_id and self.client_secret and valid_redirect_uri(self.redirect_uri))
+
+
+def valid_redirect_uri(value: str) -> bool:
+    try:
+        parsed = urlsplit(value)
+        port = parsed.port
+        return (
+            parsed.scheme in {"https", "http"}
+            and bool(parsed.hostname)
+            and parsed.path == "/oauth/google/callback"
+            and parsed.username is None
+            and parsed.password is None
+            and not parsed.query
+            and not parsed.fragment
+            and (port is None or 1 <= port <= 65535)
+            and (parsed.scheme == "https" or parsed.hostname in {"localhost", "127.0.0.1", "::1"})
+        )
+    except ValueError:
+        return False
+
+
+def new_state() -> str:
+    return secrets.token_urlsafe(32)
+
+
+def new_nonce() -> str:
+    return secrets.token_urlsafe(32)
+
+
+def new_code_verifier() -> str:
+    return secrets.token_urlsafe(64)
+
+
+def pkce_challenge(verifier: str) -> str:
+    digest = hashlib.sha256(verifier.encode("ascii")).digest()
+    return base64.urlsafe_b64encode(digest).rstrip(b"=").decode("ascii")
+
+
+class GoogleOAuthClient:
+    def __init__(self, config: GoogleOAuthConfig):
+        self.config = config
+
+    def authorization_url(self, state: str, nonce: str, code_verifier: str) -> str:
+        if not self.config.configured:
+            raise GoogleOAuthError("google_login_unconfigured")
+        query = urlencode({
+            "client_id": self.config.client_id,
+            "redirect_uri": self.config.redirect_uri,
+            "response_type": "code",
+            "scope": "openid email profile",
+            "state": state,
+            "nonce": nonce,
+            "code_challenge": pkce_challenge(code_verifier),
+            "code_challenge_method": "S256",
+            "access_type": "online",
+            "prompt": "select_account",
+        })
+        return f"{self.config.authorization_endpoint}?{query}"
+
+    def exchange_code(self, code: str, code_verifier: str, expected_nonce: str) -> dict:
+        if not self.config.configured:
+            raise GoogleOAuthError("google_login_unconfigured")
+        if not code or len(code) > 2048 or not code_verifier:
+            raise GoogleOAuthError("google_state_failed")
+        body = urlencode({
+            "code": code,
+            "client_id": self.config.client_id,
+            "client_secret": self.config.client_secret,
+            "redirect_uri": self.config.redirect_uri,
+            "grant_type": "authorization_code",
+            "code_verifier": code_verifier,
+        }).encode("ascii")
+        request = Request(
+            self.config.token_endpoint,
+            data=body,
+            headers={"Accept": "application/json", "Content-Type": "application/x-www-form-urlencoded"},
+            method="POST",
+        )
+        try:
+            with urlopen(request, timeout=10) as response:
+                payload_bytes = response.read(64 * 1024 + 1)
+            if len(payload_bytes) > 64 * 1024:
+                raise ValueError("OAuth response too large")
+            payload = json.loads(payload_bytes)
+        except (HTTPError, OSError, TimeoutError, ValueError, json.JSONDecodeError) as error:
+            raise GoogleOAuthError("google_token_exchange_failed") from error
+        token = payload.get("id_token") if isinstance(payload, dict) else None
+        if not isinstance(token, str) or not token:
+            raise GoogleOAuthError("google_token_exchange_failed")
+        return self.verify_id_token(token, expected_nonce)
+
+    def verify_id_token(self, token: str, expected_nonce: str) -> dict:
+        try:
+            from google.auth.transport import requests as google_requests
+            from google.oauth2 import id_token
+        except ImportError as error:
+            raise GoogleOAuthError("google_verifier_unavailable") from error
+        try:
+            claims = id_token.verify_oauth2_token(
+                token,
+                google_requests.Request(),
+                self.config.client_id,
+            )
+        except Exception as error:
+            raise GoogleOAuthError("invalid_google_credential") from error
+        if (
+            not isinstance(claims, dict)
+            or claims.get("iss") not in {"accounts.google.com", "https://accounts.google.com"}
+            or claims.get("email_verified") is not True
+            or not isinstance(claims.get("sub"), str)
+            or not claims["sub"]
+            or len(claims["sub"]) > 255
+            or not isinstance(claims.get("email"), str)
+            or not claims["email"].strip()
+            or not isinstance(claims.get("nonce"), str)
+            or not secrets.compare_digest(claims["nonce"], expected_nonce)
+        ):
+            raise GoogleOAuthError("invalid_google_credential")
+        return {
+            "sub": claims["sub"],
+            "email": claims["email"].strip().lower(),
+            "name": claims.get("name", "") if isinstance(claims.get("name"), str) else "",
+        }

+import time
+from uuid import uuid4
+
+from .db import get_db
+
+
+def upsert_google_account(claims: dict) -> str:
+    subject = str(claims.get("sub") or "").strip()[:255]
+    email = str(claims.get("email") or "").strip().lower()[:240]
+    name = str(claims.get("name") or "").strip()[:160]
+    if not subject or not email:
+        raise ValueError("claims Google incompletos")
+    database = get_db()
+    row = database.execute("SELECT id FROM accounts WHERE google_sub = ? OR email = ?", (subject, email)).fetchone()
+    now = int(time.time())
+    if row is None:
+        account_id = str(uuid4())
+        database.execute(
+            "INSERT INTO accounts (id, google_sub, email, name, created_at, last_login) VALUES (?, ?, ?, ?, ?, ?)",
+            (account_id, subject, email, name, now, now),
+        )
+    else:
+        account_id = row["id"]
+        database.execute(
+            "UPDATE accounts SET google_sub = ?, email = ?, name = ?, last_login = ? WHERE id = ?",
+            (subject, email, name, now, account_id),
+        )
+    database.commit()
+    return account_id
