+def get_sharing(account_id: str, project_id: str) -> dict:
+    row = get_db().execute("SELECT * FROM project_sharing WHERE account_id = ? AND project_id = ?", (account_id, project_id)).fetchone()
+    if row is None:
+        return {"project_id": project_id, "allow_admin_view": False, "allow_admin_comments": False}
+    return {"project_id": row["project_id"], "allow_admin_view": bool(row["allow_admin_view"]), "allow_admin_comments": bool(row["allow_admin_comments"])}
+
+
+def save_sharing(account_id: str, project_id: str, allow_view: bool, allow_comments: bool) -> dict | None:
+    database = get_db()
+    exists = database.execute("SELECT id FROM projects WHERE id = ? AND account_id = ?", (project_id, account_id)).fetchone()
+    if exists is None:
+        return None
+    database.execute(
+        """INSERT INTO project_sharing (project_id, account_id, allow_admin_view, allow_admin_comments, updated_at)
+           VALUES (?, ?, ?, ?, ?) ON CONFLICT(project_id) DO UPDATE SET allow_admin_view=excluded.allow_admin_view, allow_admin_comments=excluded.allow_admin_comments, updated_at=excluded.updated_at""",
+        (project_id, account_id, int(allow_view), int(allow_comments and allow_view), int(time.time())),
+    )
+    database.commit()
+    return get_sharing(account_id, project_id)
+
+
+def list_comments(project_id: str) -> list[dict]:
+    rows = get_db().execute("SELECT body, author_role, created_at FROM project_comments WHERE project_id = ? ORDER BY created_at", (project_id,)).fetchall()
+    return [dict(row) for row in rows]
+
+
+def list_shared_projects() -> list[dict]:
+    rows = get_db().execute("""SELECT p.id, p.title, p.project_type, p.updated_at, a.email, s.allow_admin_comments
+                             FROM projects p JOIN project_sharing s ON s.project_id=p.id JOIN accounts a ON a.id=p.account_id
+                             WHERE s.allow_admin_view=1 ORDER BY p.updated_at DESC""").fetchall()
+    return [dict(row) for row in rows]
+
+
+def get_shared_project(project_id: str) -> dict | None:
+    row = get_db().execute("""SELECT p.*, s.allow_admin_comments, a.email
+                            FROM projects p JOIN project_sharing s ON s.project_id=p.id JOIN accounts a ON a.id=p.account_id
+                            WHERE p.id=? AND s.allow_admin_view=1""", (project_id,)).fetchone()
+    return dict(row) if row else None
+
+
+def add_admin_comment(project_id: str, admin_account_id: str, body: str) -> bool:
+    body = str(body or "").strip()[:3000]

+    get_db().commit()
+    return True

+# Compartilhamento e administração
+
+## Privacidade por padrão
+
+Cada projeto começa privado. O proprietário pode autorizar visualização administrativa e, separadamente, comentários administrativos. A segunda autorização não é ativada quando a primeira é salva sem a segunda.
+
+Revogar visualização impede novas aberturas administrativas. O texto acadêmico permanece no snapshot do proprietário; comentários ficam em `project_comments` e nunca são misturados aos campos do projeto.
+
+## Administração
+
+A v2 não promove o primeiro usuário automaticamente. Administradores são definidos por `ASSISTENTE_V2_ADMIN_EMAILS`, uma lista de e-mails configurada fora do Git. A rota administrativa exige sessão e essa configuração.
+
+O administrador vê apenas projetos com `allow_admin_view=1` e abre uma apresentação somente leitura. Se permitido, pode escrever comentário separado. Não existe endpoint administrativo para salvar o snapshot acadêmico do proprietário.
+
+## Testes
+
+Os testes criam proprietário e administrador fictícios, verificam a revogação implícita por padrão, isolamento de conta e comentário separado. Nenhuma permissão é herdada do banco legado.
