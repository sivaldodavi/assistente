+def queue_admin_login_alert(account_id: str) -> int:
+    account = get_db().execute("SELECT email, name FROM accounts WHERE id = ?", (account_id,)).fetchone()
+    if account is None:
+        return 0
+    admin_emails = current_app.config.get("ADMIN_EMAILS", set())
+    targets = get_db().execute("SELECT id, whatsapp_phone FROM accounts WHERE whatsapp_opt_in = 1 AND whatsapp_phone <> ''").fetchall()
+    text = f"Novo login no Assistente Acadêmico: {account['name'] or 'conta sem nome'} ({account['email']})."
+    queued = 0
+    for target in targets:
+        target_account = get_db().execute("SELECT email FROM accounts WHERE id = ?", (target["id"],)).fetchone()
+        if not target_account or target_account["email"].lower() not in admin_emails or target["id"] == account_id:
+            continue
+        enqueue_notification(account_id=target["id"], destination=target["whatsapp_phone"], kind="admin_login", text=text, idempotency_key=f"admin-login-{target['id']}-{account_id}-{int(time.time())}", commit=False)
+        queued += 1
+    get_db().commit()
+    return queued
