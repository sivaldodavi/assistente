+def normalize_whatsapp_phone(value: object) -> str:
+    if value is None:
+        return ""
+    if not isinstance(value, str):
+        raise ValueError("Informe um telefone WhatsApp válido.")
+    digits = re.sub(r"[+()\s-]", "", value)
+    if not digits.isdigit() or not 10 <= len(digits) <= 15:
+        raise ValueError("Informe um telefone WhatsApp válido com DDI/DDD.")
+    return digits

+def enqueue_notification(*, account_id: str, destination: str, kind: str, text: str, idempotency_key: str, project_id: str | None = None, current_time: int | None = None, commit: bool = True):
+    destination = normalize_whatsapp_phone(destination)
+    if not text or len(text) > 1000:
+        raise ValueError("Mensagem de notificação inválida.")
+    now = int(time.time()) if current_time is None else int(current_time)
+    database = get_db()
+    database.execute(
+        """INSERT OR IGNORE INTO notification_outbox
+           (id, account_id, destination, kind, project_id, idempotency_key, text, status, attempts, next_attempt_at, created_at)
+           VALUES (?, ?, ?, ?, ?, ?, ?, 'pending', 0, ?, ?)""",
+        (str(uuid4()), account_id, destination, kind, project_id, idempotency_key, text.strip(), now, now),
+    )
+    if commit:
+        database.commit()
+    return database.execute("SELECT id, status FROM notification_outbox WHERE idempotency_key = ?", (idempotency_key,)).fetchone()
+
+
+def claim_notifications(limit: int = 20, current_time: int | None = None) -> list[dict]:
+    now = int(time.time()) if current_time is None else int(current_time)
+    database = get_db()
+    database.execute("BEGIN IMMEDIATE")
+    rows = database.execute("SELECT * FROM notification_outbox WHERE status IN ('pending', 'retry') AND next_attempt_at <= ? ORDER BY created_at LIMIT ?", (now, max(1, min(int(limit), 100)))).fetchall()
+    claimed = []
+    for row in rows:
+        database.execute("UPDATE notification_outbox SET status = 'processing' WHERE id = ?", (row["id"],))
+        claimed.append(dict(row))
+    database.commit()
+    return claimed
+
+
+def finish_notification(notification_id: str, current_time: int | None = None) -> None:
+    now = int(time.time()) if current_time is None else int(current_time)
+    database = get_db()
+    database.execute("UPDATE notification_outbox SET status = 'sent', sent_at = ?, last_error = NULL WHERE id = ? AND status = 'processing'", (now, notification_id))
+    database.commit()
+
+
+def cancel_notification(notification_id: str, reason: str = "opt_out") -> None:
+    database = get_db()
+    database.execute("UPDATE notification_outbox SET status = 'cancelled', last_error = ? WHERE id = ? AND status = 'processing'", (reason[:120], notification_id))
+    database.commit()
+
+
+def fail_notification(notification_id: str, error: object, current_time: int | None = None) -> None:
+    now = int(time.time()) if current_time is None else int(current_time)
+    database = get_db()
+    row = database.execute("SELECT attempts FROM notification_outbox WHERE id = ?", (notification_id,)).fetchone()
+    attempts = int(row["attempts"] if row else 0) + 1
+    delay = min(86400, 60 * (2 ** min(attempts, 10)))
+    database.execute("UPDATE notification_outbox SET status = 'retry', attempts = ?, next_attempt_at = ?, last_error = ? WHERE id = ?", (attempts, now + delay, type(error).__name__[:120], notification_id))
+    database.commit()

+# WhatsApp e notificações da v2
+
+## Escopo
+
+O WhatsApp envia somente mensagens úteis para a própria conta: boas-vindas após opt-in e, futuramente, lembretes de prazos dos projetos. O monitor técnico, saúde do banco, disco, fila e gateway nunca envia alertas por este canal.
+
+## Consentimento
+
+O telefone e o opt-in ficam na seção de mensagens úteis do perfil. O telefone é normalizado e nunca aparece em logs. Opt-in sem telefone é rejeitado; retirar o opt-in deixa mensagens pendentes canceladas pelo worker na próxima execução.
+
+## Fila
+
+A tabela `notification_outbox` é idempotente por `idempotency_key`. Estados: `pending`, `processing`, `sent`, `retry` e `cancelled`. A fila guarda destino, texto limitado, tipo e projeto opcional. Ela não guarda segredo do zap-go.
+
+A primeira autorização cria uma mensagem de boas-vindas pendente. Se o gateway estiver indisponível, o worker agenda retry com atraso progressivo; não perde silenciosamente a mensagem nem repete uma chave já enviada.
+
+## Worker
+
+`workers/notification_worker.py` chama o endpoint `/v1/messages` do zap-go com token carregado de arquivo. O transporte é separado da persistência e pode ser substituído por fake nos testes. O worker não importa monitor, saúde operacional ou mensagens administrativas.
+
+Antes da publicação da v2, o worker deverá receber uma unidade systemd própria e um `ZAP_TOKEN_FILE` montado fora do Git. O gateway zap-go continua sendo serviço separado.

+import json
+import os
+from pathlib import Path
+from urllib.error import HTTPError, URLError
+from urllib.request import Request, urlopen
+
+from app import create_app
+from app.store.accounts import get_account
+from app.store.notifications import cancel_notification, claim_notifications, fail_notification, finish_notification
+
+
+def _token() -> str:
+    path = os.environ.get("ZAP_TOKEN_FILE", "")
+    if not path:
+        return ""
+    try:
+        return Path(path).read_text(encoding="utf-8").strip()
+    except OSError:
+        return ""
+
+
+def send_zap(destination: str, text: str, idempotency_key: str, opener=urlopen) -> bool:
+    base_url = os.environ.get("ZAP_URL", "http://127.0.0.1:8787").rstrip("/")
+    token = _token()
+    if not token or not base_url:
+        return False
+    request = Request(
+        base_url if base_url.endswith("/v1/messages") else base_url + "/v1/messages",
+        data=json.dumps({"to": destination, "text": text, "idempotency_key": idempotency_key}).encode(),
+        headers={"Content-Type": "application/json", "Authorization": f"Bearer {token}"},
+        method="POST",
+    )
+    try:
+        with opener(request, timeout=10) as response:
+            return 200 <= getattr(response, "status", 200) < 300
+    except (HTTPError, URLError, OSError, TimeoutError):
+        return False
+
+
+def run_once(app=None, sender=send_zap) -> int:
+    app = app or create_app()
+    sent = 0
+    with app.app_context():
+        for row in claim_notifications():
+            account = get_account(row["account_id"])
+            if not account or not account["whatsapp_opt_in"]:
+                cancel_notification(row["id"])
+                continue
+            if sender(row["destination"], row["text"], row["idempotency_key"]):
+                finish_notification(row["id"])
+                sent += 1
+            else:
+                fail_notification(row["id"], "zap_unavailable")
+    return sent
+
+
+if __name__ == "__main__":
+    run_once()
