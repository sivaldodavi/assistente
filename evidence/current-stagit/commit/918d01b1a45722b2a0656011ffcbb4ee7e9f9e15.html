+from flask import Blueprint, Response, abort, request
+
+from ..auth import current_account_id, login_required
+from ..services.export import ExportUnavailable, build_project_html, render_pdf
+from ..store.projects import get_project
+from ..store.sources import list_sources
+
+
+export_bp = Blueprint("export", __name__)
+
+
+@export_bp.get("/projects/<project_id>/export")
+@login_required
+def project_export(project_id):
+    project = get_project(current_account_id(), project_id)
+    if project is None:
+        abort(404)
+    source_ids = set(project.get("source_ids", []))
+    sources = [source for source in list_sources(current_account_id()) if source["id"] in source_ids]
+    export_format = request.args.get("format", "html")
+    if export_format == "html":
+        return Response(build_project_html(project, sources), mimetype="text/html")
+    if export_format == "pdf":
+        try:
+            pdf = render_pdf(project, sources)
+        except ExportUnavailable as error:
+            abort(503, description=str(error))
+        except Exception:
+            abort(503, description="Não foi possível gerar o PDF neste momento.")
+        return Response(pdf, mimetype="application/pdf", headers={"Content-Disposition": "attachment; filename=assistente-academico.pdf"})
+    abort(400, description="Formato de exportação inválido.")

+    text = str(value or "").strip()
+    if not text:
+        return "<p class=\"empty\">Não preenchido.</p>"

+def _markdown_paragraphs(value: object) -> str:
+    text = str(value or "").strip()

+def build_project_markdown(project: dict, sources: list[dict]) -> str:
+    fields = project.get("fields", {})
+    title = fields.get("title") or project.get("title") or "Trabalho sem título"
+    authors = fields.get("authors") or "Autor não informado"
+    lines = [f"% {title}", f"% {authors}", "", "# Descrição do caso", _markdown_paragraphs(fields.get("facts")), "", "## Pergunta-problema", _markdown_paragraphs(fields.get("problem")), "", "# Decisões possíveis", _markdown_paragraphs(fields.get("decision_basis")), "", "## Alternativa A", _markdown_paragraphs(fields.get("decision_a")), "", "## Alternativa B", _markdown_paragraphs(fields.get("decision_b")), "", "# Critérios e valores"]
+    for item in project.get("criteria", []):
+        lines.extend([f"## {item.get('title', '')}", _markdown_paragraphs(item.get("definition")), f"**Base:** {item.get('basis') or 'Não informada.'}", ""])
+    lines.extend(["# Linha do Tempo dos Fatos", ""])
+    for item in project.get("timeline", []):
+        lines.append(f"- **{item.get('date_value') or 'Data não definida'}** — {item.get('fact') or 'Fato sem resumo'}")
+    lines.extend(["", "# Referências", ""])
+    for source in sources:
+        reference = source.get("reference") or source.get("title") or "Fonte sem título"
+        lines.append(f"- {reference}")
+    return "\n".join(lines) + "\n"
+
+
+def render_pdf(project: dict, sources: list[dict]) -> bytes:
+    quarto = shutil.which("quarto")
+    if not quarto:
+        raise ExportUnavailable("Quarto não está instalado neste ambiente.")
+    with tempfile.TemporaryDirectory(prefix="assistente-v2-export-") as directory:
+        root = Path(directory)
+        source = root / "project.qmd"
+        output = root / "project.pdf"
+        source.write_text(build_project_markdown(project, sources), encoding="utf-8")
+        subprocess.run(
+            [quarto, "render", str(source), "--to", "pdf", "--output", output.name],
+            cwd=root,
+            check=True,
+            capture_output=True,
+            text=True,
+            timeout=90,
+        )
+        return output.read_bytes()

+# Revisão e exportação
+
+## Prévia HTML
+
+A rota autenticada de prévia reúne o snapshot salvo do projeto: identificação, descrição, decisões, critérios, Linha do Tempo e fontes vinculadas à conta. O conteúdo é escapado antes de entrar no HTML. A prévia não lê o banco legado nem informações de outras contas.
+
+A prévia é uma revisão, não uma confirmação de qualidade acadêmica. O usuário precisa conferir citações, formatação, completude e adequação à atividade.
+
+## PDF
+
+O PDF é produzido por um adaptador para Quarto. O editor não chama subprocessos diretamente e não assume que Quarto esteja instalado em qualquer ambiente.
+
+Quando Quarto está disponível, o adaptador cria diretório temporário, gera uma fonte Markdown a partir do snapshot e lê o PDF resultante. O diretório temporário é descartado ao final. Quando Quarto não está disponível ou falha, a rota retorna indisponibilidade explícita; nunca devolve HTML com `application/pdf` falso.
+
+## Segurança
+
+A exportação exige sessão e propriedade do projeto. Fontes são filtradas pelos IDs pertencentes à conta antes de entrar no documento. O nome do arquivo é fixo e não é derivado de entrada do usuário.
+
+Exportar não marca entrega, não compartilha o projeto e não envia conteúdo a provedor externo. Uma futura integração de PDF precisa preservar essas propriedades e ganhar teste de contrato do comando executado.
