+def paper_sections(project_type: str) -> list[dict]:
+    return PAPER_FINAL_SECTIONS if project_type == "paper_final" else PAPER_PROJECT_SECTIONS
+
+
+def _rules(project_type: str) -> dict:
+    return {field["id"]: field for section in paper_sections(project_type) for field in section["fields"]}
+
+
+def normalize_paper_fields(raw: object, project_type: str) -> dict:
+    source = raw if isinstance(raw, dict) else {}
+    normalized = {}
+    for field_id, rule in _rules(project_type).items():
+        value = source.get(field_id, "")
+        if not isinstance(value, str):
+            value = str(value or "")
+        normalized[field_id] = value.strip()[: rule["limit"]]
+    return normalized
+
+
+def paper_progress(fields: object, project_type: str) -> dict:
+    normalized = normalize_paper_fields(fields, project_type)
+    required = [field_id for field_id, rule in _rules(project_type).items() if rule.get("required")]
+    done = sum(bool(normalized.get(field_id)) for field_id in required)
+    return {"done": done, "total": len(required), "percent": round(done / len(required) * 100) if required else 100}

+# Projeto de Paper e Paper Final
+
+## Distinção
+
+`paper_project` é o planejamento investigável: delimitação, problema, hipótese, justificativa, objetivos, referencial teórico e metodologia.
+
+`paper_final` é a redação final: identificação, resumo, palavras-chave, introdução, desenvolvimento e conclusão. O Paper Final é outro percurso do mesmo tipo de trabalho, mas seus campos não substituem silenciosamente os campos do Projeto.
+
+## Regras
+
+Cada tipo tem seu próprio conjunto de campos obrigatórios e progresso. O progresso mede preenchimento, não qualidade. Fontes vinculadas continuam na Biblioteca Pessoal e no snapshot do projeto.
+
+A Linha do Tempo é opcional como apoio e não vira narrativa automática. Exportação só reúne o conteúdo salvo; não inventa resultado de pesquisa nem converte uma hipótese em conclusão.
+
+## Evolução
+
+Desenvolvimento em tópicos, citações por fonte, avanço do Projeto para o Paper Final e revisão assistida serão adicionados sobre estes campos, sem reintroduzir um editor monolítico ou copiar o banco legado.
